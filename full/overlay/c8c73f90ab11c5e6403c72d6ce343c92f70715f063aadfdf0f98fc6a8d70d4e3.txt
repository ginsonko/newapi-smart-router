import assert from 'node:assert/strict'
import { describe, test } from 'node:test'

import type { SmartRouterSetting } from '@/features/smart-routing'

import {
  createEmptySmartRouterFormValues,
  smartRouterFormSchema,
  smartRouterFormToSetting,
  smartRouterSettingToForm,
} from './smart-routing-form'

const setting: SmartRouterSetting = {
  enabled: true,
  default_for_new_keys: true,
  discovery: {
    allowed_text_endpoints: ['openai', 'openai-response', 'anthropic'],
    exclude_virtual_groups: ['auto'],
  },
  channel_profiles: {
    '11': {
      cache_domain: 'cache-a',
      capacity_domain: 'capacity-a',
      max_inflight: 4,
    },
  },
  default_policy: {
    version: 4,
    enabled: true,
    strategy: 'price',
    excluded_groups: [],
    allow_future_groups: true,
    manual_group_order: [],
    max_attempts: 3,
    failure_threshold: 3,
    max_effective_ratio_ppm: 0,
    health_guard: true,
    actual_input_cost_ranking: true,
    recovery_profile: 'balanced',
    affinity_enabled: true,
    affinity_mode: 'economic_break_even',
    affinity_max_premium_percent: 10,
    affinity_ttl_seconds: 1800,
    compression_thresholds: [],
    queue_policy: {
      mode: 'no_queue',
      max_wait_ms: 0,
      min_saving_percent: 20,
    },
    ttft_policy: {
      enabled: false,
      target_ms: 1500,
      hard_max_ms: 8000,
      min_samples: 20,
    },
    balanced_weights: { stability: 45, price: 40, ttft: 15 },
    custom_weights: { cache: 25, cost: 25, ttft: 25, stability: 25 },
  },
  default_max_effective_ratio_ppm: 1_000_000,
  absolute_max_effective_ratio_ppm: 10_000_000,
  attempt_limit_enabled: false,
  recovery_fallback_enabled: true,
  max_attempts: 8,
  max_failure_threshold: 10,
  hard_failure_window_seconds: 600,
  catalog_refresh_seconds: 30,
  probe_initial_backoff_seconds: 60,
  probe_max_backoff_seconds: 60,
  max_probe_failures: 8,
  max_outage_seconds: 86400,
  probe_budget_per_minute: 30,
  probe_max_concurrency: 3,
  probe_lease_ttl_seconds: 10,
  background_probe_enabled: true,
  probe_timeout_seconds: 45,
  probe_completion_grace_seconds: 15,
  probe_scan_interval_ms: 5000,
  probe_queue_capacity: 2048,
  recovery_fast_interval_seconds: 60,
  recovery_fast_window_seconds: 3600,
  recovery_sweep_interval_seconds: 30,
  recovery_canary_initial_traffic_ppm: 250_000,
  recovery_canary_healthy_samples: 3,
  recovery_canary_fast_history_samples: 5,
  recovery_canary_fast_reliability_ppm: 800_000,
  recovery_coverage_target: 1,
  recovery_backup_evidence_ttl_seconds: 3600,
  cold_start_wave_batch_size: 2,
  cold_start_wave_max_concurrency: 2,
  cold_start_probe_budget_per_minute: 60,
  cold_start_wait_ms: 60000,
  cold_start_poll_interval_ms: 100,
  max_queue_wait_ms: 1000,
  kill_switch_mode: 'normal',
}

describe('v4 smart routing settings form', () => {
  test('round-trips safety settings while presenting ratios as multipliers', () => {
    const form = smartRouterSettingToForm(setting)
    assert.equal(form.default_max_effective_ratio, 1)
    assert.equal(form.absolute_max_effective_ratio, 10)
    assert.equal(form.actual_input_cost_ranking, true)
    assert.deepEqual(smartRouterFormToSetting(form), setting)
  })

  test('preserves explicit recovery fallback disable and defaults old settings to enabled', () => {
    const form = smartRouterSettingToForm({ ...setting, recovery_fallback_enabled: false })
    assert.equal(form.recovery_fallback_enabled, false)
    assert.equal(smartRouterFormToSetting(form).recovery_fallback_enabled, false)
    assert.equal(smartRouterSettingToForm({ ...setting, recovery_fallback_enabled: undefined }).recovery_fallback_enabled, true)
  })

  test('allows enabling automatic discovery without certified groups or contracts', () => {
    const form = createEmptySmartRouterFormValues()
    form.enabled = true
    const result = smartRouterFormSchema.safeParse(form)
    assert.equal(result.success, true)
    const rebuilt = smartRouterFormToSetting(form)
    assert.equal(rebuilt.enabled, true)
    assert.equal(rebuilt.certified_groups, undefined)
    assert.equal(rebuilt.text_contracts, undefined)
  })

  test('keeps an unconfigured disabled control plane valid', () => {
    assert.equal(
      smartRouterFormSchema.safeParse(createEmptySmartRouterFormValues())
        .success,
      true
    )
  })

  test('fills legacy documents with current hot defaults before saving', () => {
    const legacy = structuredClone(setting)
    delete legacy.probe_max_backoff_seconds
    delete legacy.attempt_limit_enabled
    delete legacy.probe_scan_interval_ms
    delete legacy.background_probe_enabled
    delete legacy.cold_start_wave_batch_size
    delete legacy.cold_start_wave_max_concurrency
    delete legacy.cold_start_probe_budget_per_minute
    delete legacy.cold_start_wait_ms
    delete legacy.cold_start_poll_interval_ms

    const form = smartRouterSettingToForm(legacy)
    assert.equal(form.probe_max_backoff_seconds, 60)
    assert.equal(form.attempt_limit_enabled, false)
    assert.equal(form.probe_scan_interval_ms, 5000)
    assert.equal(form.background_probe_enabled, true)
    assert.equal(form.cold_start_wave_batch_size, 2)
    assert.equal(form.cold_start_wave_max_concurrency, 2)
    assert.equal(form.cold_start_probe_budget_per_minute, 60)
    assert.equal(form.cold_start_wait_ms, 60000)
    assert.equal(form.cold_start_poll_interval_ms, 100)

    const rebuilt = smartRouterFormToSetting(form)
    assert.equal(rebuilt.probe_max_backoff_seconds, 60)
    assert.equal(rebuilt.attempt_limit_enabled, false)
    assert.equal(rebuilt.probe_scan_interval_ms, 5000)
    assert.equal(rebuilt.background_probe_enabled, true)
    assert.equal(rebuilt.cold_start_wave_batch_size, 2)
    assert.equal(rebuilt.cold_start_wave_max_concurrency, 2)
    assert.equal(rebuilt.cold_start_probe_budget_per_minute, 60)
    assert.equal(rebuilt.cold_start_wait_ms, 60000)
    assert.equal(rebuilt.cold_start_poll_interval_ms, 100)
    assert.equal(form.actual_input_cost_ranking, true)
  })

  test('preserves an explicit actual-input-cost ordering opt-out', () => {
    const form = smartRouterSettingToForm(setting)
    form.actual_input_cost_ranking = false
    const rebuilt = smartRouterFormToSetting(form)
    assert.equal(rebuilt.default_policy.actual_input_cost_ranking, false)
  })

  test('requires the engine before making smart routing the new-key default', () => {
    const form = createEmptySmartRouterFormValues()
    form.default_for_new_keys = true
    assert.equal(smartRouterFormSchema.safeParse(form).success, false)
  })

  test('validates cost and recovery budget boundaries', () => {
    const cost = smartRouterSettingToForm(setting)
    cost.default_max_effective_ratio = 2
    cost.absolute_max_effective_ratio = 1
    assert.equal(smartRouterFormSchema.safeParse(cost).success, false)

    const backoff = smartRouterSettingToForm(setting)
    backoff.probe_initial_backoff_seconds = 120
    backoff.probe_max_backoff_seconds = 60
    assert.equal(smartRouterFormSchema.safeParse(backoff).success, false)

    const budget = smartRouterSettingToForm(setting)
    budget.background_probe_enabled = true
    budget.probe_budget_per_minute = 0
    assert.equal(smartRouterFormSchema.safeParse(budget).success, false)

    const outage = smartRouterSettingToForm(setting)
    outage.probe_initial_backoff_seconds = 3600
    outage.max_outage_seconds = 60
    assert.equal(smartRouterFormSchema.safeParse(outage).success, false)

    const coldStartConcurrency = smartRouterSettingToForm(setting)
    coldStartConcurrency.cold_start_wave_batch_size = 2
    coldStartConcurrency.cold_start_wave_max_concurrency = 3
    assert.equal(
      smartRouterFormSchema.safeParse(coldStartConcurrency).success,
      false
    )

    const coldStartPolling = smartRouterSettingToForm(setting)
    coldStartPolling.cold_start_wait_ms = 500
    coldStartPolling.cold_start_poll_interval_ms = 1000
    assert.equal(
      smartRouterFormSchema.safeParse(coldStartPolling).success,
      false
    )
  })

  test('preserves unknown fields while dropping obsolete v3 publishing lists', () => {
    const legacy = structuredClone(setting) as SmartRouterSetting & {
      future_control_flag: string
    }
    legacy.future_control_flag = 'preserve-me'
    legacy.certified_groups = {
      old: { pools: ['old'], stable_fallback: true },
    }
    legacy.text_contracts = [
      { endpoint: 'openai', capabilities: 1, max_context: 128000 },
    ]
    legacy.default_policy = {
      ...legacy.default_policy,
      version: 3,
      strategy: 'price',
      order_mode: 'auto_price',
      allow_future_certified_routes: true,
    }

    const rebuilt = smartRouterFormToSetting(
      smartRouterSettingToForm(legacy)
    ) as SmartRouterSetting & { future_control_flag?: string }
    assert.equal(rebuilt.future_control_flag, 'preserve-me')
    assert.equal(rebuilt.certified_groups, undefined)
    assert.equal(rebuilt.text_contracts, undefined)
    assert.equal(rebuilt.default_policy.version, 4)
    assert.equal(rebuilt.default_policy.strategy, 'price')
  })
})
