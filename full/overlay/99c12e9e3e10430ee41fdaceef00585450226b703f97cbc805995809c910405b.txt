var commissionRuntimeRole = normalizeCommissionRuntimeRole(os.Getenv("COMMISSION_RUNTIME_ROLE"))

func normalizeCommissionRuntimeRole(value string) string {
	if strings.EqualFold(strings.TrimSpace(value), CommissionRoleCaptureCandidate) {
		return CommissionRoleCaptureCandidate
	}
	return CommissionRoleLegacy
}

func CommissionRuntimeRole() string { return commissionRuntimeRole }

var commissionMigrationModels = []interface{}{
	&CommissionSettingsV1{},
	&CommissionProgramV1{},
	&CommissionProfileV1{},
	&CommissionAccountV1{},
	&CommissionOperationV1{},
	&CommissionLedgerV1{},
	&CommissionWithdrawalV1{},
}

func migrateCommissionSchema() error {
	if err := DB.AutoMigrate(commissionMigrationModels...); err != nil {
		return fmt.Errorf("migrate commission schema: %w", err)
	}
	now := time.Now().UnixMilli()
	settings := CommissionSettingsV1{
		ID:            1,
		SchemaVersion: CommissionSchemaVersion,
		Mode:          CommissionModeDark,
		SpendPriority: CommissionSpendFirst,
		Version:       1,
		CreatedAt:     now,
		UpdatedAt:     now,
	}
	if err := DB.Clauses(clause.OnConflict{DoNothing: true}).Create(&settings).Error; err != nil {
		return fmt.Errorf("initialize commission settings: %w", err)
	}
	return validateCommissionSchemaReadiness(DB)
}

func validateCommissionSchemaReadiness(db *gorm.DB) error {
	if db == nil {
		return errors.New("database is not initialized")
	}
	for _, item := range commissionMigrationModels {
		if !db.Migrator().HasTable(item) {
			return fmt.Errorf("commission schema table missing for %T", item)
		}
	}
	if !db.Migrator().HasColumn(&CommissionOperationV1{}, "ReservationCorrections") {
		return errors.New("commission reservation corrections column is missing; run the deployment migration")
	}
	var settings CommissionSettingsV1
	if err := db.First(&settings, 1).Error; err != nil {
		return err
	}
	if settings.SchemaVersion != CommissionSchemaVersion {
		return fmt.Errorf("unsupported commission schema version %d", settings.SchemaVersion)
	}
	if !validCommissionMode(settings.Mode) || !validCommissionSpendPriority(settings.SpendPriority) {
		return ErrCommissionInvariant
	}
	return nil
}

func validCommissionMode(mode string) bool {
	switch mode {
	case CommissionModeDark, CommissionModeCapture, CommissionModeActive, CommissionModeSuspended:
		return true
	default:
		return false
	}
}

func validCommissionSpendPriority(priority string) bool {
	return priority == CommissionSpendFirst || priority == CommissionOrdinaryFirst
}

func GetCommissionSettings() (*CommissionSettingsV1, error) {
	var settings CommissionSettingsV1
	if err := DB.First(&settings, 1).Error; err != nil {
		return nil, err
	}
	return &settings, nil
}

func CommissionBillingMode() (string, string, error) {
	settings, err := GetCommissionSettings()
	if err != nil {
		return "", "", err
	}
	return settings.Mode, settings.SpendPriority, nil
}

func CommissionBillableAdmission() (bool, error) {
	settings, err := GetCommissionSettings()
	if err != nil {
		if CommissionRuntimeRole() == CommissionRoleCaptureCandidate {
			return false, ErrCommissionCaptureNotArmed
		}
		return false, nil
	}
	if settings.SchemaVersion != CommissionSchemaVersion || !validCommissionMode(settings.Mode) {
		return false, ErrCommissionInvariant
	}
	return settings.CurrentProgramID > 0, nil
}

func ValidateLegacyMidjourneyCommissionAdmission() error {
	_, err := CommissionBillableAdmission()
	return err
}

func commissionSchemaAvailableTx(tx *gorm.DB) bool {
	return tx != nil && tx.Migrator().HasTable(&CommissionSettingsV1{}) && tx.Migrator().HasTable(&CommissionOperationV1{})
}

func commissionHash(parts ...string) string {
	h := sha256.New()
	for _, part := range parts {
		_, _ = h.Write([]byte(part))
		_, _ = h.Write([]byte{0})
	}
	return hex.EncodeToString(h.Sum(nil))
}

func commissionPublicID(prefix string) string {
	return prefix + "_" + common.GetUUID()
}

func commissionCheckedQuota(value int64) (int, error) {
	if value < int64(common.MinQuota) || value > int64(common.MaxQuota) {
		return 0, ErrCommissionInvariant
	}
	return int(value), nil
}

// A request-level quota is capped at common.MaxQuota, but a user's wallet may
// legitimately exceed that after multiple grants or commission transfers.
func commissionCheckedWalletBalance(value int64) (int, error) {
	maxInt := int64(^uint(0) >> 1)
	if value < 0 || value > maxInt {
		return 0, ErrCommissionInvariant
	}
	return int(value), nil
}

func commissionSafeAdd(left, right int64) (int64, error) {
	if (right > 0 && left > math.MaxInt64-right) || (right < 0 && left < math.MinInt64-right) {
		return 0, ErrCommissionInvariant
	}
	return left + right, nil
}

type commissionTierV1 struct {
	ThresholdQuota int64 `json:"threshold_quota"`
	RateBps        int64 `json:"rate_bps"`
}

func normalizeCommissionTierJSON(tiersJSON string) ([]commissionTierV1, string, error) {
	var tiers []commissionTierV1
	if err := common.Unmarshal([]byte(tiersJSON), &tiers); err != nil {
		return nil, "", err
	}
	if len(tiers) == 0 {
		return nil, "", errors.New("at least one commission tier is required")
	}
	sort.Slice(tiers, func(i, j int) bool { return tiers[i].ThresholdQuota < tiers[j].ThresholdQuota })
	if tiers[0].ThresholdQuota != 0 {
		return nil, "", errors.New("the first commission threshold must be zero")
	}
	for i, tier := range tiers {
		if tier.ThresholdQuota < 0 || tier.RateBps < 0 || tier.RateBps > commissionRateBase {
			return nil, "", errors.New("commission tier is outside the supported range")
		}
		if i > 0 && tiers[i-1].ThresholdQuota == tier.ThresholdQuota {
			return nil, "", errors.New("commission thresholds must be unique")
		}
	}
	canonical, err := common.Marshal(tiers)
	if err != nil {
		return nil, "", err
	}
	return tiers, string(canonical), nil
}
