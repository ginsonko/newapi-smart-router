type TaskStatus string

func (t TaskStatus) ToVideoStatus() string {
	var status string
	switch t {
	case TaskStatusQueued, TaskStatusSubmitted:
		status = dto.VideoStatusQueued
	case TaskStatusInProgress:
		status = dto.VideoStatusInProgress
	case TaskStatusFinalizing:
		status = dto.VideoStatusInProgress
	case TaskStatusSuccess:
		status = dto.VideoStatusCompleted
	case TaskStatusFailure:
		status = dto.VideoStatusFailed
	default:
		status = dto.VideoStatusUnknown // Default fallback
	}
	return status
}

const (
	TaskStatusNotStart   TaskStatus = "NOT_START"
	TaskStatusSubmitted             = "SUBMITTED"
	TaskStatusQueued                = "QUEUED"
	TaskStatusInProgress            = "IN_PROGRESS"
	TaskStatusFinalizing            = "FINALIZING"
	TaskStatusFailure               = "FAILURE"
	TaskStatusSuccess               = "SUCCESS"
	TaskStatusUnknown               = "UNKNOWN"
)

type Task struct {
	ID        int64                 `json:"id" gorm:"primary_key;AUTO_INCREMENT"`
	CreatedAt int64                 `json:"created_at" gorm:"index"`
	UpdatedAt int64                 `json:"updated_at"`
	TaskID    string                `json:"task_id" gorm:"type:varchar(191);index"` // 第三方id，不一定有/ song id\ Task id
	Platform  constant.TaskPlatform `json:"platform" gorm:"type:varchar(30);index"` // 平台
	UserId    int                   `json:"user_id" gorm:"index"`
	Group     string                `json:"group" gorm:"type:varchar(50)"` // 修正计费用
	ChannelId int                   `json:"channel_id" gorm:"index"`
	Quota     int                   `json:"quota"`
	// BillingReservationID associates an async task with the already-created
	// durable wallet reservation. It is metadata only; task billing continues
	// to use its existing lifecycle and the commission worker waits for a
	// terminal task status before projecting a reward.
	BillingReservationID string     `json:"-" gorm:"type:varchar(128);index"`
	ReconcileSince       int64      `json:"billing_reconcile_since,omitempty" gorm:"not null;default:0"`
	NextPollAt           int64      `json:"-" gorm:"not null;default:0;index"`
	Action               string     `json:"action" gorm:"type:varchar(40);index"` // 任务类型, song, lyrics, description-mode
	Status               TaskStatus `json:"status" gorm:"type:varchar(20);index"` // 任务状态
	FailReason           string     `json:"fail_reason"`
	SubmitTime           int64      `json:"submit_time" gorm:"index"`
	StartTime            int64      `json:"start_time" gorm:"index"`
	FinishTime           int64      `json:"finish_time" gorm:"index"`
	Progress             string     `json:"progress" gorm:"type:varchar(20);index"`
	Properties           Properties `json:"properties" gorm:"type:json"`
	Username             string     `json:"username,omitempty" gorm:"-"`
	// 禁止返回给用户，内部可能包含key等隐私信息
	PrivateData TaskPrivateData `json:"-" gorm:"column:private_data;type:json"`
	Data        json.RawMessage `json:"data" gorm:"type:json"`
}

func (t *Task) SetData(data any) {
	b, _ := common.Marshal(data)
	t.Data = json.RawMessage(b)
}

func (t *Task) GetData(v any) error {
	return common.Unmarshal(t.Data, &v)
}

type Properties struct {
	Input               string `json:"input"`
	UpstreamModelName   string `json:"upstream_model_name,omitempty"`
	OriginModelName     string `json:"origin_model_name,omitempty"`
	MediaMode           string `json:"media_mode,omitempty"`
	MediaSize           string `json:"media_size,omitempty"`
	MediaDuration       int    `json:"media_duration,omitempty"`
	MediaReferenceCount int    `json:"media_reference_count,omitempty"`
}

func (m *Properties) Scan(val interface{}) error {
	bytesValue, _ := val.([]byte)
	if len(bytesValue) == 0 {
		*m = Properties{}
		return nil
	}
	return common.Unmarshal(bytesValue, m)
}

func (m Properties) Value() (driver.Value, error) {
	if m == (Properties{}) {
		return nil, nil
	}
	return common.Marshal(m)
}

// SyncTaskQueryParams 用于包含所有搜索条件的结构体，可以根据需求添加更多字段
type SyncTaskQueryParams struct {
	Platform       constant.TaskPlatform
	ChannelID      string
	TaskID         string
	UserID         string
	Action         string
	Status         string
	StartTimestamp int64
	EndTimestamp   int64
	UserIDs        []int
}

// TaskInitParams is the persistence-layer projection needed to create a task.
// Keeping relay request types out of model preserves the package dependency
// direction and lets relay/common use task media persistence safely.
type TaskInitParams struct {
	UserID            int
	Group             string
	ChannelID         int
	ChannelType       int
	ChannelAPIKey     string
	UpstreamModelName string
	OriginModelName   string
	PublicTaskID      string
}

func InitTask(platform constant.TaskPlatform, params TaskInitParams) *Task {
	properties := Properties{}
	privateData := TaskPrivateData{}
	if params.ChannelAPIKey != "" {
		privateData.Key = params.ChannelAPIKey
	}
	properties.UpstreamModelName = params.UpstreamModelName
	properties.OriginModelName = params.OriginModelName

	// 使用预生成的公开 ID（如果有），否则新生成
	taskID := params.PublicTaskID
	if taskID == "" {
		taskID = GenerateTaskID()
	}

	t := &Task{
		TaskID:      taskID,
		UserId:      params.UserID,
		Group:       params.Group,
		SubmitTime:  time.Now().Unix(),
		Status:      TaskStatusNotStart,
		Progress:    "0%",
		ChannelId:   params.ChannelID,
		Platform:    platform,
		Properties:  properties,
		PrivateData: privateData,
	}
	return t
}

func TaskGetAllUserTask(userId int, startIdx int, num int, queryParams SyncTaskQueryParams) []*Task {
	var tasks []*Task
	var err error

	// 初始化查询构建器
	query := DB.Where("user_id = ?", userId)

	if queryParams.TaskID != "" {
		query = query.Where("task_id = ?", queryParams.TaskID)
	}
	if queryParams.Action != "" {
		query = query.Where("action = ?", queryParams.Action)
	}
	if queryParams.Status != "" {
		query = query.Where("status = ?", queryParams.Status)
	}
	if queryParams.Platform != "" {
		query = query.Where("platform = ?", queryParams.Platform)
	}
	if queryParams.StartTimestamp != 0 {
		// 假设您已将前端传来的时间戳转换为数据库所需的时间格式，并处理了时间戳的验证和解析
		query = query.Where("submit_time >= ?", queryParams.StartTimestamp)
	}
	if queryParams.EndTimestamp != 0 {
		query = query.Where("submit_time <= ?", queryParams.EndTimestamp)
	}

	// 获取数据
	err = query.Omit("channel_id").Order("id desc").Limit(num).Offset(startIdx).Find(&tasks).Error
	if err != nil {
		return nil
	}

	return tasks
}

// ListUserMediaTasks returns user-owned async media tasks without exposing
// channel credentials or mixing in legacy Suno and Midjourney task surfaces.
func ListUserMediaTasks(userID int, startIdx int, limit int) ([]Task, int64, error) {
	if userID <= 0 || startIdx < 0 || limit <= 0 {
		return nil, 0, fmt.Errorf("invalid media task list range")
	}
	query := DB.Model(&Task{}).
		Where("user_id = ?", userID).
		Where("task_id <> ?", "").
		Where("platform NOT IN ?", []constant.TaskPlatform{constant.TaskPlatformSuno, constant.TaskPlatformMidjourney})
	var total int64
	if err := query.Count(&total).Error; err != nil {
		return nil, 0, err
	}
	var tasks []Task
	if err := query.Order("id DESC").Offset(startIdx).Limit(limit).Find(&tasks).Error; err != nil {
		return nil, 0, err
	}
	return tasks, total, nil
}

// MediaTaskTimelineItem is the common history identity for asynchronous Task
// rows and synchronous Studio outputs archived directly as MediaBlob rows.
// Pagination is applied after both sources are merged, so new synchronous
// images cannot disappear behind an already-paginated list of old videos.
type MediaTaskTimelineItem struct {
	PublicID  string
	Kind      string
	CreatedAt int64
	Task      *Task
	Blobs     []MediaBlob
}

func ListUserMediaTaskTimeline(userID int, startIdx int, limit int) ([]MediaTaskTimelineItem, int64, error) {
	if userID <= 0 || startIdx < 0 || limit <= 0 || startIdx > int(^uint(0)>>1)-limit {
		return nil, 0, fmt.Errorf("invalid media task timeline range")
	}
	needed := startIdx + limit
	legacyPlatforms := []constant.TaskPlatform{constant.TaskPlatformSuno, constant.TaskPlatformMidjourney}
	taskQuery := DB.Model(&Task{}).
		Where("user_id = ?", userID).
		Where("task_id <> ?", "").
		Where("platform NOT IN ?", legacyPlatforms)
	var taskTotal int64
	if err := taskQuery.Count(&taskTotal).Error; err != nil {
		return nil, 0, err
	}
	var tasks []Task
	if err := taskQuery.Order("created_at DESC, id DESC").Limit(needed).Find(&tasks).Error; err != nil {
		return nil, 0, err
	}

	studioQuery := DB.Model(&MediaBlob{}).
		Where("media_blobs.user_id = ? AND media_blobs.deleted_at = 0 AND media_blobs.status = ?", userID, MediaBlobStatusReady).
		Where("media_blobs.task_public_id LIKE ?", "studio_%").
		Where(`NOT EXISTS (
			SELECT 1 FROM tasks
			WHERE tasks.user_id = media_blobs.user_id
			  AND tasks.task_id = media_blobs.task_public_id
			  AND tasks.task_id <> ''
			  AND tasks.platform NOT IN ?
		)`, legacyPlatforms)
	var studioTotal int64
	if err := studioQuery.Select("COUNT(DISTINCT media_blobs.task_public_id)").Scan(&studioTotal).Error; err != nil {
		return nil, 0, err
	}
	type studioTimelineRow struct {
		TaskPublicID string `gorm:"column:task_public_id"`
		CreatedAt    int64  `gorm:"column:created_at"`
	}
	var studioRows []studioTimelineRow
	if err := studioQuery.
		Select("media_blobs.task_public_id, MAX(media_blobs.created_at) AS created_at").
		Group("media_blobs.task_public_id").
		Order("created_at DESC, media_blobs.task_public_id DESC").
		Limit(needed).Scan(&studioRows).Error; err != nil {
		return nil, 0, err
	}

	operationIDs := make([]string, 0, len(studioRows))
	for _, row := range studioRows {
		operationIDs = append(operationIDs, row.TaskPublicID)
	}
	blobsByOperation := make(map[string][]MediaBlob, len(operationIDs))
	if len(operationIDs) > 0 {
		var blobs []MediaBlob
		if err := DB.Where("user_id = ? AND deleted_at = 0 AND status = ? AND task_public_id IN ?", userID, MediaBlobStatusReady, operationIDs).
			Order("created_at ASC, id ASC").Find(&blobs).Error; err != nil {
			return nil, 0, err
		}
		for index := range blobs {
			blob := blobs[index]
			blobsByOperation[blob.TaskPublicID] = append(blobsByOperation[blob.TaskPublicID], blob)
		}
	}

	timeline := make([]MediaTaskTimelineItem, 0, len(tasks)+len(studioRows))
	for index := range tasks {
		task := tasks[index]
		createdAt := task.CreatedAt
		if createdAt == 0 {
			createdAt = task.SubmitTime
		}
		timeline = append(timeline, MediaTaskTimelineItem{
			PublicID: task.TaskID, Kind: "task", CreatedAt: createdAt, Task: &task,
		})
	}
	for _, row := range studioRows {
		timeline = append(timeline, MediaTaskTimelineItem{
			PublicID: row.TaskPublicID, Kind: "studio_blob", CreatedAt: row.CreatedAt,
			Blobs: blobsByOperation[row.TaskPublicID],
		})
	}
	sort.SliceStable(timeline, func(left, right int) bool {
		if timeline[left].CreatedAt == timeline[right].CreatedAt {
			return timeline[left].PublicID > timeline[right].PublicID
		}
		return timeline[left].CreatedAt > timeline[right].CreatedAt
	})
	if startIdx >= len(timeline) {
		return []MediaTaskTimelineItem{}, taskTotal + studioTotal, nil
	}
	end := startIdx + limit
	if end > len(timeline) {
		end = len(timeline)
	}
	return timeline[startIdx:end], taskTotal + studioTotal, nil
}

func TaskGetAllTasks(startIdx int, num int, queryParams SyncTaskQueryParams) []*Task {
	var tasks []*Task
	var err error

	// 初始化查询构建器
	query := DB

	// 添加过滤条件
	if queryParams.ChannelID != "" {
		query = query.Where("channel_id = ?", queryParams.ChannelID)
	}
	if queryParams.Platform != "" {
		query = query.Where("platform = ?", queryParams.Platform)
	}
	if queryParams.UserID != "" {
		query = query.Where("user_id = ?", queryParams.UserID)
	}
	if len(queryParams.UserIDs) != 0 {
		query = query.Where("user_id in (?)", queryParams.UserIDs)
	}
	if queryParams.TaskID != "" {
		query = query.Where("task_id = ?", queryParams.TaskID)
	}
	if queryParams.Action != "" {
		query = query.Where("action = ?", queryParams.Action)
	}
	if queryParams.Status != "" {
		query = query.Where("status = ?", queryParams.Status)
	}
	if queryParams.StartTimestamp != 0 {
		query = query.Where("submit_time >= ?", queryParams.StartTimestamp)
	}
	if queryParams.EndTimestamp != 0 {
		query = query.Where("submit_time <= ?", queryParams.EndTimestamp)
	}

	// 获取数据
	err = query.Order("id desc").Limit(num).Offset(startIdx).Find(&tasks).Error
	if err != nil {
		return nil
	}

	return tasks
}

// HasUnfinishedSyncTasks reports whether at least one async (Suno/video) task is
// still in progress. It is a cheap existence check (LIMIT 1) used to decide
// whether the async_task_poll system task needs to run; when no task is pending
// the scheduler skips creating a row entirely.
func HasUnfinishedSyncTasks() bool {
	var id int64
	err := DB.Model(&Task{}).
		Where("platform <> ?", "smart_image").
		Where("progress != ?", "100%").
		Where("status != ?", TaskStatusFailure).
		Where("status != ?", TaskStatusSuccess).
		Where("next_poll_at <= ?", time.Now().Unix()).
		Limit(1).
		Pluck("id", &id).Error
	return err == nil && id != 0
}

type taskSnapshot struct {
	Status     TaskStatus
	Progress   string
	StartTime  int64
	FinishTime int64
	FailReason string
	ResultURL  string
	Data       json.RawMessage
}

func (s taskSnapshot) Equal(other taskSnapshot) bool {
	return s.Status == other.Status &&
		s.Progress == other.Progress &&
		s.StartTime == other.StartTime &&
		s.FinishTime == other.FinishTime &&
		s.FailReason == other.FailReason &&
		s.ResultURL == other.ResultURL &&
		bytes.Equal(s.Data, other.Data)
}

func (t *Task) Snapshot() taskSnapshot {
	return taskSnapshot{
		Status:     t.Status,
		Progress:   t.Progress,
		StartTime:  t.StartTime,
		FinishTime: t.FinishTime,
		FailReason: t.FailReason,
		ResultURL:  t.PrivateData.ResultURL,
		Data:       t.Data,
	}
}

// UpdateWithStatus performs a conditional UPDATE guarded by fromStatus (CAS).
// Returns (true, nil) if this caller won the update, (false, nil) if
// another process already moved the task out of fromStatus.
//
// Uses Model().Select("*").Updates() instead of Save() because GORM's Save
// falls back to INSERT ON CONFLICT when the WHERE-guarded UPDATE matches
// zero rows, which silently bypasses the CAS guard.
func (t *Task) UpdateWithStatus(fromStatus TaskStatus) (bool, error) {
	// Scheduling is updated separately; stale poll results must not reset it.
	result := DB.Model(t).Where("status = ?", fromStatus).Select("*").Omit("ReconcileSince", "NextPollAt").Updates(t)
	if result.Error != nil {
		return false, result.Error
	}
	return result.RowsAffected > 0, nil
}

// TaskBulkUpdate performs an unconditional bulk UPDATE by upstream task_id strings.
// Same caveats as TaskBulkUpdateByID — no CAS guard.
func TaskBulkUpdate(taskIds []string, params map[string]any) error {
	if len(taskIds) == 0 {
		return nil
	}
	return DB.Model(&Task{}).
		Where("task_id in (?)", taskIds).
		Updates(params).Error
}

// TaskBulkUpdateByID performs an unconditional bulk UPDATE by primary key IDs.
// WARNING: This function has NO CAS (Compare-And-Swap) guard — it will overwrite
// any concurrent status changes. DO NOT use in billing/quota lifecycle flows
// (e.g., timeout, success, failure transitions that trigger refunds or settlements).
// For status transitions that involve billing, use Task.UpdateWithStatus() instead.
func TaskBulkUpdateByID(ids []int64, params map[string]any) error {
	if len(ids) == 0 {
		return nil
	}
	return DB.Model(&Task{}).
		Where("id in (?)", ids).
		Updates(params).Error
}

// TaskCountAllTasks returns total tasks that match the given query params (admin usage)
func TaskCountAllTasks(queryParams SyncTaskQueryParams) int64 {
	var total int64
	query := DB.Model(&Task{})
	if queryParams.ChannelID != "" {
		query = query.Where("channel_id = ?", queryParams.ChannelID)
	}
	if queryParams.Platform != "" {
		query = query.Where("platform = ?", queryParams.Platform)
	}
	if queryParams.UserID != "" {
		query = query.Where("user_id = ?", queryParams.UserID)
	}
	if len(queryParams.UserIDs) != 0 {
		query = query.Where("user_id in (?)", queryParams.UserIDs)
	}
	if queryParams.TaskID != "" {
		query = query.Where("task_id = ?", queryParams.TaskID)
	}
	if queryParams.Action != "" {
		query = query.Where("action = ?", queryParams.Action)
	}
	if queryParams.Status != "" {
		query = query.Where("status = ?", queryParams.Status)
	}
	if queryParams.StartTimestamp != 0 {
		query = query.Where("submit_time >= ?", queryParams.StartTimestamp)
	}
	if queryParams.EndTimestamp != 0 {
		query = query.Where("submit_time <= ?", queryParams.EndTimestamp)
	}
	_ = query.Count(&total).Error
	return total
}

func (t *Task) ToOpenAIVideo() *dto.OpenAIVideo {
	openAIVideo := dto.NewOpenAIVideo()
	openAIVideo.ID = t.TaskID
	openAIVideo.Status = t.Status.ToVideoStatus()
	openAIVideo.Model = t.Properties.OriginModelName
	openAIVideo.SetProgressStr(t.Progress)
	openAIVideo.CreatedAt = t.CreatedAt
	if t.Status == TaskStatusSuccess {
		openAIVideo.CompletedAt = t.FinishTime
	}
	openAIVideo.SetMetadata("url", t.GetResultURL())
	return openAIVideo
}
