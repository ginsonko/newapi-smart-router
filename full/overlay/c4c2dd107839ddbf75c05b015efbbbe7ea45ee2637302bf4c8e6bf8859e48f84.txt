import assert from 'node:assert/strict'
import { readFileSync } from 'node:fs'
import { describe, test } from 'node:test'

import {
  markApplicationReady,
  reportApplicationBootFailure,
  type BootLifecycleController,
} from './boot-lifecycle.ts'

describe('application boot lifecycle', () => {
  test('forwards ready and failure signals to the pre-React controller', () => {
    const calls: string[] = []
    const controller: BootLifecycleController = {
      ready: () => calls.push('ready'),
      fail: () => calls.push('fail'),
    }

    markApplicationReady(controller)
    reportApplicationBootFailure(new Error('chunk failed'), controller)

    assert.deepEqual(calls, ['ready', 'fail'])
  })

  test('keeps the shell outside React and bounds automatic recovery to once', () => {
    const html = readFileSync(
      new URL('../../index.html', import.meta.url),
      'utf8'
    )
    const shellPosition = html.indexOf('id="boot-shell"')
    const rootPosition = html.indexOf('id="root"')
    const visibleBootMarkup = html.slice(
      shellPosition,
      html.indexOf('<script', shellPosition)
    )

    assert.ok(shellPosition >= 0)
    assert.ok(rootPosition > shellPosition)
    assert.ok(html.indexOf('__NEWAPI_EARLY_BOOT__') < html.indexOf('</head>'))
    assert.match(html, /hasFailure: \(\) => earlyFailure/)
    assert.match(
      html,
      /__NEWAPI_EARLY_BOOT__\?\.hasFailure\?\.\(\)\) recoverOnce\(\)/
    )
    assert.match(html, /sessionStorage\.getItem\(retryKey\) === '1'/)
    assert.match(html, /window\.location\.replace\(window\.location\.href\)/)
    assert.doesNotMatch(visibleBootMarkup, /\d{1,3}\s*%/)
  })
})
