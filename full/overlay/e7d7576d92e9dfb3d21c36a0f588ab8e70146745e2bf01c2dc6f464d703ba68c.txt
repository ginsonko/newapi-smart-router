import { api, type ApiRequestConfig } from '@/lib/api'

type ApiEnvelope<T> = {
  success: boolean
  message?: string
  data: T
}

const noStoreHeaders = {
  'Cache-Control': 'no-store',
  Pragma: 'no-cache',
}

function noStoreConfig(config: ApiRequestConfig = {}): ApiRequestConfig {
  return {
    ...config,
    headers: {
      ...noStoreHeaders,
      ...(config.headers as Record<string, string> | undefined),
    },
    // Tenant control-plane reads contain operational metadata and must not be
    // served from the generic in-flight GET deduplicator.
    disableDuplicate: true,
  }
}

function tenantParams(tenantId?: string) {
  const value = tenantId?.trim()
  return noStoreConfig(value ? { params: { tenant_id: value } } : {})
}

export async function getTenantCostDownstreams() {
  const response = await api.get<ApiEnvelope<TenantCostDownstream[]>>(
    '/api/tenant-cost/downstreams',
    noStoreConfig()
  )
  return response.data.data
}

    '/api/tenant-cost/downstreams',

export async function getTenantCostEvents(tenantId?: string) {
  const params: Record<string, string | number> = { limit: 100 }
  const value = tenantId?.trim()

export async function createTenantReportingToken(
  expiresAt?: number,
  tenantId?: string,
  requestId?: string
) {
  const body: { request_id: string; expires_at?: number } = {
    request_id: requestId?.trim() || crypto.randomUUID(),
  }
  if (expiresAt) body.expires_at = expiresAt
  const response = await api.post<ApiEnvelope<OneTimeTenantSecret>>(
    '/api/tenant-cost/reporting-tokens',
    body,
    noStoreConfig(tenantParams(tenantId))
  )
  return response.data.data
}

export async function rotateTenantReportingToken(
  tokenId: string,
  expiresAt?: number,
  tenantId?: string,
  requestId?: string
) {
  const body: { request_id: string; expires_at?: number } = {
    request_id: requestId?.trim() || crypto.randomUUID(),
  }
  if (expiresAt) body.expires_at = expiresAt
  const response = await api.post<ApiEnvelope<OneTimeTenantSecret>>(
    `/api/tenant-cost/reporting-tokens/${encodeURIComponent(tokenId)}/rotate`,
    body,
    noStoreConfig(tenantParams(tenantId))
  )
  return response.data.data
}

export async function revokeTenantReportingToken(

    noStoreConfig(tenantParams(tenantId))
  )
  return response.data.data
}

export async function replayTenantCostOutbox(
  eventIds: string[],
  tenantId?: string
) {
  const response = await api.post<
    ApiEnvelope<{ replayed: number; bounded: boolean }>
  >(
    '/api/tenant-cost/outbox/replay',
    {
      event_ids: eventIds.slice(0, 100),
      limit: Math.min(eventIds.length, 100) || 100,
    },
    noStoreConfig(tenantParams(tenantId))
  )
  return response.data.data
}
