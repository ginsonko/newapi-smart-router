package model

import (
	"github.com/QuantumNous/new-api/common"
	"github.com/QuantumNous/new-api/constant"
)

// ModelRequiresPassiveRecovery treats media capability as a reason never to
// generate a synthetic request. A text endpoint declaration cannot override
// known media generation capability; both canonical and mapped names matter.
func ModelRequiresPassiveRecovery(channelType int, names ...string) bool {
	for _, name := range names {
		endpoints := append(common.GetEndpointTypesByChannelType(constant.ChannelTypeOpenAI, name), ResolveModelEndpointTypes(channelType, name)...)
		for _, endpoint := range endpoints {
			switch endpoint {
			case constant.EndpointTypeImageGeneration, constant.EndpointTypeOpenAIVideo,
				constant.EndpointTypeAudioSpeech, constant.EndpointTypeAudioTranscription, constant.EndpointTypeAudioTranslation:
				return true
			}
		}
	}
	return false
}
