func TestTieredExpressionInputPredictionUsesStableDefaultRequestBranch(t *testing.T) {
	require.True(t, tieredExpressionSupportsInputPrediction(
		`param("service_tier") == "priority" ? p * 4 : p * 2`,
	), "request-dependent expressions have a deterministic empty/default context and learn real contexts separately")
	require.True(t, tieredExpressionSupportsInputPrediction(
		`has(header("anthropic-beta"), "fast-mode") ? p * 4 : p * 2`,
	))
	require.False(t, tieredExpressionSupportsInputPrediction(
		`hour("Asia/Hong_Kong") < 12 ? p * 2 : p * 3`,
	), "wall-clock expressions must wait for matching historical evidence")
}

type fakeSmartRatioSource struct {
	mu       sync.Mutex
	revision uint64
	ratios   map[string]map[string]int64
}

func (source *fakeSmartRatioSource) Revision() uint64 {
	source.mu.Lock()
	defer source.mu.Unlock()
	return source.revision
}

func (source *fakeSmartRatioSource) RouteRatioPPM(userGroup, usingGroup string) (int64, bool) {
	source.mu.Lock()
	defer source.mu.Unlock()
	groups := source.ratios[userGroup]
	ratio, exists := groups[usingGroup]
	return ratio, exists
}

func (source *fakeSmartRatioSource) update(revision uint64, userGroup, group string, ratio int64) {
	source.mu.Lock()
	defer source.mu.Unlock()
	source.revision = revision
	if source.ratios[userGroup] == nil {
		source.ratios[userGroup] = make(map[string]int64)
	}
	source.ratios[userGroup][group] = ratio
}

func (source *fakeSmartRatioSource) updateWithoutRevision(userGroup, group string, ratio int64) {
	source.mu.Lock()
	defer source.mu.Unlock()
	if source.ratios[userGroup] == nil {
		source.ratios[userGroup] = make(map[string]int64)
	}
	source.ratios[userGroup][group] = ratio
}

func TestPriceSnapshotRevisionRefreshesContentAndExplicitInvalidation(t *testing.T) {
	catalogs := publishedCatalogManager(t)
	ratios := &fakeSmartRatioSource{
		revision: 1,
		ratios: map[string]map[string]int64{
			"standard": {"economy": 20_000, "fallback": 50_000},
		},
	}
	manager := NewPriceSnapshotManager(catalogs, ratios)

	first, err := manager.GetOrRefresh(context.Background(), "standard")
	require.NoError(t, err)
	second, err := manager.GetOrRefresh(context.Background(), "standard")
	require.NoError(t, err)
	assert.Same(t, first, second)

	ratios.update(2, "standard", "economy", 30_000)
	_, exists := manager.Current("standard")
	assert.False(t, exists)
	changed, err := manager.GetOrRefresh(context.Background(), "standard")
	require.NoError(t, err)
	assert.NotEqual(t, first.Version, changed.Version)

	manager.Invalidate("standard")
	_, exists = manager.Current("standard")
	assert.False(t, exists)
	rebuilt, err := manager.GetOrRefresh(context.Background(), "standard")
	require.NoError(t, err)
	assert.Equal(t, changed.Version, rebuilt.Version)
}

func TestRatioToFixedPPMRoundsDeterministicallyAndRejectsInvalidValues(t *testing.T) {
	ppm, ok := ratioToFixedPPM(0.0200004)

	require.True(t, ok)

func TestPriceSnapshotUsesGroupModelPriceAndRefreshesAfterContractChange(t *testing.T) {
	originalGroups := rootsetting.UserUsableGroups2JSONString()
	originalRatios := ratio_setting.GroupRatio2JSONString()
	originalOverrides := ratio_setting.GroupGroupRatio2JSONString()
	originalContract := billing_setting.GetGroupModelPricingJSONString()
	t.Cleanup(func() {
		require.NoError(t, rootsetting.UpdateUserUsableGroupsByJSONString(originalGroups))
		require.NoError(t, ratio_setting.UpdateGroupRatioByJSONString(originalRatios))
		require.NoError(t, ratio_setting.UpdateGroupGroupRatioByJSONString(originalOverrides))
		_, _, err := billing_setting.UpdateGroupModelPricingFromJSONString(originalContract)
		require.NoError(t, err)
	})

	require.NoError(t, rootsetting.UpdateUserUsableGroupsByJSONString(`{"economy":"Economy","fallback":"Fallback"}`))
	require.NoError(t, ratio_setting.UpdateGroupRatioByJSONString(`{"economy":0.02,"fallback":0.05}`))
	require.NoError(t, ratio_setting.UpdateGroupGroupRatioByJSONString(`{}`))
	_, _, err := billing_setting.UpdateGroupModelPricingFromJSONString(`{
		"schema_version":1,
		"groups":{
			"economy":{"gpt-test":{"billing_mode":"fixed_price","model_price":0.5,"task_profile":{"unit":"per_second"}}},
			"fallback":{"gpt-test":{"billing_mode":"fixed_price","model_price":0.1,"task_profile":{"unit":"per_second"}}}
		}
	}`)
	require.NoError(t, err)

	manager := NewPriceSnapshotManager(publishedCatalogManager(t), NewAPIRouteRatioSource{})
	first, err := manager.GetOrRefresh(context.Background(), "standard")
	require.NoError(t, err)
	require.NotEmpty(t, first.RoutePrices)

	pricesByGroup := make(map[string]smartrouter.RoutePrice)
	catalog, exists := manager.catalogs.Current()
	require.True(t, exists)
	for _, contract := range catalog.Contracts {
		for _, pool := range contract.Pools {
			for _, route := range pool.Candidates {
				if price, ok := first.RoutePrices[route.RouteID]; ok {
					pricesByGroup[route.Group] = price
				}
			}
		}
	}
	assert.Equal(t, int64(10_000), pricesByGroup["economy"].ScorePPM)
	assert.Equal(t, int64(5_000), pricesByGroup["fallback"].ScorePPM)
	assert.True(t, pricesByGroup["economy"].ComparableWith(pricesByGroup["fallback"]))

	_, _, err = billing_setting.UpdateGroupModelPricingFromJSONString(`{
		"schema_version":1,
		"groups":{
			"economy":{"gpt-test":{"billing_mode":"fixed_price","model_price":0.5,"task_profile":{"unit":"per_second"}}},
			"fallback":{"gpt-test":{"billing_mode":"fixed_price","model_price":0.2,"task_profile":{"unit":"per_second"}}}
		}
	}`)
	require.NoError(t, err)
	_, current := manager.Current("standard")
	assert.False(t, current)
	refreshed, err := manager.GetOrRefresh(context.Background(), "standard")
	require.NoError(t, err)
	assert.NotEqual(t, first.Version, refreshed.Version)
}

	assert.Equal(t, int64(375_000), inherited.ScorePPM)
	assert.Equal(t, int64(400_000), explicit.ScorePPM)
	assert.True(t, inherited.ComparableWith(explicit))
	assert.False(t, inherited.Synthetic)
	assert.False(t, explicit.Synthetic)
}

func TestPriceSnapshotAppliesUserGroupOverrideAndSeparatesBillingUnits(t *testing.T) {
	originalGroups := rootsetting.UserUsableGroups2JSONString()
	originalRatios := ratio_setting.GroupRatio2JSONString()
	originalOverrides := ratio_setting.GroupGroupRatio2JSONString()
	originalContract := billing_setting.GetGroupModelPricingJSONString()
	t.Cleanup(func() {
		require.NoError(t, rootsetting.UpdateUserUsableGroupsByJSONString(originalGroups))
		require.NoError(t, ratio_setting.UpdateGroupRatioByJSONString(originalRatios))
		require.NoError(t, ratio_setting.UpdateGroupGroupRatioByJSONString(originalOverrides))
		_, _, err := billing_setting.UpdateGroupModelPricingFromJSONString(originalContract)
		require.NoError(t, err)
	})

	require.NoError(t, rootsetting.UpdateUserUsableGroupsByJSONString(`{"economy":"Economy","fallback":"Fallback"}`))
	require.NoError(t, ratio_setting.UpdateGroupRatioByJSONString(`{"economy":0.02,"fallback":0.05}`))
	require.NoError(t, ratio_setting.UpdateGroupGroupRatioByJSONString(`{"vip":{"economy":0.01,"fallback":0.06}}`))
	_, _, err := billing_setting.UpdateGroupModelPricingFromJSONString(`{
		"schema_version":1,
		"groups":{
			"economy":{"gpt-test":{"billing_mode":"fixed_price","model_price":0.5,"task_profile":{"unit":"per_second"}}},
			"fallback":{"gpt-test":{"billing_mode":"fixed_price","model_price":0.1,"task_profile":{"unit":"per_request"}}}
		}
	}`)
	require.NoError(t, err)

	manager := NewPriceSnapshotManager(publishedCatalogManager(t), NewAPIRouteRatioSource{})
	snapshot, err := manager.GetOrRefresh(context.Background(), "vip")
	require.NoError(t, err)

	pricesByGroup := make(map[string]smartrouter.RoutePrice)
	catalog, exists := manager.catalogs.Current()
	require.True(t, exists)
	for _, contract := range catalog.Contracts {
		for _, pool := range contract.Pools {
			for _, route := range pool.Candidates {
				if price, ok := snapshot.RoutePrices[route.RouteID]; ok {
					pricesByGroup[route.Group] = price
				}
			}
		}
	}
	assert.Equal(t, int64(5_000), pricesByGroup["economy"].ScorePPM)
	assert.Equal(t, int64(6_000), pricesByGroup["fallback"].ScorePPM)
	assert.False(t, pricesByGroup["economy"].ComparableWith(pricesByGroup["fallback"]))
}

func TestRoutePriceDescriptorKeepsDurationAndTieredPriceClassesCoherent(t *testing.T) {
	originalContract := billing_setting.GetGroupModelPricingJSONString()
	t.Cleanup(func() {
		_, _, err := billing_setting.UpdateGroupModelPricingFromJSONString(originalContract)
		require.NoError(t, err)
	})

	_, _, err := billing_setting.UpdateGroupModelPricingFromJSONString(`{
		"groups":{
			"per-request":{"video-model":{"billing_mode":"fixed_price","model_price":0.4,"task_profile":{"unit":"per_request"}}},
			"per-second":{"video-model":{"billing_mode":"fixed_price","model_price":0.2,"task_profile":{"unit":"per_second"}}},
			"fixed-15-a":{"video-model":{"billing_mode":"fixed_price","model_price":1,"task_profile":{"unit":"fixed_duration","fixed_duration_seconds":15}}},
			"fixed-15-b":{"video-model":{"billing_mode":"fixed_price","model_price":1,"task_profile":{"unit":"fixed_duration","fixed_duration_seconds":15}}},
			"fixed-20":{"video-model":{"billing_mode":"fixed_price","model_price":1,"task_profile":{"unit":"fixed_duration","fixed_duration_seconds":20}}},
			"tier-a":{"token-model":{"billing_mode":"tiered_expr","billing_expr":"tier(\"base\", p * 2 + c * 12)"}},
			"tier-b":{"token-model":{"billing_mode":"tiered_expr","billing_expr":"tier(\"base\", p * 2 + c * 12)"}},
			"tier-other":{"token-model":{"billing_mode":"tiered_expr","billing_expr":"tier(\"base\", p * 3 + c * 12)"}}
		}
	}`)
	require.NoError(t, err)

	source := NewAPIRouteRatioSource{}
	perRequest, err := source.RoutePriceDescriptor("video-model", "per-request", 500_000)
	require.NoError(t, err)
	perSecond, err := source.RoutePriceDescriptor("video-model", "per-second", 500_000)
	require.NoError(t, err)
	fixed15A, err := source.RoutePriceDescriptor("video-model", "fixed-15-a", 100_000)
	require.NoError(t, err)
	fixed15B, err := source.RoutePriceDescriptor("video-model", "fixed-15-b", 200_000)
	require.NoError(t, err)
	fixed20, err := source.RoutePriceDescriptor("video-model", "fixed-20", 100_000)
	require.NoError(t, err)

	assert.Equal(t, int64(200_000), perRequest.ScorePPM)
	assert.Equal(t, int64(100_000), perSecond.ScorePPM)
	assert.Equal(t, int64(100_000), fixed15A.ScorePPM)
	assert.Equal(t, int64(200_000), fixed15B.ScorePPM)
	assert.False(t, perRequest.ComparableWith(perSecond))
	assert.True(t, fixed15A.ComparableWith(fixed15B))
	assert.False(t, fixed15A.ComparableWith(fixed20))

	tierA, err := source.RoutePriceDescriptor("token-model", "tier-a", 100_000)
	require.NoError(t, err)
	tierB, err := source.RoutePriceDescriptor("token-model", "tier-b", 200_000)
	require.NoError(t, err)
	tierOther, err := source.RoutePriceDescriptor("token-model", "tier-other", 50_000)
	require.NoError(t, err)
	assert.Equal(t, int64(100_000), tierA.ScorePPM)
	assert.Equal(t, int64(200_000), tierB.ScorePPM)
	assert.True(t, tierA.ComparableWith(tierB))
	assert.False(t, tierA.ComparableWith(tierOther))
}
