package media

import (
	"github.com/QuantumNous/new-api/common"
	"strings"
)

// Exact image models and endpoints only. Arbitrary names and video requests
// never inherit image pricing dimensions. Route overrides remain authoritative.
func applyImageDimensionDefaults(capability *Capability, model, endpoint string) {
	switch strings.ToLower(strings.TrimSpace(model)) {
	case "image2", "image2.5", "gpt-image-2", "gpt-image-2.5":
	default:
		return
	}
	switch endpoint {
	case "openai-images-generations-v1", "openai-images-edits-v1":
	default:
		return
	}
	capability.Resolutions = []string{"1k", "2k", "4k"}
	capability.DefaultResolution = "1k"
	capability.AspectRatios = []string{"1:1", "3:2", "16:9", "9:16", "4:3", "3:4", "21:9"}
	capability.ResolutionAliases = make(map[string]string)
	capability.SizeAspectRatios = make(map[string]string)
	for _, preset := range common.Image2DimensionPresets() {
		capability.ResolutionAliases[preset.Size] = preset.Resolution
		capability.SizeAspectRatios[preset.Size] = preset.AspectRatio
	}
}

// MergeSizeAspectRatios excludes ambiguous labels and unavailable pixel aliases.
func MergeSizeAspectRatios(routes []Capability, aliases map[string]string, intersection bool) map[string]string {
	result := make(map[string]string)
	conflicts := make(map[string]bool)
	for _, route := range routes {
		for size, ratio := range route.SizeAspectRatios {
			if previous, ok := result[size]; ok && previous != ratio {
				conflicts[size] = true
			}
			result[size] = ratio
		}
	}
	for size, ratio := range result {
		if aliases[size] == "" || conflicts[size] {
			delete(result, size)
			continue
		}
		if intersection {
			for _, route := range routes {
				if route.SizeAspectRatios[size] != ratio {
					delete(result, size)
					break
				}
			}
		}
	}
	return result
}
