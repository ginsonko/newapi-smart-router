func setupRouteAnalyticsProjectionTestDB(t *testing.T) *gorm.DB {
	t.Helper()
	previousDB := model.DB
	dsn := fmt.Sprintf("file:route-analytics-service-%d?mode=memory&cache=shared", time.Now().UnixNano())
	db, err := gorm.Open(sqlite.Open(dsn), &gorm.Config{})
	require.NoError(t, err)
	require.NoError(t, db.AutoMigrate(
		&model.RouteAnalyticsDimensionV1{},
		&model.RouteAnalyticsObservationV1{},
		&model.RouteAnalyticsBucketV1{},
		&model.KeyUsageAnalyticsBucketV1{},
		&model.RouteAnalyticsHistoryV1{},
		&model.RouteAnalyticsWorkerStateV1{},
	))
	model.DB = db
	common.SetDatabaseTypes(common.DatabaseTypeSQLite, common.DatabaseTypeSQLite)
	t.Cleanup(func() {
		model.DB = previousDB
		common.SetDatabaseTypes(common.DatabaseTypeSQLite, common.DatabaseTypeSQLite)
	})
	return db
}

func applyRouteAnalyticsProjectionBuildResult(
	t *testing.T,
	result routeAnalyticsProjectionBuildResultV1,
) int {
	t.Helper()
	detailIDs := make(map[string]string, len(result.Dimensions))
	for _, dimension := range result.Dimensions {
		stored, err := model.EnsureRouteAnalyticsDimensionV1(context.Background(), dimension)
		require.NoError(t, err)
		detailIDs[stored.RouteFingerprintDigest] = stored.DetailID
	}
	require.NoError(t, attachRouteAnalyticsDetailIDsV1(result.Projections, detailIDs))
	applied, err := model.ApplyRouteAnalyticsProjectionsV1(context.Background(), result.Projections)
	require.NoError(t, err)
	return applied
}

func TestRouteAnalyticsProjectionCorrectsErrorToRecoveredConsume(t *testing.T) {
	db := setupRouteAnalyticsProjectionTestDB(t)
	catalog, route := routeAnalyticsProjectionTestCatalog(t)
	requestID := "raw-request-must-not-persist"
	errorLog := routeAnalyticsProjectionTestLog(t, 1_800, model.LogTypeError, requestID, routeAnalyticsProjectionTestAttempts(route, false))
	consumeLog := routeAnalyticsProjectionTestLog(t, 1_801, model.LogTypeConsume, requestID, routeAnalyticsProjectionTestAttempts(route, true))

	result := buildRouteAnalyticsProjectionsV1(catalog, []*model.Log{errorLog, consumeLog})
	assert.Zero(t, result.Skipped)
	assert.NotEmpty(t, result.Dimensions)
	assert.Greater(t, applyRouteAnalyticsProjectionBuildResult(t, result), 0)

	var routeBucket model.RouteAnalyticsBucketV1
	require.NoError(t, db.Where("granularity = ?", model.RouteAnalyticsGranularity5m).First(&routeBucket).Error)
	assert.Equal(t, int64(1), routeBucket.LogicalRequests)
	assert.Equal(t, int64(1), routeBucket.FinalSuccesses)
	assert.Zero(t, routeBucket.FirstHitSuccesses)
	assert.Equal(t, int64(1), routeBucket.RetryRecoveries)
	assert.Zero(t, routeBucket.ConfirmedFailures)
	assert.Equal(t, int64(1), routeBucket.TTFTCount)
	assert.Equal(t, int64(2_000), routeBucket.TTFTSumMS)
	assert.Equal(t, int64(900), routeBucket.CacheReadTokens)
	assert.Equal(t, int64(1_000), routeBucket.TotalInputTokens)
	assert.Equal(t, int64(190_000_000), routeBucket.PredictedInputMicros)
	assert.Equal(t, int64(1_000), routeBucket.PredictedInputTokens)
	assert.JSONEq(t, `{"price-v2":1}`, routeBucket.PriceRevisionCountsJSON)

	var keyBucket model.KeyUsageAnalyticsBucketV1
	require.NoError(t, db.Where("granularity = ?", model.RouteAnalyticsGranularity5m).First(&keyBucket).Error)
	assert.Equal(t, int64(1), keyBucket.LogicalRequests)
	assert.Equal(t, int64(1), keyBucket.Successes)
	assert.Zero(t, keyBucket.Failures)
	assert.Equal(t, int64(500), keyBucket.PositiveQuota)
	assert.Equal(t, int64(1_000), keyBucket.InputTokens)
	assert.Equal(t, int64(100), keyBucket.OutputTokens)
	assert.Equal(t, int64(900), keyBucket.CacheReadTokens)
	assert.Equal(t, route.Group, keyBucket.Group)

	var observations []model.RouteAnalyticsObservationV1
	require.NoError(t, db.Find(&observations).Error)
	encoded, err := common.Marshal(observations)
	require.NoError(t, err)
	assert.NotContains(t, string(encoded), requestID)
	assert.NotContains(t, string(encoded), "client_gone")
}

func TestRouteAnalyticsProjectionSelfHealsPrunedBucketsDuringReplay(t *testing.T) {
	db := setupRouteAnalyticsProjectionTestDB(t)
	catalog, route := routeAnalyticsProjectionTestCatalog(t)
	requestID := "pruned-bucket-replay"
	errorLog := routeAnalyticsProjectionTestLog(
		t, 1_800, model.LogTypeError, requestID, routeAnalyticsProjectionTestAttempts(route, false),
	)
	consumeLog := routeAnalyticsProjectionTestLog(
		t, 1_801, model.LogTypeConsume, requestID, routeAnalyticsProjectionTestAttempts(route, true),
	)

	initial := buildRouteAnalyticsProjectionsV1(catalog, []*model.Log{errorLog})
	assert.Greater(t, applyRouteAnalyticsProjectionBuildResult(t, initial), 0)
	require.NoError(t, db.Where("1 = 1").Delete(&model.RouteAnalyticsBucketV1{}).Error)
	require.NoError(t, db.Where("1 = 1").Delete(&model.KeyUsageAnalyticsBucketV1{}).Error)

	recovered := buildRouteAnalyticsProjectionsV1(catalog, []*model.Log{consumeLog})
	assert.Greater(t, applyRouteAnalyticsProjectionBuildResult(t, recovered), 0)
	assert.Zero(t, applyRouteAnalyticsProjectionBuildResult(t, recovered))

	var routeBuckets []model.RouteAnalyticsBucketV1
	require.NoError(t, db.Order("granularity ASC").Find(&routeBuckets).Error)
	require.Len(t, routeBuckets, 2)
	for _, bucket := range routeBuckets {
		assert.Equal(t, int64(1), bucket.LogicalRequests)
		assert.Equal(t, int64(1), bucket.FinalSuccesses)
		assert.Zero(t, bucket.ConfirmedFailures)
		assert.Equal(t, int64(1), bucket.RetryRecoveries)
	}

	var keyBuckets []model.KeyUsageAnalyticsBucketV1
	require.NoError(t, db.Order("granularity ASC").Find(&keyBuckets).Error)
	require.Len(t, keyBuckets, 2)
	for _, bucket := range keyBuckets {
		assert.Equal(t, int64(1), bucket.LogicalRequests)
		assert.Equal(t, int64(1), bucket.Successes)
		assert.Zero(t, bucket.Failures)
		assert.Equal(t, int64(500), bucket.PositiveQuota)
	}
}
