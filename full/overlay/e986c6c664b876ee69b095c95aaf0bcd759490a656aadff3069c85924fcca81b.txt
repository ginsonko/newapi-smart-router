func coldStartWorkerHint(
	session *SmartRoutingSession,
	routeCount int,
) SmartRouteRecoveryHint {
	routes := make([]smartrouter.CertifiedRoute, 0, routeCount)
	prices := make(map[string]int64, routeCount)
	quality := make(map[string]smartrouter.QualityState, routeCount)
	for index := 1; index <= routeCount; index++ {
		routeID := fmt.Sprintf("cold-%02d", index)
		routes = append(routes, smartrouter.CertifiedRoute{
			RouteID: routeID, Group: "economy", ChannelID: 100 + index,
			ChannelGeneration: fmt.Sprintf("generation-%02d", index), UpstreamModel: "gpt-test",
			Capabilities: smartroutersetting.CapabilityText, MaxContext: 128_000,
			CapacityDomain: "capacity-" + routeID,
		})
		prices[routeID] = int64(index) * 1_000
		quality[routeID] = smartrouter.QualityState{Phase: smartrouter.QualityUnknown, Epoch: 1}
	}
	policy := session.policy
	policy.AllowFutureGroups = true
	policy.MaxEffectiveRatioPPM = 100_000
	hint := SmartRouteRecoveryHint{
		CatalogVersion: "cold-catalog", ScopeKey: "token-cold",
		Request: session.request, Policy: policy, UserGroup: session.userGroup,
		AllowedGroups: []string{"economy"}, Routes: routes,
		Prices: smartrouter.PriceSnapshot{Version: "cold-prices", RatiosPPM: prices},
		Runtime: SmartRouteRuntimeSnapshot{
			Quality:     smartrouter.QualitySnapshot{Version: "cold-quality", Routes: quality},
			Credentials: smartrouter.CredentialSnapshot{Domains: map[string]smartrouter.CredentialState{}},
		},
	}
	if len(routes) > 0 {
		hint.ForegroundPhysicalKey = smartRoutePhysicalProbeKey(
			hint.Request.ContractID,
			hint.Request.Endpoint,
			routes[0],
		)
	}
	return hint
}

func publishProbeWorkerHint(t *testing.T, worker *SmartRouteProbeWorker, session *SmartRoutingSession) {
	t.Helper()
	// These tests exercise the ordinary failed-route recovery worker. Virgin
	// discovery has its own post-plan cold-start tests below and is intentionally
	// not scheduled from a pre-plan hint in V4.4.
	if state, ok := worker.engine.State.(*MemorySmartRouteStateStore); ok {
		state.mu.Lock()
		quality, exists := state.quality["cheap"]
		if !exists {
			quality = smartrouter.InitialQuality(false)
		}
		if smartrouter.IsVirginUnknown(quality) {
			nowMS := time.Now().UnixMilli()
			quality.Phase = smartrouter.QualityOpen
			quality.Revision = 1
			quality.LastFailureMS = nowMS - 1
			quality.LastUpdatedMS = nowMS - 1
			quality.NextProbeAtMS = nowMS - 1
			state.quality["cheap"] = quality
		}
		state.mu.Unlock()
	}
	catalog, ok := session.engine.Catalogs.Current()
	require.True(t, ok)
	routes, err := session.contractRoutes(*catalog)
	require.NoError(t, err)
	runtime, err := session.engine.State.Snapshot(context.Background(), SmartRouteStateSnapshotRequest{
		Routes: smartRouteStateRoutes(routes),
	})
	require.NoError(t, err)
	prices, err := session.engine.Prices.GetOrRefresh(context.Background(), session.userGroup)
	require.NoError(t, err)
	worker.Hint(SmartRouteRecoveryHint{
		CatalogVersion: catalog.Version, Request: session.request, Policy: session.policy,
		UserGroup: session.userGroup, AllowedGroups: session.allowedGroups,
		Routes: routes, Prices: *prices, Runtime: runtime,
	})
}

func TestBackgroundProbeWorkerPromotesSemanticTextSuccessImmediately(t *testing.T) {
	var calls atomic.Int64
	worker, state, session, _ := probeWorkerFixture(t, SmartRouteProbeExecutorFunc(func(
		_ context.Context,
		request SmartRouteProbeExecutionRequest,
	) SmartRouteProbeExecutionResult {
		calls.Add(1)
		assert.Equal(t, 45*time.Second, request.FirstTextTimeout)
		assert.Equal(t, 15*time.Second, request.CompletionGrace)
		return SmartRouteProbeExecutionResult{
			Outcome: SmartRouteOutcome{Outcome: smartrouter.OutcomeSuccess},
			TTFE:    100 * time.Millisecond, TTFEObserved: true,
			TTFT: 8 * time.Second, TTFTObserved: true, Completed: true,
		}
	}))
	publishProbeWorkerHint(t, worker, session)

	require.True(t, worker.RunOne())
	assert.Equal(t, int64(1), calls.Load())
	snapshot, err := state.Snapshot(context.Background(), SmartRouteStateSnapshotRequest{Routes: []SmartRouteStateRoute{
		{RouteID: "cheap", CapacityDomain: "cheap-capacity"},
	}})
	require.NoError(t, err)
	assert.Equal(t, smartrouter.QualityHealthy, snapshot.Quality.Routes["cheap"].Phase)
	assert.Equal(t, int64(8_000), snapshot.TTFT.Routes["cheap"].P95MS)
	assert.Equal(t, int64(1), worker.Stats().Succeeded)
	assert.Zero(t, worker.Stats().Queued, "the first successful probe ends recovery immediately")

	publishProbeWorkerHint(t, worker, session)
	assert.Zero(t, worker.Stats().Queued, "healthy routes do not restart synthetic probing")
}

func TestBackgroundProbeWorkerStopsAfterUnavailableRouteRecovers(t *testing.T) {
	worker, state, session, _ := probeWorkerFixture(t, SmartRouteProbeExecutorFunc(func(
		context.Context,
		SmartRouteProbeExecutionRequest,
	) SmartRouteProbeExecutionResult {
		return SmartRouteProbeExecutionResult{
			Outcome:      SmartRouteOutcome{Outcome: smartrouter.OutcomeSuccess},
			TTFTObserved: true, TTFT: time.Second, Completed: true,
		}
	}))
	nowMS := time.Now().UnixMilli()
	state.mu.Lock()
	state.quality["cheap"] = smartrouter.QualityState{
		Phase: smartrouter.QualityOpen, Epoch: 1,
		LastFailureMS: nowMS - 1_000, NextProbeAtMS: nowMS - 1,
	}
	state.mu.Unlock()

	publishProbeWorkerHint(t, worker, session)
	require.True(t, worker.RunOne())
	snapshot, err := state.Snapshot(context.Background(), SmartRouteStateSnapshotRequest{Routes: []SmartRouteStateRoute{
		{RouteID: "cheap", CapacityDomain: "cheap-capacity"},
	}})
	require.NoError(t, err)
	quality := snapshot.Quality.Routes["cheap"]
	assert.Equal(t, smartrouter.QualityHealthy, quality.Phase)
	assert.Equal(t, int64(1_000_000), quality.WarmingTrafficPPM)
	assert.Greater(t, quality.LastSuccessMS, quality.LastFailureMS)
	assert.Zero(t, worker.Stats().Queued)

	publishProbeWorkerHint(t, worker, session)
	assert.Zero(t, worker.Stats().Queued, "a confirmed recovery stops all further probes")
}

			Outcome: backgroundProbeFailure(ctx.Err(), "probe deadline"),
		}

	assert.Zero(t, snapshot.Quality.Routes["cheap"].TotalHardFailures)
	assert.Zero(t, snapshot.Quality.Routes["cheap"].TotalProbeFailures)
	assert.Equal(t, int64(1), worker.Stats().Queued)
}

func TestBackgroundProbeWorkerYieldsToForegroundCapacity(t *testing.T) {
	var calls atomic.Int64
	worker, _, session, setting := probeWorkerFixture(t, SmartRouteProbeExecutorFunc(func(
		context.Context,
		SmartRouteProbeExecutionRequest,
	) SmartRouteProbeExecutionResult {
		calls.Add(1)
		return SmartRouteProbeExecutionResult{
			Outcome:      SmartRouteOutcome{Outcome: smartrouter.OutcomeSuccess},
			TTFTObserved: true, TTFT: time.Second, Completed: true,
		}
	}))
	foreground, err := session.engine.State.TryAcquireCapacity(context.Background(), SmartRouteCapacityRequest{
		CapacityDomain: "cheap-capacity", MaxInflight: 2,
	})
	require.NoError(t, err)
	require.True(t, foreground.Granted())
	publishProbeWorkerHint(t, worker, session)

	require.True(t, worker.RunOne())
	assert.Zero(t, calls.Load())
	assert.Equal(t, int64(1), worker.Stats().Deferred)
	require.NoError(t, foreground.Lease.Release(context.Background()))

	worker.mu.Lock()
	for _, job := range worker.pending {
		job.notBefore = time.Now()
	}
	worker.mu.Unlock()
	require.True(t, worker.RunOne())
	assert.Equal(t, int64(1), calls.Load())
	assert.Equal(t, 45, setting.ProbeTimeoutSeconds)
}

func TestSmartRouteRecoveryDueOnlySchedulesUnavailableRoutes(t *testing.T) {
	nowMS := int64(100_000)
	tests := []struct {
		name    string
		quality smartrouter.QualityState
		due     bool
	}{
		{
			name: "open route whose deadline arrived is due",
			quality: smartrouter.QualityState{
				Phase: smartrouter.QualityOpen, NextProbeAtMS: nowMS,
				LastFailureMS: nowMS - 1_000,
			},
			due: true,
		},
		{
			name: "open route before its deadline is not due",
			quality: smartrouter.QualityState{
				Phase: smartrouter.QualityOpen, NextProbeAtMS: nowMS + 1,
				LastFailureMS: nowMS - 1_000,
			},
		},
		{
			name: "legacy warming without evidence normalizes to unknown",
			quality: smartrouter.QualityState{
				Phase: smartrouter.QualityWarming, SampleDueAtMS: nowMS,
			},
		},
		{name: "unknown is usable and never auto probed", quality: smartrouter.InitialQuality(false)},
		{name: "healthy needs no probe", quality: smartrouter.QualityState{Phase: smartrouter.QualityHealthy}},
	}
	for _, test := range tests {
		t.Run(test.name, func(t *testing.T) {
			assert.Equal(t, test.due, smartRouteRecoveryDue(test.quality, nowMS))
		})
	}
	assert.False(t, smartRouteQualityHasObservedSuccess(smartrouter.InitialQuality(true)),
		"an unobserved stable bootstrap alias is available but is not shared success evidence")
	assert.True(t, smartRouteQualityHasObservedSuccess(smartrouter.QualityState{
		Phase: smartrouter.QualityHealthy,
	}))
}

func TestProbeDispatcherHonorsBoundedRecoveryConcurrency(t *testing.T) {
	started := make(chan string, 2)
	release := make(chan struct{}, 2)
	worker, _, session, setting := probeWorkerFixture(t, SmartRouteProbeExecutorFunc(func(
		_ context.Context,
		request SmartRouteProbeExecutionRequest,
	) SmartRouteProbeExecutionResult {
		started <- request.Route.RouteID
		<-release
		return SmartRouteProbeExecutionResult{
			Outcome:      SmartRouteOutcome{Outcome: smartrouter.OutcomeSuccess},
			TTFTObserved: true, TTFT: time.Millisecond, Completed: true,
		}
	}))
	setting.ProbeMaxConcurrency = 1
	now := time.Now()
	routes := []smartrouter.CertifiedRoute{
		{RouteID: "recovery-a", Group: "economy", ChannelID: 900, ChannelGeneration: "a", UpstreamModel: "gpt-test", CapacityDomain: "capacity-a"},
		{RouteID: "recovery-b", Group: "economy", ChannelID: 901, ChannelGeneration: "b", UpstreamModel: "gpt-test", CapacityDomain: "capacity-b"},
	}
	for _, route := range routes {
		worker.enqueue(smartRouteProbeJob{
			key: "physical-" + route.RouteID, catalogVersion: "catalog",
			contractID: session.request.ContractID, canonicalModel: session.request.CanonicalModel,
			endpoint: session.request.Endpoint, route: route,
			aliases:   map[string]smartrouter.CertifiedRoute{route.RouteID: route},
			notBefore: now, lastHint: now, forceProbe: true,
		}, setting.ProbeQueueCapacity)
	}

	worker.dispatch(*setting)
	var first string
	select {
	case first = <-started:
	case <-time.After(2 * time.Second):
		require.FailNow(t, "first bounded recovery probe did not start")
	}
	worker.dispatch(*setting)
	select {
	case second := <-started:
		require.FailNow(t, "dispatcher exceeded recovery concurrency", "%s then %s", first, second)
	case <-time.After(50 * time.Millisecond):
	}
	release <- struct{}{}
	require.Eventually(t, func() bool {
		worker.mu.Lock()
		defer worker.mu.Unlock()
		return len(worker.running) == 0
	}, time.Second, 10*time.Millisecond)
	worker.dispatch(*setting)
	var second string
	select {
	case second = <-started:
	case <-time.After(2 * time.Second):
		require.FailNow(t, "second bounded recovery probe did not start")
	}
	assert.NotEqual(t, first, second)
	release <- struct{}{}
	worker.wg.Wait()
}

func TestRecoveryProbePhysicalLeaseSingleFlightsAcrossWorkersAndAliases(t *testing.T) {
	started := make(chan struct{}, 2)
	release := make(chan struct{})
	var calls atomic.Int64
	executor := SmartRouteProbeExecutorFunc(func(
		_ context.Context,
		_ SmartRouteProbeExecutionRequest,
	) SmartRouteProbeExecutionResult {
		calls.Add(1)
		started <- struct{}{}
		<-release
		return SmartRouteProbeExecutionResult{Outcome: SmartRouteOutcome{
			Outcome: smartrouter.OutcomeInfrastructure, RouteFailure: true, AffectsRouteHealth: true,
		}}
	})
	workerA, state, session, setting := probeWorkerFixture(t, executor)
	workerB, err := NewSmartRouteProbeWorker(session.engine, executor)
	require.NoError(t, err)
	now := time.Now()
	physicalKey := "shared-physical-key"
	routeA := smartrouter.CertifiedRoute{RouteID: "alias-a", ChannelID: 1, CapacityDomain: "capacity-a"}
	routeB := smartrouter.CertifiedRoute{RouteID: "alias-b", ChannelID: 1, CapacityDomain: "capacity-b"}
	for worker, route := range map[*SmartRouteProbeWorker]smartrouter.CertifiedRoute{workerA: routeA, workerB: routeB} {
		worker.enqueue(smartRouteProbeJob{
			key: physicalKey, catalogVersion: "catalog", contractID: session.request.ContractID,
			canonicalModel: session.request.CanonicalModel, endpoint: session.request.Endpoint,
			route: route, aliases: map[string]smartrouter.CertifiedRoute{route.RouteID: route},
			notBefore: now, lastHint: now, forceProbe: true,
		}, setting.ProbeQueueCapacity)
	}
	_ = state

	doneA := make(chan struct{})
	go func() {
		workerA.RunOne()
		close(doneA)
	}()
	select {
	case <-started:
	case <-time.After(2 * time.Second):
		require.FailNow(t, "first physical probe did not start")
	}
	require.True(t, workerB.RunOne())
	assert.Equal(t, int64(1), calls.Load(), "the alias must not dispatch a second physical probe")
	close(release)
	<-doneA
}

func TestProbeWorkerHotDisableClearsAndRebuildsPendingRecovery(t *testing.T) {
	worker, _, session, setting := probeWorkerFixture(t, SmartRouteProbeExecutorFunc(func(
		context.Context,
		SmartRouteProbeExecutionRequest,
	) SmartRouteProbeExecutionResult {
		return SmartRouteProbeExecutionResult{}
	}))
	hint := coldStartWorkerHint(session, 1)
	nowMS := time.Now().UnixMilli()
	hint.Runtime.Quality.Routes["cold-01"] = smartrouter.QualityState{
		Phase: smartrouter.QualityOpen, Epoch: 2, LastFailureMS: nowMS - 1_000, NextProbeAtMS: nowMS - 1,
	}
	worker.Hint(hint)
	worker.mu.Lock()
	require.Len(t, worker.pending, 1)
	assert.Empty(t, worker.coldStartWaves)
	worker.mu.Unlock()

	disabled := *setting
	disabled.BackgroundProbeEnabled = false
	worker.dispatch(disabled)
	worker.mu.Lock()
	assert.Empty(t, worker.coldStartWaves)
	assert.Len(t, worker.pending, 1, "disabling execution must retain the bounded recovery work item")
	worker.mu.Unlock()

	worker.Hint(hint)
	worker.mu.Lock()
	assert.Empty(t, worker.coldStartWaves)
	assert.Len(t, worker.pending, 1, "re-enabling must merge rather than duplicate the physical recovery item")
	worker.mu.Unlock()
}

func TestSmartRouteRecoveryHintHasDueWorkIgnoresHealthyAndExcludedRoutes(t *testing.T) {
	nowMS := time.Now().UnixMilli()
	policy := smartrouter.NormalizePolicy(smartrouter.Policy{
		Version: smartrouter.PolicyVersion, Enabled: true, AllowFutureGroups: true,
		AllowFutureCertifiedRoutes: true,
	})
	hint := SmartRouteRecoveryHint{
		Request:       smartrouter.RouteRequest{ReplayClass: smartrouter.ReplaySafeText},
		Policy:        policy,
		AllowedGroups: []string{"default"},
		Routes: []smartrouter.CertifiedRoute{
			{RouteID: "route", Group: "default"},
		},
		Runtime: SmartRouteRuntimeSnapshot{Quality: smartrouter.QualitySnapshot{Routes: map[string]smartrouter.QualityState{
			"route": {Phase: smartrouter.QualityHealthy, Epoch: 1},
		}}},
	}
	assert.False(t, smartRouteRecoveryHintHasDueWork(hint, nowMS))

	hint.Runtime.Quality.Routes["route"] = smartrouter.QualityState{
		Phase: smartrouter.QualityOpen, Epoch: 1, NextProbeAtMS: nowMS,
	}
	assert.True(t, smartRouteRecoveryHintHasDueWork(hint, nowMS))

	hint.ExcludedRoutes = []string{"route"}
	assert.False(t, smartRouteRecoveryHintHasDueWork(hint, nowMS))
}
