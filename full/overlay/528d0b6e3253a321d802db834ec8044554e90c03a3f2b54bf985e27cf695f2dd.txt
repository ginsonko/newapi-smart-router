func TestSmartContractIdentityUsesOnlyModelAndProtocol(t *testing.T) {
	chatID, chatFingerprint, err := BuildSmartContractIdentity("gpt-5.6", SmartEndpointChat)
	require.NoError(t, err)
	require.NotEmpty(t, chatID)
	assert.Equal(t, SmartEndpointChat, chatFingerprint)

	repeatedID, repeatedFingerprint, err := BuildSmartContractIdentity(" gpt-5.6 ", SmartEndpointChat)
	require.NoError(t, err)
	assert.Equal(t, chatID, repeatedID)
	assert.Equal(t, chatFingerprint, repeatedFingerprint)

	responsesID, _, err := BuildSmartContractIdentity("gpt-5.6", SmartEndpointResponses)
	require.NoError(t, err)
	assert.NotEqual(t, chatID, responsesID)

	otherModelID, _, err := BuildSmartContractIdentity("gpt-5.6-mini", SmartEndpointChat)
	require.NoError(t, err)
	assert.NotEqual(t, chatID, otherModelID)

	_, _, err = BuildSmartContractIdentity("gpt-5.6", "openai-chat-v2")
	assert.ErrorIs(t, err, ErrUnsupportedSmartRouteEndpoint)
}

func TestBuildSmartRouteRequestReplayBoundaries(t *testing.T) {
	tests := []struct {
		name   string
		body   string
		replay smartrouter.ReplayClass
	}{
		{
			name:   "previous response binds upstream state",
			body:   `{"model":"gpt-5.6","previous_response_id":"resp_123","input":"continue"}`,
			replay: smartrouter.ReplayStateBound,
		},
		{
			name:   "conversation object binds upstream state",
			body:   `{"model":"gpt-5.6","conversation":{"id":"conv_123"},"input":"continue"}`,
			replay: smartrouter.ReplayStateBound,
		},
		{
			name:   "empty state references remain replay safe",
			body:   `{"model":"gpt-5.6","previous_response_id":"","conversation":null,"input":"hello"}`,
			replay: smartrouter.ReplaySafeText,
		},
		{
			name:   "store true is side effecting",
			body:   `{"model":"gpt-5.6","store":true,"input":"hello"}`,
			replay: smartrouter.ReplaySideEffecting,
		},
		{
			name:   "background is side effecting",
			body:   `{"model":"gpt-5.6","background":true,"input":"hello"}`,
			replay: smartrouter.ReplaySideEffecting,
		},
		{
			name:   "hosted tool is side effecting",
			body:   `{"model":"gpt-5.6","tools":[{"type":"web_search_preview"}],"input":"latest news"}`,
			replay: smartrouter.ReplaySideEffecting,
		},
		{
			name:   "versioned hosted tool is side effecting",
			body:   `{"model":"gpt-5.6","tools":[{"type":"web_search_20250305"}],"input":"latest news"}`,
			replay: smartrouter.ReplaySideEffecting,
		},
		{
			name:   "future hosted tool fails closed",
			body:   `{"model":"gpt-5.6","tools":[{"type":"future_hosted_tool_v2"}],"input":"do work"}`,
			replay: smartrouter.ReplaySideEffecting,
		},
		{
			name:   "custom client tool remains replay safe",
			body:   `{"model":"gpt-5.6","tools":[{"type":"custom","name":"local_tool"}],"input":"do work"}`,
			replay: smartrouter.ReplaySafeText,
		},
		{
			name:   "embedded audio is unsupported in phase one",
			body:   `{"model":"gpt-5.6","input":[{"type":"input_audio","audio":"AA=="}]}`,
			replay: smartrouter.ReplayUnsupported,
		},
		{
			name:   "provider file id is unsupported in phase one",
			body:   `{"model":"gpt-5.6","input":[{"type":"input_file","file_id":"file_123"}]}`,
			replay: smartrouter.ReplayUnsupported,
		},
		{
			name:   "state binding wins over side effect",
			body:   `{"model":"gpt-5.6","store":true,"previous_response_id":"resp_123","input":"continue"}`,
			replay: smartrouter.ReplayStateBound,
		},
	}

	for _, test := range tests {
		t.Run(test.name, func(t *testing.T) {
			c := newSmartRequestContext(t, http.MethodPost, "/v1/responses", test.body)
			request, err := BuildSmartRouteRequest(c, "gpt-5.6", 100)
			require.NoError(t, err)
			assert.Equal(t, test.replay, request.ReplayClass)
		})
	}
}

func TestBuildSmartRouteRequestClassifiesMediaContracts(t *testing.T) {
	tests := []struct {
		path         string
		model        string
		body         string
		endpoint     string
		replay       smartrouter.ReplayClass
		capabilities uint64
	}{
		{path: "/v1/images/generations", model: "gpt-image-1", body: `{"model":"gpt-image-1","prompt":"a cat"}`, endpoint: SmartEndpointImages, replay: smartrouter.ReplaySafeImage, capabilities: SmartCapabilityImageGeneration},

		{path: "/v1/videos", model: "sora-2", body: `{"model":"sora-2","prompt":"a cat"}`, endpoint: SmartEndpointVideos, replay: smartrouter.ReplaySafeVideo, capabilities: SmartCapabilityVideoGeneration | SmartCapabilityAsyncTask},

		{path: "/v1/audio/translations", model: "whisper-1", body: `{"model":"whisper-1"}`, endpoint: SmartEndpointAudioTranslation, replay: smartrouter.ReplaySafeAudio, capabilities: SmartCapabilityAudioTranscript},
	}
	for _, test := range tests {
		t.Run(test.path, func(t *testing.T) {
			c := newSmartRequestContext(t, http.MethodPost, test.path, test.body)
			request, err := BuildSmartRouteRequest(c, test.model, 0)

			assert.Equal(t, test.endpoint, request.Endpoint)
			assert.Equal(t, test.replay, request.ReplayClass)
			assert.Equal(t, test.capabilities, request.RequiredCapabilities)
			assert.NotEmpty(t, request.ContractID)
		})

	assert.Equal(t, "720p", request.Media.Resolution)
	require.Len(t, request.Media.References, 1)
	assert.Equal(t, smartrouter.MediaKindImage, request.Media.References[0].Type)
	assert.True(t, request.Media.References[0].MetadataTrusted)
	assert.Equal(t, SmartCapabilityVideoGeneration|SmartCapabilityAsyncTask|SmartCapabilityReferenceImage, request.RequiredCapabilities)

	c, imageBody := newSmartMultipartRequestContext(t, "/v1/images/edits", map[string]string{
		"model": "gpt-image-1", "prompt": "remove the background", "size": "1024x1024",

	assert.Equal(t, SmartCapabilityImageGeneration|SmartCapabilityReferenceImage, request.RequiredCapabilities)
	require.NotNil(t, request.Media)
	require.Len(t, request.Media.References, 1)
	assert.Equal(t, smartrouter.MediaKindImage, request.Media.References[0].Type)
	assertSmartRequestBodyReusable(t, c, imageBody)

	assert.Equal(t, smartrouter.ReplaySafeAudio, request.ReplayClass)
	require.NotNil(t, request.Media)
	assert.Equal(t, int64(1), request.Media.OutputCount)
	assertSmartRequestBodyReusable(t, c, audioBody)
}

func TestSmartMediaAttemptReplayBoundary(t *testing.T) {
	transport := types.NewErrorWithStatusCode(errors.New("dial tcp timeout"), types.ErrorCodeDoRequestFailed, http.StatusInternalServerError)
	assert.Equal(t, SmartMediaAttemptAcceptanceUnknown, ClassifySmartMediaAttempt(transport))
	parsed := types.NewErrorWithStatusCode(errors.New("invalid successful response"), types.ErrorCodeBadResponseBody, http.StatusOK)
	assert.Equal(t, SmartMediaAttemptAcceptanceUnknown, ClassifySmartMediaAttempt(parsed))
	rejected := types.NewErrorWithStatusCode(errors.New("upstream rejected"), types.ErrorCodeBadResponseStatusCode, http.StatusBadGateway)
	assert.Equal(t, SmartMediaAttemptNotAccepted, ClassifySmartMediaAttempt(rejected))
	assert.True(t, ClassifySmartRouteOutcome(rejected, nil).CanTryNextRoute())
	marked := MarkSmartMediaSubmissionAmbiguous(transport)
	assert.ErrorIs(t, marked, ErrSmartMediaSubmissionAmbiguous)
	assert.False(t, ClassifySmartRouteOutcome(marked, nil).CanTryNextRoute())

	dnsFailure := types.NewOpenAIError(&net.OpError{
		Op: "dial", Net: "tcp", Err: &net.DNSError{Name: "provider.example", Err: "server misbehaving"},
	}, types.ErrorCodeDoRequestFailed, http.StatusInternalServerError)
	assert.Equal(t, SmartMediaAttemptNotAccepted, ClassifySmartMediaAttempt(dnsFailure))
	assert.True(t, IsSmartMediaRoutePreparationFailure(dnsFailure))
	dnsMarked := MarkSmartMediaRouteNotAccepted(dnsFailure)
	assert.ErrorIs(t, dnsMarked, ErrSmartMediaRouteNotAccepted)
	assert.True(t, ClassifySmartRouteOutcome(dnsMarked, nil).CanTryNextRoute())

	postConnectFailure := types.NewOpenAIError(&net.OpError{
		Op: "read", Net: "tcp", Err: errors.New("connection reset by peer"),
	}, types.ErrorCodeDoRequestFailed, http.StatusInternalServerError)
	assert.Equal(t, SmartMediaAttemptAcceptanceUnknown, ClassifySmartMediaAttempt(postConnectFailure))
	assert.False(t, IsSmartMediaRoutePreparationFailure(postConnectFailure))
}

func TestBuildSmartRouteRequestLeavesDiskBodyReusableOnSuccessAndFailure(t *testing.T) {
	tests := []struct {
		name      string
		body      string
		wantError bool
	}{
		{
			name: "success",
			body: `{"model":"gpt-5.6","messages":[{"role":"user","content":"hello"}]}`,
		},
		{
			name:      "invalid JSON",
			body:      `{"model":"gpt-5.6","messages":[}`,
			wantError: true,
		},
	}

	for _, test := range tests {
		t.Run(test.name, func(t *testing.T) {
			c := newSmartRequestContext(t, http.MethodPost, "/v1/chat/completions", test.body)
			storage := installSmartDiskBodyStorage(t, c, test.body)
			require.True(t, storage.IsDisk())

			_, err := BuildSmartRouteRequest(c, "gpt-5.6", 10)
			if test.wantError {
				assert.ErrorIs(t, err, ErrInvalidSmartRouteRequest)
			} else {
				require.NoError(t, err)
			}
			storedBody, readErr := io.ReadAll(c.Request.Body)
			require.NoError(t, readErr)
			assert.Equal(t, test.body, string(storedBody))

			common.CleanupBodyStorage(c)
			_, closedErr := storage.Bytes()
			assert.ErrorIs(t, closedErr, common.ErrStorageClosed)
		})
	}
}

	deadline := time.Now().Add(time.Minute).Truncate(time.Millisecond)
	ctx, cancel := context.WithDeadline(c.Request.Context(), deadline)

func TestClassifySmartRouteOutcome(t *testing.T) {
	tests := []struct {
		name          string
		apiErr        *types.NewAPIError
		contextErr    error
		outcome       smartrouter.QualityOutcome
		scope         SmartRouteFailureScope
		retry         bool
		affectsHealth bool
		quarantineMS  int64
	}{
		{
			name:          "success",
			outcome:       smartrouter.OutcomeSuccess,
			scope:         SmartFailureNone,
			affectsHealth: true,
		},
		{
			name: "model not found retries another certified route without degrading channel",
			apiErr: types.WithOpenAIError(types.OpenAIError{
				Message: "model not found", Type: "invalid_request_error", Code: "model_not_found",
			}, http.StatusNotFound),
			outcome: smartrouter.OutcomeInfrastructure, scope: SmartFailureRouteModel, retry: true,
			affectsHealth: true,
		},
		{
			name: "unsupported parameter retries matching contract",
			apiErr: types.WithOpenAIError(types.OpenAIError{
				Message: "unsupported parameter response_format", Type: "invalid_request_error", Code: "unsupported_parameter",
			}, http.StatusBadRequest),
			outcome: smartrouter.OutcomeInfrastructure, scope: SmartFailureRouteContract, retry: true,
			affectsHealth: true,
		},
		{
			name: "unsupported parameter text wins over generic invalid request code",
			apiErr: types.NewOpenAIError(
				errors.New("unsupported_parameter: response_format"),
				types.ErrorCodeInvalidRequest,
				http.StatusBadRequest,
			),
			outcome: smartrouter.OutcomeInfrastructure, scope: SmartFailureRouteContract, retry: true,
			affectsHealth: true,
		},
		{
			name: "skip retry does not suppress explicit model route fallback",
			apiErr: types.NewErrorWithStatusCode(
				errors.New("model not found"), types.ErrorCodeModelNotFound, http.StatusNotFound,
				types.ErrOptionWithSkipRetry(),
			),
			outcome: smartrouter.OutcomeInfrastructure, scope: SmartFailureRouteModel, retry: true,
			affectsHealth: true,
		},
		{
			name: "skip retry does not suppress explicit unsupported contract fallback",
			apiErr: types.NewErrorWithStatusCode(
				errors.New("unsupported_parameter response_format"), types.ErrorCodeInvalidRequest, http.StatusBadRequest,
				types.ErrOptionWithSkipRetry(),
			),
			outcome: smartrouter.OutcomeInfrastructure, scope: SmartFailureRouteContract, retry: true,
			affectsHealth: true,
		},
		{
			name: "Responses single route overrides generic upstream 502",
			apiErr: types.NewErrorWithStatusCode(
				errors.New("ambiguous malformed stream"), types.ErrorCodeSmartResponsesSingleRoute, http.StatusBadGateway,
				types.ErrOptionWithSkipRetry(),
			),
			outcome: smartrouter.OutcomeInfrastructure, scope: SmartFailureInfrastructure,
		},
		{
			name: "Responses single route overrides capacity classification",
			apiErr: types.NewErrorWithStatusCode(
				errors.New("ambiguous capacity event"), types.ErrorCodeSmartResponsesSingleRoute, http.StatusTooManyRequests,
				types.ErrOptionWithSkipRetry(),
			),
			outcome: smartrouter.OutcomeInfrastructure, scope: SmartFailureInfrastructure,
		},
		{
			name: "Responses single route overrides credential classification",
			apiErr: types.NewErrorWithStatusCode(
				errors.New("ambiguous authentication event"), types.ErrorCodeSmartResponsesSingleRoute, http.StatusUnauthorized,
				types.ErrOptionWithSkipRetry(),
			),
			outcome: smartrouter.OutcomeInfrastructure, scope: SmartFailureInfrastructure,
		},
		{

func TestClassifySmartRouteOutcomeUsesUpstreamStatusAfterMapping(t *testing.T) {
	tests := []struct {
		name   string
		status int
		code   types.ErrorCode
		body   types.OpenAIError
		want   smartrouter.QualityOutcome
		scope  SmartRouteFailureScope
	}{
		{
			name: "mapped 429 remains capacity failure", status: http.StatusTooManyRequests,
			code: types.ErrorCodeBadResponseStatusCode,
			body: types.OpenAIError{Message: "rate limit", Type: "rate_limit_error", Code: "rate_limit_exceeded"},
			want: smartrouter.OutcomeCapacityLimited, scope: SmartFailureCapacity,
		},
		{
			name: "mapped 403 remains model access failure", status: http.StatusForbidden,
			code: types.ErrorCodeBadResponseStatusCode,
			body: types.OpenAIError{Message: "account does not have access to model", Type: "permission_denied", Code: "model_access_denied"},
			want: smartrouter.OutcomeInfrastructure, scope: SmartFailureRouteModel,
		},

			name: "mapped 404 remains model mismatch", status: http.StatusNotFound,

func TestPreserveSmartRouteUpstreamResponseEvidenceKeepsClientGoneNeutral(t *testing.T) {
	c := newSmartRequestContext(t, http.MethodPost, "/v1/chat/completions", `{}`)
	downstreamCtx, cancel := context.WithCancel(c.Request.Context())
	c.Request = c.Request.WithContext(downstreamCtx)
	cancel()

	// Receiving a successful response header does not prove that a subsequent
	// write failure came from the provider. Once the original downstream is
	// cancelled, a wrapped cancellation must remain client_gone and health
	// neutral rather than becoming a retryable 502.
	common.SetContextKey(c, constant.ContextKeyUpstreamResponseReceived, true)
	common.SetContextKey(c, constant.ContextKeyUpstreamStatusCode, http.StatusOK)
	apiErr := types.NewErrorWithStatusCode(
		context.Canceled,
		types.ErrorCodeDoRequestFailed,
		http.StatusBadGateway,
	)

	apiErr = PreserveSmartRouteUpstreamResponseEvidence(c, apiErr)
	require.False(t, apiErr.HasUpstreamResponse())
	outcome := ClassifySmartRouteOutcome(apiErr, c.Request.Context().Err())
	assert.Equal(t, smartrouter.OutcomeClientCancelled, outcome.Outcome)
	assert.Equal(t, SmartFailureCancellation, outcome.FailureScope)
	assert.False(t, outcome.CanTryNextRoute())
	assert.False(t, outcome.AffectsRouteHealth)
}

func newSmartMultipartRequestContext(
	t *testing.T,
	path string,
	fields map[string]string,
	files map[string]string,
) (*gin.Context, []byte) {
	t.Helper()
	var body bytes.Buffer
	writer := multipart.NewWriter(&body)
	for key, value := range fields {
		require.NoError(t, writer.WriteField(key, value))
	}
	for field, value := range files {
		part, err := writer.CreateFormFile(field, field+".bin")
		require.NoError(t, err)
		_, err = io.WriteString(part, value)
		require.NoError(t, err)
	}
	require.NoError(t, writer.Close())
	raw := append([]byte(nil), body.Bytes()...)
	c := newSmartRequestContext(t, http.MethodPost, path, string(raw))
	c.Request.Header.Set("Content-Type", writer.FormDataContentType())
	return c, raw
}

func installSmartDiskBodyStorage(t *testing.T, c *gin.Context, body string) common.BodyStorage {
	t.Helper()
	previous := common.GetDiskCacheConfig()
	common.SetDiskCacheConfig(common.DiskCacheConfig{
		Enabled: true, ThresholdMB: 0, MaxSizeMB: 16, Path: t.TempDir(),
	})
	storage, err := common.CreateBodyStorage([]byte(body))
	common.SetDiskCacheConfig(previous)
	require.NoError(t, err)
	t.Cleanup(func() { _ = storage.Close() })
	c.Set(common.KeyBodyStorage, storage)
	return storage
}
