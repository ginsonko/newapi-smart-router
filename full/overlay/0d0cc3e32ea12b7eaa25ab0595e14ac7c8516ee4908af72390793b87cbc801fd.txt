import { api } from '@/lib/api'

import type { ParameterPricingContract } from './parameter-pricing-contract'

export const PARAMETER_PRICING_QUERY_KEY = ['parameter-pricing'] as const

export type ParameterPricingData = {
  supported: boolean
  schema_version: number
  revision: string
  entry_count: number
  contract: ParameterPricingContract
  changed?: boolean
  current_revision?: string
  expected_revision_matches?: boolean
}

export type ParameterPricingResponse = {
  success: boolean
  message?: string
  code?: string
  data: ParameterPricingData
}

export async function getParameterPricing(): Promise<ParameterPricingResponse> {
  const response = await api.get<ParameterPricingResponse>(
    '/api/option/parameter-pricing',
    { disableDuplicate: true, skipErrorHandler: true }
  )
  return response.data
}

export async function previewParameterPricing(
  expectedRevision: string,
  contract: ParameterPricingContract
): Promise<ParameterPricingResponse> {
  const response = await api.post<ParameterPricingResponse>(
    '/api/option/parameter-pricing/preview',
    { expected_revision: expectedRevision, contract },
    { skipErrorHandler: true }
  )
  return response.data
}

export async function updateParameterPricing(
  expectedRevision: string,
  contract: ParameterPricingContract
): Promise<ParameterPricingResponse> {
  const response = await api.put<ParameterPricingResponse>(
    '/api/option/parameter-pricing',
    { expected_revision: expectedRevision, contract },
    { skipErrorHandler: true }
  )
  return response.data
}
