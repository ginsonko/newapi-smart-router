function normalizeHealth(data: unknown) {
  if (Array.isArray(data)) return data as SmartRouterHealthRoute[]
  if (data && typeof data === 'object' && 'items' in data) {
    const items = (data as { items?: unknown }).items
    return Array.isArray(items) ? (items as SmartRouterHealthRoute[]) : []
  }
  return []
}

function normalizeMeta(data: unknown): HealthMeta | undefined {
  if (!data || Array.isArray(data) || typeof data !== 'object') return undefined
  const meta = data as HealthMeta
  return {
    ...meta,
    updated_at_ms: meta.updated_at_ms ?? meta.snapshot?.generated_at_ms,
    snapshot_version: meta.snapshot_version ?? meta.snapshot?.id,
    stale: meta.stale ?? meta.snapshot?.stale,
  }
}

function formatActualInputCacheRate(rate?: number) {
  if (
    typeof rate !== 'number' ||
    !Number.isFinite(rate) ||
    rate < 0 ||
    rate > 1_000_000
  ) {
    return ''
  }
  return `${new Intl.NumberFormat(undefined, {
    maximumFractionDigits: 1,
  }).format(rate / 10_000)}%`
}

function probeState(route: SmartRouterHealthRoute) {
  if (route.probe_state) return route.probe_state.toLowerCase()
  const state = canonicalHealthState(route)
  if (state === 'probing') return 'active'
  if (route.probe_overdue) return 'due'
  return 'idle'
}

function routeMatchesSearch(route: SmartRouterHealthRoute, search: string) {
  const query = search.trim().toLowerCase()
  if (!query) return true
  return [
    route.model,
    route.group,
    route.endpoint,
    route.channel_id,
    route.route_id,
    route.contract_id,
    route.sanitized_error,
    route.last_error,
    route.last_probe_error,
  ].some((value) =>
    String(value ?? '')
      .toLowerCase()
      .includes(query)
  )
}

function routeMatchesProbeFilter(
  route: SmartRouterHealthRoute,
  filter: AdminProbeFilter
) {
  if (filter === 'all') return true
  if (filter === 'due') {
    return route.probe_overdue || probeState(route) === 'due'
  }
  if (filter === 'active') {
    const state = probeState(route)
    return state === 'active' || state === 'queued' || state === 'running'
  }
  return isMediaRoute(route)
}

function routeNeedsAdministratorAction(route: SmartRouterHealthRoute) {
  return (
    route.requires_admin_review === true ||
    canonicalHealthState(route) === 'needs_review'
  )
}

function routeMatchesStateFilter(
  route: SmartRouterHealthRoute,
  filter: AdminStateFilter
) {
  if (filter === 'all') return true
  const state = canonicalHealthState(route)
  if (filter === 'evidence_all') {
    return state === 'pending_sample' || state === 'telemetry_unavailable'
  }
  if (filter === 'review_all') {
    return routeNeedsAdministratorAction(route)
  }
  if (filter === 'recovering_all') {
    return state === 'unavailable'
  }
  return state === filter
}

function probeOutcomeLabelKey(outcome?: string) {
  if (!outcome) return null
  if (outcome === 'success') return 'Probe succeeded'
  if (
    outcome === 'infrastructure_failure' ||
    outcome === 'credential_failure'
  ) {
    return 'Probe needs another check'
  }
  return 'Probe completed with a non-success result'
}

function probeDisposition(response?: SmartRouterHealthActionResponse) {
  return String(
    response?.data?.disposition ?? response?.data?.status ?? ''
  ).toLowerCase()
}

function probeNotice(
  disposition: string,
  requestSucceeded: boolean
): ProbeNotice {
  if (disposition === 'queued') {
    return { tone: 'success', labelKey: 'Recovery check queued' }
  }
  if (
    disposition === 'duplicate' ||
    disposition === 'already_queued' ||
    disposition === 'already_running'
  ) {
    return { tone: 'info', labelKey: 'Recovery check queued' }
  }
  if (
    disposition === 'cooldown' ||
    disposition === 'not_due' ||
    disposition === 'scheduled'
  ) {
    return { tone: 'info', labelKey: 'Next check scheduled' }
  }
  if (disposition === 'waiting_for_budget') {
    return { tone: 'info', labelKey: 'Waiting for budget' }
  }
  if (disposition === 'natural_traffic_only') {
    return {
      tone: 'info',
      labelKey: 'Media recovery uses bounded natural traffic',
    }
  }
  if (disposition === 'healthy' || disposition === 'already_healthy') {
    return { tone: 'info', labelKey: 'Healthy' }
  }
  if (disposition === 'queue_full') {
    return {
      tone: 'warning',
      labelKey: 'Recovery queue is full; automatic checks will continue',
    }
  }
  if (disposition === 'worker_unavailable') {
    return {
      tone: 'warning',
      labelKey: 'Background recovery is temporarily unavailable',
    }
  }
  if (disposition === 'requires_reset') {
    return {
      tone: 'warning',
      labelKey: 'Release this route before requesting a check',
    }
  }
  if (requestSucceeded) {
    return { tone: 'success', labelKey: 'Recovery check queued' }
  }
  return { tone: 'error', labelKey: 'Unable to queue recovery check' }
}

function bulkRecoveryNotice(
  response?: SmartRouterBulkRecoveryResponse
): BulkRecoveryNotice {
  const status = response?.data?.status
  const values = {
    queued: response?.data?.queued ?? 0,
    existing: response?.data?.already_queued ?? 0,
    full: response?.data?.queue_full ?? 0,
  }
  if (status === 'queued') {
    return {
      tone: 'success',
      labelKey:
        'Recovery tasks requeued: {{queued}} new, {{existing}} already queued',
      values,
    }
  }
  if (status === 'partially_queued') {
    return {
      tone: 'warning',
      labelKey:
        'Recovery tasks partially requeued: {{queued}} new, {{existing}} already queued, {{full}} waiting for space',
      values,
    }
  }
  if (status === 'already_queued' || status === 'already_scheduled') {
    return {
      tone: 'info',
      labelKey: 'All automatic recovery tasks are already queued',
    }
  }
  if (status === 'nothing_to_requeue') {
    return {
      tone: 'info',
      labelKey: 'No automatic recovery tasks need requeueing',
    }
  }
  if (status === 'queue_full') {
    return {
      tone: 'warning',
      labelKey: 'Recovery queue is full; automatic checks will continue',
    }
  }
  if (status === 'worker_unavailable') {
    return {
      tone: 'warning',
      labelKey: 'Background recovery is temporarily unavailable',
    }
  }
  if (status === 'recovery_disabled') {
    return {
      tone: 'warning',
      labelKey: 'Background probes are currently disabled in site settings.',
    }
  }
  if (status === 'catalog_unavailable') {
    return { tone: 'warning', labelKey: 'Catalog not ready' }
  }
  if (status === 'timeout' || status === 'canceled') {
    return {
      tone: 'warning',
      labelKey:
        'Recovery scheduling did not finish; accepted tasks stay queued',
    }
  }
  return {
    tone: response?.success ? 'info' : 'error',
    labelKey: 'Unable to requeue automatic recovery tasks',
  }
}

function backgroundProbeMessage(meta?: HealthMeta) {
  if (!meta?.shared_state_backend) return null
  if (meta.shared_state_backend !== 'redis') {
    return 'Background probes are unavailable without a shared multi-node lease backend.'
  }
  if (!meta.background_probe_enabled) {
    return 'Background probes are currently disabled in site settings.'
  }
  if (meta.background_probe_available === false) {
    return 'Background recovery worker is not connected.'
  }
  return null
}

type RouteActions = {
  probe: (route: SmartRouterHealthRoute) => void
  confirmBlock: (route: SmartRouterHealthRoute) => void
  probePending: (routeID: string) => boolean
  blockPending: (routeID: string) => boolean
}

function AdminRouteRow(props: {
  route: SmartRouterHealthRoute
  actions: RouteActions
}) {
  const { t } = useTranslation()
  const [open, setOpen] = useState(false)
  const { route, actions } = props
  const state = canonicalHealthState(route)
  const successRate = routeSuccessRate(route)
  const lastReal = formatSmartRouterTimestamp(
    route.last_updated_ms ??
      route.last_success_at_ms ??
      route.last_failure_at_ms
  )
  const lastProbe = formatSmartRouterTimestamp(
    route.last_probe_completed_at_ms ??
      route.last_probe_at_ms ??
      route.last_probe_started_at_ms
  )
  const nextProbe = formatSmartRouterTimestamp(
    route.next_probe_at_ms ?? route.sample_due_at_ms
  )
  const confidenceKey = evidenceConfidenceLabelKey(evidenceConfidence(route))
  const probeOutcomeKey = probeOutcomeLabelKey(route.last_probe_outcome)
  const error = route.sanitized_error || route.last_error || route.last_outcome
  const keyCeiling =
    route.price_known === false
      ? t('Price pending synchronization')
      : ratioText(route.effective_ratio_ppm)
  const modelPrice =
    route.model_price_known === false
      ? t('Price pending synchronization')
      : formatSmartRouterRoutePrice(route.price, t)
  const recentCacheRate = formatActualInputCacheRate(
    route.recent_cache_rate_ppm
  )
  const estimatedInputPrice =
    typeof route.estimated_input_usd_per_million === 'number' &&
    Number.isFinite(route.estimated_input_usd_per_million) &&
    route.estimated_input_usd_per_million >= 0
      ? formatSmartRouterInputCostUSD(route.estimated_input_usd_per_million)
      : ''
  const probeRequestPending = actions.probePending(route.route_id)
  const pending = probeRequestPending || actions.blockPending(route.route_id)
  const canProbe =
    supportsSyntheticProbe(route) &&
    state !== 'healthy' &&
    state !== 'needs_review'
  const routeProbeState = probeState(route)
  const probeRunning =
    routeProbeState === 'active' || routeProbeState === 'running'
  const probeQueued = routeProbeState === 'queued'
  const probeActive = probeRunning || probeQueued
  const probeDisabled = pending || probeActive
  let probeControlLabel = t('Queue recovery probe')
  if (probeRequestPending) {
    probeControlLabel = t('Processing...')
  } else if (probeRunning) {
    probeControlLabel = t('Probe active')
  } else if (probeQueued) {
    probeControlLabel = t('Recovery check queued')
  }
  const probePromoted =
    canProbe && (state === 'unavailable' || state === 'pending_sample')
  let probeMenuItem: ReactNode = null
  if (canProbe) {
    probeMenuItem = (
      <DropdownMenuItem
        disabled={probeDisabled}
        onClick={() => actions.probe(route)}
      >
        {probeRequestPending ? (
          <LoaderCircle className='animate-spin' />
        ) : (
          <Stethoscope />
        )}
        {probeControlLabel}
      </DropdownMenuItem>
    )
  } else if (isMediaRoute(route)) {
    probeMenuItem = (
      <DropdownMenuItem disabled>
        {t('Media recovery uses bounded natural traffic')}
      </DropdownMenuItem>
    )
  }

  return (
    <Collapsible open={open} onOpenChange={setOpen}>
      <div className='relative border-t pl-2 first:border-t-0 sm:pl-5 lg:pl-10'>
        <span
          className={cn(
            'absolute inset-y-2 left-0 w-0.5 rounded-full bg-border',

            </div>

            </div>
            <div>
              <span className='text-muted-foreground'>{t('Probe result')}</span>

                  })}

              </div>
            ) : null}
            {error ? (
              <div className='sm:col-span-2'>
                <span className='text-muted-foreground'>

                </span>
                <p className='mt-0.5 break-words'>
                  {route.last_http_status
                    ? `HTTP ${route.last_http_status} · `
                    : ''}
                  {error}
                </p>
              </div>
            ) : null}
          </div>
        </CollapsibleContent>
      </div>
    </Collapsible>
  )
}

function buildHierarchy(routes: SmartRouterHealthRoute[]) {
  const models = new Map<
    string,
    Map<string, Map<string, SmartRouterHealthRoute[]>>
  >()
  for (const route of routes) {
    const model = route.model || ''
    const endpoint = route.endpoint || ''
    const group = route.group || ''
    let endpoints = models.get(model)
    if (!endpoints) {
      endpoints = new Map()
      models.set(model, endpoints)
    }
    let groups = endpoints.get(endpoint)
    if (!groups) {
      groups = new Map()
      endpoints.set(endpoint, groups)
    }
    const current = groups.get(group)
    if (current) current.push(route)
    else groups.set(group, [route])
  }
  return models
}

function worstState(routes: SmartRouterHealthRoute[]) {
  return canonicalHealthState(
    [...routes].sort(
      (left, right) => healthSortRank(left) - healthSortRank(right)
    )[0]
  )
}

            {t('Recovery and fluctuation')}
          </NativeSelectOption>
          <NativeSelectOption value='review_all'>
            {t('Needs review')}
          </NativeSelectOption>
          <NativeSelectOption value='evidence_all'>
            {t('Awaiting health evidence')}
          </NativeSelectOption>
          {(
            [
              'needs_review',
              'telemetry_unavailable',
              'unavailable',
              'unknown',
              'probing',
              'recovering',
              'degraded_usable',
              'healthy',
              'pending_sample',
            ] as SmartRouterPrimaryState[]
          ).map((state) => (
            <NativeSelectOption key={state} value={state}>
              {t(adminHealthStateLabelKey(state))}
            </NativeSelectOption>
          ))}
        </NativeSelect>
        <NativeSelect
          value={probeFilter}
          onChange={(event) =>
            setProbeFilter(event.target.value as AdminProbeFilter)
          }
          aria-label={t('Probe state')}
          className='min-w-40 flex-1 sm:flex-none'
        >
          <NativeSelectOption value='all'>
            {t('All probe states')}
          </NativeSelectOption>
          <NativeSelectOption value='due'>
            {t('Probe overdue')}
          </NativeSelectOption>
          <NativeSelectOption value='active'>
            {t('Probe active')}
          </NativeSelectOption>
          <NativeSelectOption value='natural'>
            {t('Natural traffic only')}
          </NativeSelectOption>
        </NativeSelect>
        <NativeSelect
          value={endpointFilter}
          onChange={(event) => setEndpointFilter(event.target.value)}
          aria-label={t('Endpoint filter')}
          className='min-w-40 flex-1 sm:flex-none'
        >
          <NativeSelectOption value=''>{t('All endpoints')}</NativeSelectOption>
          {endpoints.map((endpoint) => (
            <NativeSelectOption key={endpoint} value={endpoint}>
              {t(smartRouterEndpointLabelKey(endpoint))}
            </NativeSelectOption>
          ))}
        </NativeSelect>
        <NativeSelect
          value={groupFilter}
          onChange={(event) => setGroupFilter(event.target.value)}
          aria-label={t('Group filter')}
          className='min-w-40 flex-1 sm:flex-none'
        >
          <NativeSelectOption value=''>{t('All groups')}</NativeSelectOption>
          {groups.map((group) => (
            <NativeSelectOption key={group} value={group}>
              {group}
            </NativeSelectOption>
          ))}
        </NativeSelect>
        <Button
          type='button'
          variant={reviewOnly ? 'secondary' : 'outline'}
          aria-pressed={reviewOnly}
          onClick={() => setReviewOnly((current) => !current)}
        >
          {t('Only needs administrator action')}
        </Button>
        {hasActiveFilters ? (
          <Button
            type='button'
            variant='ghost'
            size='icon'
            aria-label={t('Clear filters')}
            title={t('Clear filters')}
            onClick={() => {
              setSearch('')
              setStateFilter('all')
              setProbeFilter('all')
              setEndpointFilter('')
              setGroupFilter('')
              setReviewOnly(false)
            }}
          >
            <X />
          </Button>
        ) : null}
      </div>

      {query.isError ? (
        <Alert>
          <CircleAlert />
          <AlertTitle>{t('Unable to load route health')}</AlertTitle>
          <AlertDescription>
            {t(
              'The health console is temporarily unavailable. API requests continue independently.'
            )}
          </AlertDescription>
        </Alert>
      ) : null}

      {!query.isError && routes.length === 0 ? (
        <div className='text-muted-foreground rounded-lg border border-dashed p-5 text-center text-sm'>
          {t(
            'No route health samples yet. Matching real requests will populate this view.'
          )}
        </div>
      ) : null}

      {routes.length > 0 && filteredRoutes.length === 0 ? (
        <div className='text-muted-foreground rounded-lg border border-dashed p-5 text-center text-sm'>
          {t('No physical routes match these filters.')}
        </div>
      ) : null}

      {visibleRoutes.length > 0 ? (
        <div className='space-y-2'>
          <ScrollArea className='h-[min(68vh,48rem)] min-h-80 rounded-lg border'>
            <div className='space-y-2 p-1 pr-3'>
              {[...hierarchy.entries()].map(([model, endpointMap]) => {
                const modelRoutes = [...endpointMap.values()].flatMap(
                  (groupMap) => [...groupMap.values()].flat()
                )
                const modelState = worstState(modelRoutes)
                return (
                  <section
                    key={model}
                    className='overflow-hidden rounded-lg border'
                  >
                    <GroupDisclosure
                      level='model'
                      defaultOpen={modelState !== 'pending_sample'}
                      forceOpen={forceHierarchyOpen}
                      label={
                        <div className='flex min-w-0 flex-wrap items-center gap-2'>
                          <span className='min-w-0 truncate text-sm font-semibold'>
                            {model || t('Unknown model')}
                          </span>
                          <Badge
                            variant={adminHealthStateBadgeVariant(modelState)}
                          >
                            {t(adminHealthStateLabelKey(modelState))}
                          </Badge>
                        </div>
                      }
                      meta={
                        <span className='text-muted-foreground text-xs'>
                          {t('{{count}} routes', { count: modelRoutes.length })}
                        </span>
                      }
                    >
                      {[...endpointMap.entries()]
                        .sort(([left], [right]) =>
                          compareSmartRouterEndpoints(left, right)
                        )
                        .map(([endpoint, groupMap]) => {
                          const endpointRoutes = [...groupMap.values()].flat()
                          const endpointState = worstState(endpointRoutes)
                          return (
                            <GroupDisclosure
                              key={endpoint}
                              level='endpoint'
                              defaultOpen={endpointState !== 'pending_sample'}
                              forceOpen={forceHierarchyOpen}
                              label={
                                <div className='flex flex-wrap items-center gap-2'>
                                  <Badge variant='outline'>
                                    {t(smartRouterEndpointLabelKey(endpoint))}
                                  </Badge>
                                  <span className='text-muted-foreground text-xs'>
                                    {t('{{count}} groups', {
                                      count: groupMap.size,
                                    })}
                                  </span>
                                </div>
                              }
                            >
                              {[...groupMap.entries()].map(
                                ([group, groupRoutes]) => {
                                  const groupState = worstState(groupRoutes)
                                  return (
                                    <GroupDisclosure
                                      key={group}
                                      level='group'
                                      defaultOpen={
                                        groupState !== 'pending_sample'
                                      }
                                      forceOpen={forceHierarchyOpen}
                                      label={
                                        <div className='flex flex-wrap items-center gap-2'>
                                          <span className='text-sm font-medium'>
                                            {group || t('Unknown group')}
                                          </span>
                                          <Badge
                                            variant={adminHealthStateBadgeVariant(
                                              groupState
                                            )}
                                          >
                                            {t(
                                              adminHealthStateLabelKey(
                                                groupState
                                              )
                                            )}
                                          </Badge>
                                          <span className='text-muted-foreground text-xs'>
                                            {t('{{count}} physical routes', {
                                              count: groupRoutes.length,
                                            })}
                                          </span>
                                        </div>
                                      }
                                    >
                                      <div className='divide-y'>
                                        {groupRoutes.map((route) => (
                                          <AdminRouteRow
                                            key={route.route_id}
                                            route={route}
                                            actions={routeActions}
                                          />
                                        ))}
                                      </div>
                                    </GroupDisclosure>
                                  )
                                }
                              )}
                            </GroupDisclosure>
                          )
                        })}
                    </GroupDisclosure>
                  </section>
                )
              })}
            </div>
          </ScrollArea>

          <div className='flex flex-wrap items-center justify-between gap-2 pt-1'>
            <span className='text-muted-foreground text-xs'>
              {t('Showing {{shown}} of {{total}} routes', {
                shown: visibleRoutes.length,
                total: filteredRoutes.length,
              })}
            </span>
            {visibleRoutes.length < filteredRoutes.length ? (
              <Button
                type='button'
                size='sm'
                variant='outline'
                onClick={() =>
                  setVisibleState({
                    filterKey,
                    limit: visibleLimit + HEALTH_PAGE_SIZE,
                  })
                }
              >
                {t('Show {{count}} more', {
                  count: Math.min(
                    HEALTH_PAGE_SIZE,
                    filteredRoutes.length - visibleRoutes.length
                  ),
                })}
              </Button>
            ) : null}
          </div>
        </div>
      ) : null}

      <AlertDialog
        open={confirmation !== null}
        onOpenChange={(open) => {
          if (!open) setConfirmation(null)
        }}
      >
        <AlertDialogContent>
          <AlertDialogHeader>
            <AlertDialogTitle>
              {confirmation?.kind === 'reset-all'
                ? t('Reset every automatic route state to unknown?')
                : t('Exclude this route from smart routing?')}
            </AlertDialogTitle>
            <AlertDialogDescription>
              {confirmation?.kind === 'reset-all'
                ? t(
                    'This explicit operation clears only automatic health failures and makes those physical routes directly eligible for real traffic again. It preserves manual and model-not-found exclusions and does not queue background probes.'
                  )
                : t(
                    'New smart-routing requests will no longer select this physical route until the exclusion is removed.'
                  )}
            </AlertDialogDescription>
          </AlertDialogHeader>
          <p className='text-muted-foreground font-mono text-xs break-all'>
            {confirmation?.kind === 'reset-all'
              ? t(
                  'No channel, price, authorization, credential, or manual policy is changed.'
                )
              : confirmation?.route?.route_id}
          </p>
          <AlertDialogFooter>
            <AlertDialogCancel>{t('Cancel')}</AlertDialogCancel>
            <AlertDialogAction
              onClick={() => {
                if (!confirmation) return
                if (confirmation.kind === 'reset-all') {
                  resetAllMutation.mutate()
                } else if (confirmation.route) {
                  blockMutation.mutate(confirmation.route.route_id)
                }
                setConfirmation(null)
              }}
            >
              {confirmation?.kind === 'reset-all'
                ? t('Reset automatic health')
                : t('Exclude route')}
            </AlertDialogAction>
          </AlertDialogFooter>
        </AlertDialogContent>
      </AlertDialog>
    </div>
  )
}
