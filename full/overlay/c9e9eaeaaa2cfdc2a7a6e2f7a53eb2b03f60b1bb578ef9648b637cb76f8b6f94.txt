type managerTestRatioSource struct {
	mu       sync.Mutex
	revision uint64
	ratios   map[string]int64
}

type managerTestRecoveryHintSink struct{}

func (managerTestRecoveryHintSink) Hint(SmartRouteRecoveryHint) {}

type managerTestColdStartSuccessSink struct {
	calls        int
	physicalKeys []string
	hints        []SmartRouteRecoveryHint
}

func (sink *managerTestColdStartSuccessSink) Hint(hint SmartRouteRecoveryHint) {
	sink.hints = append(sink.hints, hint)
}

func (sink *managerTestColdStartSuccessSink) ColdStartSucceeded(
	_ smartrouter.RouteRequest,
	physicalKey string,
) {
	sink.calls++
	sink.physicalKeys = append(sink.physicalKeys, physicalKey)
}

func (source *managerTestRatioSource) Revision() uint64 {
	source.mu.Lock()
	defer source.mu.Unlock()
	return source.revision
}

func (source *managerTestRatioSource) RouteRatioPPM(_ string, usingGroup string) (int64, bool) {
	source.mu.Lock()
	defer source.mu.Unlock()
	ratio, exists := source.ratios[usingGroup]
	return ratio, exists
}

func (source *managerTestRatioSource) update(group string, ratio int64) {
	source.mu.Lock()
	source.revision++
	source.ratios[group] = ratio
	source.mu.Unlock()
}

type managerOrderingStateStore struct {
	mu      sync.Mutex
	order   []string
	records []SmartRouteAttemptRecord
	onProbe func()
}

func (store *managerOrderingStateStore) append(value string) {
	store.mu.Lock()
	store.order = append(store.order, value)
	store.mu.Unlock()
}

func (store *managerOrderingStateStore) values() []string {
	store.mu.Lock()
	defer store.mu.Unlock()
	return append([]string(nil), store.order...)
}

func (store *managerOrderingStateStore) attemptRecords() []SmartRouteAttemptRecord {
	store.mu.Lock()
	defer store.mu.Unlock()
	return append([]SmartRouteAttemptRecord(nil), store.records...)
}

func (store *managerOrderingStateStore) Snapshot(
	_ context.Context,
	request SmartRouteStateSnapshotRequest,
) (SmartRouteRuntimeSnapshot, error) {
	quality := make(map[string]smartrouter.QualityState, len(request.Routes))
	capacity := make(map[string]smartrouter.CapacityState, len(request.Routes))
	for _, route := range request.Routes {
		quality[route.RouteID] = smartrouter.InitialQuality(route.StableFallback)
		capacity[route.CapacityDomain] = smartrouter.CapacityState{
			Known: true, MaxInflight: route.MaxInflight,
		}
	}
	return SmartRouteRuntimeSnapshot{
		Quality:  smartrouter.QualitySnapshot{Version: "quality", Routes: quality},
		TTFT:     smartrouter.TTFTSnapshot{Version: "ttft", Routes: map[string]smartrouter.TTFTState{}},
		Capacity: smartrouter.CapacitySnapshot{Version: "capacity", Domains: capacity},
	}, nil
}

func (store *managerOrderingStateStore) TryAcquireProbe(
	_ context.Context,
	_ SmartRouteProbeRequest,
) (*SmartRouteLease, bool, error) {
	store.append("probe")
	if store.onProbe != nil {
		store.onProbe()
	}
	return managerTestLease(func() { store.append("probe_release") }), true, nil
}

func (store *managerOrderingStateStore) TryAcquireCapacity(
	_ context.Context,
	_ SmartRouteCapacityRequest,
) (SmartRouteCapacityAcquisition, error) {
	store.append("capacity")
	return SmartRouteCapacityAcquisition{
		Disposition: SmartRouteCapacityGranted,
		Lease:       managerTestLease(func() { store.append("capacity_release") }),
	}, nil
}

func (store *managerOrderingStateStore) RecordAttempt(_ context.Context, record SmartRouteAttemptRecord) error {
	store.mu.Lock()
	store.order = append(store.order, "record:"+string(record.Admission))
	store.records = append(store.records, record)
	store.mu.Unlock()
	return nil
}

func (store *managerOrderingStateStore) QuarantineRoute(context.Context, SmartRouteQuarantineRequest) error {
	return nil
}

func (store *managerOrderingStateStore) RouteExclusions(
	context.Context,
	SmartRouteExclusionRequest,
) (SmartRouteExclusionSnapshot, error) {
	return SmartRouteExclusionSnapshot{Version: "none"}, nil
}

func managerTestLease(onRelease func()) *SmartRouteLease {
	return &SmartRouteLease{
		ID:          "manager-test-lease",
		expiresAtMS: time.Now().Add(time.Minute).UnixMilli(),
		renewTTL:    time.Minute,
		release: func(context.Context) error {
			onRelease()
			return nil
		},
		renew: func(context.Context, time.Duration) (int64, error) {
			return time.Now().Add(time.Minute).UnixMilli(), nil
		},
	}
}

func managerTestSession(t *testing.T, state SmartRouteStateStore) (*SmartRoutingSession, *managerTestRatioSource) {
	t.Helper()
	contractID, fingerprint, err := BuildSmartContractIdentity("gpt-test", SmartEndpointChat)
	require.NoError(t, err)
	catalog := smartrouter.CertifiedRouteCatalog{
		Version: "manager-catalog",
		Contracts: map[string]smartrouter.ContractCatalog{
			contractID: {
				Contract: smartrouter.RouteContract{
					ContractID: contractID, CanonicalModel: "gpt-test",
					Endpoint: SmartEndpointChat, CapabilityFingerprint: fingerprint,
				},
				Pools: map[string]smartrouter.CertifiedPool{
					"balanced-v1": {Candidates: []smartrouter.CertifiedRoute{
						{
							RouteID: "cheap", Group: "economy", ChannelID: 1, UpstreamModel: "gpt-test",
							Capabilities: smartroutersetting.CapabilityText, MaxContext: 128_000,
							CacheDomain: "cheap-cache", CapacityDomain: "cheap-capacity", MaxInflight: 2,
						},
						{
							RouteID: "fallback", Group: "fallback", ChannelID: 2, UpstreamModel: "gpt-test",
							Capabilities: smartroutersetting.CapabilityText, MaxContext: 128_000,
							CacheDomain: "fallback-cache", CapacityDomain: "fallback-capacity", MaxInflight: 2,
							StableFallback: true,
						},
					}},
				},
			},
		},
	}
	catalogs := &CatalogManager{settings: func() smartroutersetting.Setting {
		return smartroutersetting.Setting{Enabled: true, CatalogRefreshSeconds: 3600}
	}}
	catalogs.current.Store(&catalog)
	catalogs.lastRefresh.Store(time.Now().Unix())
	ratioSource := &managerTestRatioSource{revision: 1, ratios: map[string]int64{
		"economy": 20_000, "fallback": 50_000,
	}}
	prices := NewPriceSnapshotManager(catalogs, ratioSource)

		Version: smartrouter.PolicyVersion, PolicyID: "manager-policy", Enabled: true,
		PoolPolicy: "balanced-v1", OrderMode: smartrouter.OrderAutoPrice,
		MaxEffectiveRatioPPM: 50_000, HealthGuard: true,
		AllowFutureGroups: true,
		RecoveryProfile:   smartrouter.RecoveryBalanced,
	})
	immediateProbe := 0
	setting := smartroutersetting.Setting{
		TextRecoveryProbeAfterSeconds: &immediateProbe,
		Enabled:                       true, MaxAttempts: 3, CatalogRefreshSeconds: 3600,
		KillSwitchMode: smartroutersetting.KillSwitchNormal,
	}
	engine := &SmartRouterEngine{
		Catalogs: catalogs, Prices: prices, State: state,
		Affinity: NewMemorySmartRouteAffinityStore(nil, 100),
		Settings: func() smartroutersetting.Setting { return setting },
	}

		request: smartrouter.RouteRequest{
			ContractID: contractID, CanonicalModel: "gpt-test", Endpoint: SmartEndpointChat,
			CapabilityFingerprint: fingerprint, RequiredCapabilities: smartroutersetting.CapabilityText,

		attempt: smartrouter.AttemptState{MaxAttempts: 3}, blockedRoutes: map[string]struct{}{},
		probeSubject: "token-1",
	}, ratioSource
}

func TestSmartRoutingSessionUsesNormalAdmissionForSafeTextAndRecordsBeforeRelease(t *testing.T) {
	state := &managerOrderingStateStore{}
	session, _ := managerTestSession(t, state)

	require.NoError(t, err)

		Outcome: smartrouter.OutcomeSuccess,
	}, 0, 100*time.Millisecond, true)
	assert.Equal(t, []string{
		"capacity", "record:normal", "capacity_release",
	}, state.values())
}

func TestSmartRoutingSessionMediaVirginRetryRequiresExplicitNonAcceptance(t *testing.T) {
	for _, test := range []struct {
		name        string
		replayClass smartrouter.ReplayClass
		disposition SmartMediaAttemptDisposition
		committed   bool
		wantNext    bool
	}{
		{name: "image explicitly not accepted", replayClass: smartrouter.ReplaySafeImage, disposition: SmartMediaAttemptNotAccepted, wantNext: true},
		{name: "video explicitly not accepted", replayClass: smartrouter.ReplaySafeVideo, disposition: SmartMediaAttemptNotAccepted, wantNext: true},
		{name: "image acceptance unknown", replayClass: smartrouter.ReplaySafeImage, disposition: SmartMediaAttemptAcceptanceUnknown},
		{name: "video acceptance unknown", replayClass: smartrouter.ReplaySafeVideo, disposition: SmartMediaAttemptAcceptanceUnknown},
		{name: "committed image", replayClass: smartrouter.ReplaySafeImage, disposition: SmartMediaAttemptNotAccepted, committed: true},
	} {
		t.Run(test.name, func(t *testing.T) {
			state := &managerOrderingStateStore{}
			session, _ := managerTestSession(t, state)
			session.request.ReplayClass = test.replayClass
			catalog, ok := session.engine.Catalogs.Current()
			require.True(t, ok)
			contract := catalog.Contracts[session.request.ContractID]
			pool := contract.Pools["balanced-v1"]
			pool.Candidates[1].StableFallback = false
			contract.Pools["balanced-v1"] = pool
			catalog.Contracts[session.request.ContractID] = contract

			first, err := session.Next(context.Background())
			require.NoError(t, err)
			require.Equal(t, "cheap", first.Route.RouteID)
			session.RecordAttempt(context.Background(), first, SmartRouteOutcome{
				Outcome: smartrouter.OutcomeInfrastructure, RouteFailure: true,
				RetryNextRoute: true, AffectsRouteHealth: true,
			}, 0, 0, false)

			authorized := session.CanTryNextMedia(SmartRouteOutcome{
				Outcome: smartrouter.OutcomeInfrastructure, RouteFailure: true,
				RetryNextRoute: true, AffectsRouteHealth: true,
			}, test.committed, test.disposition)
			assert.Equal(t, test.wantNext, authorized)
			if !test.wantNext {
				assert.False(t, session.attempt.VirginMediaRetryAuthorized)
				return
			}

			second, err := session.Next(context.Background())
			require.NoError(t, err)
			require.Equal(t, "fallback", second.Route.RouteID)
			assert.Equal(t, smartrouter.AdmissionNormal, second.Plan.Admission)
			assert.False(t, session.attempt.VirginMediaRetryAuthorized, "the grant must be consumed when the next attempt starts")
			second.stopAndRelease()
		})
	}
}

func TestSmartRoutingSessionWaitsForOrderedColdStartDiscoveryThenReplans(t *testing.T) {
	state, err := NewMemorySmartRouteStateStore(DefaultSmartRouteStateConfig())
	require.NoError(t, err)
	session, _ := managerTestSession(t, state)
	catalog, ok := session.engine.Catalogs.Current()
	require.True(t, ok)
	contract := catalog.Contracts[session.request.ContractID]
	pool := contract.Pools["balanced-v1"]
	pool.Candidates[1].StableFallback = false
	contract.Pools["balanced-v1"] = pool
	catalog.Contracts[session.request.ContractID] = contract
	setting := smartroutersetting.Normalize(smartroutersetting.Setting{
		Enabled: true, BackgroundProbeEnabled: true, MaxAttempts: 3,
		CatalogRefreshSeconds: 3600, KillSwitchMode: smartroutersetting.KillSwitchNormal,
		ColdStartWaitMS: 500, ColdStartPollIntervalMS: 10,
	})
	session.engine.Settings = func() smartroutersetting.Setting { return setting }
	session.engine.SetRecoveryHintSink(managerTestRecoveryHintSink{})

	first, err := session.Next(context.Background())
	require.NoError(t, err)
	require.Equal(t, "cheap", first.Route.RouteID)
	session.RecordAttempt(context.Background(), first, SmartRouteOutcome{
		Outcome: smartrouter.OutcomeInfrastructure, RouteFailure: true,
		RetryNextRoute: true, AffectsRouteHealth: true,
	}, 0, 0, false)

	type nextResult struct {
		handle *SmartRouteAttemptHandle
		err    error
	}
	next := make(chan nextResult, 1)
	go func() {
		handle, nextErr := session.Next(context.Background())
		next <- nextResult{handle: handle, err: nextErr}
	}()
	time.Sleep(30 * time.Millisecond)
	require.NoError(t, state.RecordAttempt(context.Background(), SmartRouteAttemptRecord{
		RouteID: "fallback", Group: "fallback", Model: session.request.CanonicalModel,
		ChannelID: 2, ContractID: session.request.ContractID, Endpoint: session.request.Endpoint,
		CapacityDomain: "fallback-capacity", StableFallback: false,
		Outcome: smartrouter.OutcomeSuccess, Admission: smartrouter.AdmissionVirginBootstrap,
		RouteEpoch: 1,
	}))
	notifySmartRouteDiscovery(session.request)

	select {
	case result := <-next:
		require.NoError(t, result.err)
		require.NotNil(t, result.handle)
		assert.Equal(t, "fallback", result.handle.Route.RouteID)
		assert.Equal(t, smartrouter.AdmissionNormal, result.handle.Plan.Admission)
		result.handle.stopAndRelease()
	case <-time.After(time.Second):
		require.Fail(t, "cold-start discovery did not wake and replan the request")
	}
}

		capacity: &SmartRouteLease{renewTTL: 2 * time.Minute},
	}
	assert.Equal(t, 10*time.Second/3, smartRouteLeaseHeartbeatIntervalFor(handle))

	handle.probe.renewTTL = 60 * time.Second
	assert.Equal(t, smartRouteLeaseHeartbeatInterval, smartRouteLeaseHeartbeatIntervalFor(handle),
		"long-lived leases retain the 15-second ceiling")
}

			},
		},
		{
			name: "capacity failure keeps capacity accounting",

func TestSmartRoutingSessionDiscardsUnusedPlanWhenPriceChangesBeforeDispatch(t *testing.T) {
	state := &managerOrderingStateStore{}
	session, ratios := managerTestSession(t, state)

	require.NoError(t, err)
	require.Equal(t, "cheap", handle.Route.RouteID)
	require.Equal(t, 1, session.attempt.StartedAttempts)

	// The planner selected economy at 0.02x. A control-plane hot update lands
	// before billing reservation or upstream dispatch and makes fallback cheaper.
	ratios.update("economy", 40_000)
	ratios.update("fallback", 30_000)
	changed, err := session.ReplanIfPriceQuoteChanged(context.Background(), handle, "")
	require.NoError(t, err)
	require.True(t, changed)

	require.NoError(t, err)
	assert.Equal(t, "fallback", replanned.Route.RouteID)
	assert.Equal(t, int64(30_000), replanned.Plan.RatioPPM)
	assert.Equal(t, 1, session.attempt.StartedAttempts,
		"a price-only replan must not consume the upstream attempt budget")
	replanned.stopAndRelease()
}

func TestSmartRoutingSessionExhaustsFinitePhysicalSetBeyondNumericLimit(t *testing.T) {
	state := &managerOrderingStateStore{}
	session, ratios := managerTestSession(t, state)
	catalog, ok := session.engine.Catalogs.Current()
	require.True(t, ok)
	updated := *catalog
	updated.Contracts = make(map[string]smartrouter.ContractCatalog, len(catalog.Contracts))
	for contractID, contract := range catalog.Contracts {
		cloned := contract
		cloned.Pools = make(map[string]smartrouter.CertifiedPool, len(contract.Pools))
		for poolID, pool := range contract.Pools {
			pool.Candidates = append([]smartrouter.CertifiedRoute(nil), pool.Candidates...)
			cloned.Pools[poolID] = pool
		}
		updated.Contracts[contractID] = cloned
	}
	contract := updated.Contracts[session.request.ContractID]
	pool := contract.Pools["balanced-v1"]
	for channelID := 3; channelID <= 10; channelID++ {
		group := fmt.Sprintf("route-%d", channelID)
		pool.Candidates = append(pool.Candidates, smartrouter.CertifiedRoute{
			RouteID: group, Group: group, ChannelID: channelID, UpstreamModel: "gpt-test",
			Capabilities: smartroutersetting.CapabilityText, MaxContext: 128_000,
			CacheDomain: group + "-cache", CapacityDomain: group + "-capacity", MaxInflight: 2,
			StableFallback: true,
		})
		session.allowedGroups = append(session.allowedGroups, group)
		ratios.update(group, int64(20_000+channelID))
	}
	contract.Pools["balanced-v1"] = pool
	updated.Contracts[session.request.ContractID] = contract
	session.engine.Catalogs.current.Store(&updated)
	session.engine.Prices.InvalidateAll()
	session.attempt.MaxAttempts = 8
	session.attempt.ExhaustEligibleChannels = true

	var attempted []int
	for attempt := range 10 {
		handle, err := session.Next(context.Background())
		require.NoErrorf(t, err, "attempt %d must still have an eligible physical channel", attempt+1)
		attempted = append(attempted, handle.Route.ChannelID)
		session.RecordAttempt(context.Background(), handle, SmartRouteOutcome{
			Outcome: smartrouter.OutcomeInfrastructure, RouteFailure: true, RetryNextRoute: true,
		}, 0, 0, false)
	}
	assert.Len(t, attempted, 10)
	assert.Len(t, session.attempt.AttemptedChannels, 10)
	uniqueChannels := make(map[int]struct{}, len(attempted))
	for _, channelID := range attempted {
		uniqueChannels[channelID] = struct{}{}
	}
	assert.Len(t, uniqueChannels, 10)
	_, err := session.Next(context.Background())
	require.ErrorIs(t, err, smartrouter.ErrNoCandidate)
}

func TestSmartRouteCapacityOverflowDoesNotReplaceStrongAffinity(t *testing.T) {
	store := NewMemorySmartRouteAffinityStore(nil, 100)
	const affinityKey = "affinity-key"
	original := smartrouter.AffinityState{
		Strength: smartrouter.AffinityStrong, ContractID: "contract", RouteID: "original",
		ChannelID: 1, CacheDomain: "cache-a", PolicyID: "policy",
		LastSuccessMS: 100, LastSwitchMS: 90, LastActivityMS: 100,
	}
	require.NoError(t, store.Put(context.Background(), affinityKey, original, time.Hour))
	session := &SmartRoutingSession{
		engine: &SmartRouterEngine{Affinity: store}, affinityKey: affinityKey,
		affinity: &original, request: smartrouter.RouteRequest{ContractID: "contract"},
		policy: smartrouter.Policy{PolicyID: "policy"},
	}
	session.recordAffinity(&SmartRouteAttemptHandle{
		Plan:  smartrouter.PlanResult{AffinityDisposition: smartrouter.AffinityPreserve, CapacityOverflow: true},
		Route: smartrouter.CertifiedRoute{RouteID: "overflow", ChannelID: 2, CacheDomain: "cache-b"},
	})

	stored, found, err := store.Get(context.Background(), affinityKey)
	require.NoError(t, err)
	require.True(t, found)
	assert.Equal(t, "original", stored.RouteID)
	assert.Equal(t, 1, stored.ChannelID)

	assert.Greater(t, stored.LastActivityMS, original.LastActivityMS)
