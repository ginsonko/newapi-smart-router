package controller

import (
	"net/http"
	"testing"

	"github.com/QuantumNous/new-api/model"
	"github.com/stretchr/testify/require"
)

func TestTokenSecurityOwnershipAndGroupMutation(t *testing.T) {
	database := setupTokenControllerTestDB(t)
	require.NoError(t, database.Exec("CREATE TABLE users (id integer PRIMARY KEY, `group` text, deleted_at datetime)").Error)
	require.NoError(t, database.Exec("INSERT INTO users(id, `group`) VALUES (1, 'default'), (2, 'default')").Error)
	token := model.Token{UserId: 1, Key: "isolated-test-key", Name: "keep", Group: "auto", ExpiredTime: -1, UnlimitedQuota: true}
	require.NoError(t, database.Create(&token).Error)
	for _, userID := range []int{2, 0, 1} {
		body := tokenMutationBody(&token, "private-security-test-group", false)
		body["user_id"] = 1
		ctx, recorder := newAuthenticatedContext(t, http.MethodPut, "/api/token/", body, userID)
		UpdateToken(ctx)
		require.False(t, decodeAPIResponse(t, recorder).Success)
		var stored model.Token
		require.NoError(t, database.First(&stored, token.Id).Error)
		require.Equal(t, "auto", stored.Group)
		require.Equal(t, 1, stored.UserId)
	}
	body := tokenMutationBody(&token, "default", false)
	ctx, recorder := newAuthenticatedContext(t, http.MethodPut, "/api/token/", body, 1)
	UpdateToken(ctx)
	require.True(t, decodeAPIResponse(t, recorder).Success)
	create := tokenMutationBody(&model.Token{Name: "denied", ExpiredTime: -1, UnlimitedQuota: true}, "private-security-test-group", false)
	ctx, recorder = newAuthenticatedContext(t, http.MethodPost, "/api/token/", create, 1)
	AddToken(ctx)
	require.False(t, decodeAPIResponse(t, recorder).Success)
	var count int64
	require.NoError(t, database.Model(&model.Token{}).Count(&count).Error)
	require.EqualValues(t, 1, count)
}
