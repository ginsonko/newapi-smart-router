func TestNormalizeMediaStudioResponseFormatByMode(t *testing.T) {
	gin.SetMode(gin.TestMode)

	newContext := func(t *testing.T) *gin.Context {
		var body bytes.Buffer
		writer := multipart.NewWriter(&body)
		require.NoError(t, writer.WriteField("model", "gpt-image-2"))
		require.NoError(t, writer.WriteField("prompt", "edit this image"))
		require.NoError(t, writer.WriteField("response_format", "mp3"))
		part, err := writer.CreateFormFile("image", "input.png")
		require.NoError(t, err)
		_, err = part.Write([]byte("image bytes"))
		require.NoError(t, err)
		require.NoError(t, writer.Close())

		context, _ := gin.CreateTestContext(httptest.NewRecorder())
		context.Request = httptest.NewRequest(http.MethodPost, "/api/media/studio/image-edits", &body)
		context.Request.Header.Set("Content-Type", writer.FormDataContentType())
		return context
	}

	for _, test := range []struct {
		mode string
		want string
	}{
		{mode: "image-edits", want: "url"},
		{mode: "transcriptions", want: "json"},
		{mode: "translations", want: "json"},
	} {
		t.Run(test.mode, func(t *testing.T) {
			context := newContext(t)
			require.NoError(t, normalizeMediaStudioResponseFormat(context, test.mode))
			require.NotNil(t, context.Request.MultipartForm)
			assert.Equal(t, test.want, context.Request.PostForm.Get("response_format"))
			assert.Equal(t, test.want, context.Request.MultipartForm.Value["response_format"][0])
			require.Len(t, context.Request.MultipartForm.File["image"], 1)
			file, err := context.Request.MultipartForm.File["image"][0].Open()
			require.NoError(t, err)
			defer file.Close()
			content, err := io.ReadAll(file)
			require.NoError(t, err)
			assert.Equal(t, []byte("image bytes"), content)
		})
	}

	t.Run("speech remains client selected", func(t *testing.T) {
		context := newContext(t)
		require.NoError(t, normalizeMediaStudioResponseFormat(context, "speech"))
		assert.Nil(t, context.Request.MultipartForm)
	})
}

func TestMediaStudioOperationIDIsStrictAndPrivateToImageStudio(t *testing.T) {
	gin.SetMode(gin.TestMode)
	valid := "studio_3f37de08-c768-4ab8-a2ec-fd33d6f4d5ee"

	for _, test := range []struct {
		name    string
		value   string
		want    string
		wantErr bool
	}{
		{name: "absent remains compatible"},
		{name: "canonical UUID", value: valid, want: valid},
		{name: "missing prefix", value: strings.TrimPrefix(valid, mediaStudioOperationPrefix), wantErr: true},
		{name: "upper case is not canonical", value: strings.ToUpper(valid), wantErr: true},
		{name: "invalid UUID", value: "studio_not-a-uuid", wantErr: true},
	} {
		t.Run(test.name, func(t *testing.T) {
			result, err := validateMediaStudioOperationID(test.value)
			if test.wantErr {
				assert.Error(t, err)
				return
			}
			require.NoError(t, err)
			assert.Equal(t, test.want, result)
		})
	}

	newContext := func(mode, operationID string) *gin.Context {
		context, _ := gin.CreateTestContext(httptest.NewRecorder())
		context.Request = httptest.NewRequest(http.MethodPost, "/api/media/studio/"+mode, nil)
		context.Request.Header.Set(mediaStudioOperationHeader, operationID)
		return context
	}

	context := newContext("images", valid)
	require.NoError(t, setMediaStudioOperationID(context, "images"))
	stored, exists := context.Get(mediaStudioOperationContextKey)
	require.True(t, exists)
	assert.Equal(t, valid, stored)

	context = newContext("videos", valid)
	require.NoError(t, setMediaStudioOperationID(context, "videos"))
	stored, exists = context.Get(mediaStudioOperationContextKey)
	assert.True(t, exists, "video children need a stable idempotency identity")
	assert.Equal(t, valid, stored)

	context = newContext("image-edits", "studio_not-a-uuid")
	assert.Error(t, setMediaStudioOperationID(context, "image-edits"))
	_, exists = context.Get(mediaStudioOperationContextKey)
	assert.False(t, exists)
}

func TestMediaStudioFallbackGroupsRemainOrderedAndUserScoped(t *testing.T) {
	usable := map[string]string{"cheap": "1", "plus": "2", "primary": "3"}

	groups, err := resolveMediaStudioFallbackGroups(
		`["cheap","plus","cheap"]`, "primary", usable,
	)
	require.NoError(t, err)
	assert.Equal(t, []string{"cheap", "plus", "primary"}, groups)

	groups, err = resolveMediaStudioFallbackGroups("", "primary", usable)
	require.NoError(t, err)
	assert.Nil(t, groups, "an ordinary fixed-group studio request must remain fixed")

	_, err = resolveMediaStudioFallbackGroups(`["cheap","forbidden"]`, "primary", usable)
	assert.ErrorContains(t, err, "not available")
	_, err = resolveMediaStudioFallbackGroups(`not-json`, "primary", usable)
	assert.ErrorContains(t, err, "invalid media studio fallback")
}

func TestMediaResponseCaptureKeepsOrdinaryJSONBufferedAfterWriteHeaderNow(t *testing.T) {
	gin.SetMode(gin.TestMode)
	recorder := httptest.NewRecorder()
	context, _ := gin.CreateTestContext(recorder)
	file, err := os.CreateTemp(t.TempDir(), "capture-*")
	require.NoError(t, err)
	defer file.Close()
	writer := &mediaResponseCaptureWriter{
		ResponseWriter: context.Writer, file: file, maxBytes: 1024, hold: true,
	}
	context.Writer = writer
	context.Header("Content-Type", "application/json")
	context.Status(http.StatusOK)
	context.Writer.WriteHeaderNow()

	_, err = context.Writer.Write([]byte(`{"data":[{"url":"https://provider.example/image.png"}]}`))
	require.NoError(t, err)
	assert.True(t, writer.hold)
	assert.Empty(t, recorder.Body.String(), "ordinary image JSON must remain rewritable until relay completion")

	require.NoError(t, writer.commitBuffered([]byte(`{"data":[{"asset_id":"asset_1"}]}`), true))
	assert.JSONEq(t, `{"data":[{"asset_id":"asset_1"}]}`, recorder.Body.String())
}

func TestMediaDownloadAssetArchiveCreatesOneOwnedZipWithFailureManifest(t *testing.T) {
	gin.SetMode(gin.TestMode)
	previousDB := model.DB
	db, err := gorm.Open(sqlite.Open("file:media-archive?mode=memory&cache=shared"), &gorm.Config{})
	require.NoError(t, err)
	model.DB = db
	t.Cleanup(func() { model.DB = previousDB })
	require.NoError(t, db.AutoMigrate(
		&model.MediaBlob{}, &model.TaskArtifact{}, &model.MediaStorageState{},
		&model.MediaStorageReservation{},
	))
	previousSetting := mediasetting.Get()
	nextSetting := previousSetting
	nextSetting.StoragePath = t.TempDir()
	nextSetting.ReservedFreeBytes = 1
	require.NoError(t, mediasetting.Replace(nextSetting))
	t.Cleanup(func() { require.NoError(t, mediasetting.Replace(previousSetting)) })
	require.NoError(t, mediaservice.EnsureStorage())

	blobs := []model.MediaBlob{
		{PublicID: "owned-a", UserID: 7, Filename: "图片.png", MediaType: "image", MimeType: "image/png", StorageBackend: model.MediaStorageBackendLocal, ObjectKey: filepath.Join("objects", "owned-a.png"), SizeBytes: 3, Status: model.MediaBlobStatusReady, ModerationStatus: model.MediaModerationApproved},
		{PublicID: "owned-b", UserID: 7, Filename: "图片.png", MediaType: "image", MimeType: "image/png", StorageBackend: model.MediaStorageBackendLocal, ObjectKey: filepath.Join("objects", "owned-b.png"), SizeBytes: 4, Status: model.MediaBlobStatusReady, ModerationStatus: model.MediaModerationApproved},
		{PublicID: "foreign", UserID: 8, Filename: "foreign.png", MediaType: "image", MimeType: "image/png", StorageBackend: model.MediaStorageBackendLocal, ObjectKey: filepath.Join("objects", "foreign.png"), SizeBytes: 5, Status: model.MediaBlobStatusReady, ModerationStatus: model.MediaModerationApproved},
	}
	for index := range blobs {
		path, pathErr := mediaservice.ResolveObjectPath(blobs[index].ObjectKey)
		require.NoError(t, pathErr)
		require.NoError(t, os.WriteFile(path, bytes.Repeat([]byte{byte(index + 1)}, int(blobs[index].SizeBytes)), 0o600))
	}
	require.NoError(t, db.Create(&blobs).Error)

	body, err := json.Marshal(map[string]any{"asset_ids": []string{"owned-a", "owned-b", "foreign"}})
	require.NoError(t, err)
	recorder := httptest.NewRecorder()
	context, _ := gin.CreateTestContext(recorder)
	context.Request = httptest.NewRequest(http.MethodPost, "/api/media/assets/download-archive", bytes.NewReader(body))
	context.Request.Header.Set("Content-Type", "application/json")
	context.Set("id", 7)

	MediaDownloadAssetArchive(context)

	require.Equal(t, http.StatusOK, recorder.Code)
	assert.Equal(t, "application/zip", recorder.Header().Get("Content-Type"))
	archive, err := zip.NewReader(bytes.NewReader(recorder.Body.Bytes()), int64(recorder.Body.Len()))
	require.NoError(t, err)
	names := make([]string, 0, len(archive.File))
	contents := make(map[string][]byte, len(archive.File))
	for _, file := range archive.File {
		names = append(names, file.Name)
		entry, openErr := file.Open()
		require.NoError(t, openErr)
		data, readErr := io.ReadAll(entry)
		require.NoError(t, readErr)
		require.NoError(t, entry.Close())
		contents[file.Name] = data
	}
	assert.Contains(t, names, "图片.png")
	assert.Contains(t, names, "图片 (2).png")
	assert.Contains(t, names, "下载说明.txt")
	assert.NotContains(t, names, "foreign.png")
	assert.Equal(t, bytes.Repeat([]byte{1}, 3), contents["图片.png"])
	assert.Equal(t, bytes.Repeat([]byte{2}, 4), contents["图片 (2).png"])
	assert.Contains(t, string(contents["下载说明.txt"]), "foreign")
	var storageState model.MediaStorageState
	require.NoError(t, db.First(&storageState, 1).Error)
	assert.Zero(t, storageState.ReservedBytes, "ZIP and remote-source spool reservations must be released")
}
