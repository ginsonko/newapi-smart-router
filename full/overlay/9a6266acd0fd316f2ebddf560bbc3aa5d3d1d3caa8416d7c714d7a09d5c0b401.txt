func TestResponsesToolCounterFinalOutputAndDeduplication(t *testing.T) {
	for _, toolType := range []string{"web_search", dto.BuildInToolWebSearchPreview} {
		t.Run(toolType, func(t *testing.T) {
			info := &relaycommon.RelayInfo{ResponsesUsageInfo: &relaycommon.ResponsesUsageInfo{BuiltInTools: map[string]*relaycommon.BuildInToolInfo{toolType: {ToolName: toolType}}}}
			counter := newResponsesToolCounter(info)
			first := dto.ResponsesOutput{Type: "web_search_call", ID: "search1", Status: "completed"}
			counter.observe(first, nil)
			counter.observe(first, nil)
			output := []dto.ResponsesOutput{first, {Type: "web_search_call", ID: "search2", Status: "completed"}, {Type: "file_search_call", ID: "file1", Status: "completed"}, {Type: "function_call", ID: "function1", Status: "completed"}, {Type: "web_search_call", ID: "failed", Status: "failed"}}
			counter.observeOutputs(output)
			counter.observeOutputs(output)
			require.Equal(t, 2, info.ResponsesUsageInfo.BuiltInTools[toolType].CallCount)
			require.Equal(t, 1, info.ResponsesUsageInfo.BuiltInTools[dto.BuildInToolFileSearch].CallCount)
			require.Len(t, info.ResponsesUsageInfo.BuiltInTools, 2)
		})
	}
	info := &relaycommon.RelayInfo{}
	counter := newResponsesToolCounter(info)
	index := 0
	counter.observe(dto.ResponsesOutput{Type: "web_search_call", Status: "completed"}, &index)
	counter.observeOutputs([]dto.ResponsesOutput{{Type: "web_search_call", ID: "late-id", Status: "completed"}})
	require.Equal(t, 1, info.ResponsesUsageInfo.BuiltInTools["web_search"].CallCount)
}

func TestResponsesHandlerCountsCallsInsteadOfToolDeclarations(t *testing.T) {
	for _, example := range []struct {
		output string
		count  int
	}{
		{`[]`, 0},
		{`[{"type":"web_search_call","id":"a","status":"completed"},{"type":"web_search_call","id":"b","status":"completed"}]`, 2},
	} {
		context, _ := gin.CreateTestContext(httptest.NewRecorder())
		info := &relaycommon.RelayInfo{ResponsesUsageInfo: &relaycommon.ResponsesUsageInfo{BuiltInTools: map[string]*relaycommon.BuildInToolInfo{"web_search": {ToolName: "web_search"}}}}
		info.ChannelMeta = &relaycommon.ChannelMeta{}
		body := `{"status":"completed","tools":[{"type":"web_search"}],"output":` + example.output + `,"usage":{"input_tokens":10,"output_tokens":3,"total_tokens":13}}`
		_, err := OaiResponsesHandler(context, info, &http.Response{StatusCode: 200, Header: http.Header{}, Body: io.NopCloser(strings.NewReader(body))})
		require.Nil(t, err)
		require.Equal(t, example.count, info.ResponsesUsageInfo.BuiltInTools["web_search"].CallCount)
	}
}
