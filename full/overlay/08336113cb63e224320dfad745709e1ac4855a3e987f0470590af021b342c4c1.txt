func setupTenantCostTestDB(t *testing.T) *gorm.DB {
	t.Helper()
	previousDB := DB
	previousMain := common.MainDatabaseType()
	dsn := fmt.Sprintf("file:tenant-cost-%s?mode=memory&cache=shared", strings.ReplaceAll(t.Name(), "/", "_"))
	db, err := gorm.Open(sqlite.Open(dsn), &gorm.Config{})
	require.NoError(t, err)
	DB = db
	common.SetMainDatabaseType(common.DatabaseTypeSQLite)
	t.Cleanup(func() {
		DB = previousDB
		common.SetMainDatabaseType(previousMain)
	})
	require.NoError(t, db.AutoMigrate(
		&TenantCostDownstream{}, &TenantCostProductSlot{}, &TenantCostSlotProvisionReceipt{}, &TenantCostKeyRotationReceipt{},
		&TenantCredentialBinding{}, &TenantRouteProfile{}, &TenantCostSequence{},
		&TenantCostEvent{}, &TenantCostOutbox{}, &TenantReportingToken{}, &TenantCostAuditEvent{}, &TenantCostProjectionRetry{},
		&TenantCostWorkerControl{},
	))
	require.NoError(t, EnsureDefaultTenantCostDownstream())
	return db
}

func TestTenantCostEventReplayIgnoresReconstructedTerminalTimes(t *testing.T) {
	setupTenantCostTestDB(t)
	spec := tenantCostTestSpec("uce_time_replay")
	first, err := AppendTenantCostEvent(spec)
	require.NoError(t, err)

	// A worker restart can rebuild the same terminal fact with new wall-clock
	// timestamps.  The durable event ID and all financial/correlation fields
	// still identify one fact, so this must replay the original row rather than
	// create a conflict or a second outbox item.
	retry := spec
	retry.OccurredAt = spec.OccurredAt.Add(17 * time.Minute)
	retry.FinalizedAt = spec.FinalizedAt.Add(19 * time.Minute)
	second, err := AppendTenantCostEvent(retry)
	require.NoError(t, err)
	assert.True(t, second.Replayed)
	assert.Equal(t, first.Event.Sequence, second.Event.Sequence)

	conflict := retry
	conflict.GatewayChargeQuota++
	_, err = AppendTenantCostEvent(conflict)
	assert.ErrorIs(t, err, ErrTenantCostConflict)
}

func TestTenantCostOutboxClaimRetryAndSent(t *testing.T) {
	db := setupTenantCostTestDB(t)
	_, err := AppendTenantCostEvent(tenantCostTestSpec("uce_outbox"))
	require.NoError(t, err)
	now := time.Now().UTC()
	rows, err := ClaimTenantCostOutbox(10, now.UnixNano())
	require.NoError(t, err)
	require.Len(t, rows, 1)
	assert.Equal(t, TenantOutboxPending, rows[0].State)
	require.NoError(t, MarkTenantCostOutboxFailed(rows[0].EventID, fmt.Errorf("temporary"), now.UnixNano()))
	var failed TenantCostOutbox
	require.NoError(t, db.Where("event_id = ?", rows[0].EventID).First(&failed).Error)
	assert.Equal(t, TenantOutboxFailed, failed.State)
	// A failed row is not immediately eligible; forcing its due time models a
	// worker restart after the bounded backoff.
	require.NoError(t, db.Model(&TenantCostOutbox{}).Where("id = ?", failed.ID).Updates(map[string]interface{}{
		"next_attempt_at": now.Add(-time.Second).UnixNano(),
	}).Error)
	rows, err = ClaimTenantCostOutbox(10, now.Add(time.Second).UnixNano())
	require.NoError(t, err)
	require.Len(t, rows, 1)
	require.NoError(t, MarkTenantCostOutboxSent(rows[0].EventID, now.Add(2*time.Second).UnixNano()))
	var sent TenantCostOutbox
	require.NoError(t, db.Where("event_id = ?", rows[0].EventID).First(&sent).Error)
	assert.Equal(t, TenantOutboxSent, sent.State)
}

func TestTenantCostProjectionRetryQueueIsBoundedAndIdempotent(t *testing.T) {
	db := setupTenantCostTestDB(t)
	spec := tenantCostTestSpec("uce_projection_retry")
	require.NoError(t, EnqueueTenantCostProjection(spec, fmt.Errorf("temporary database error")))
	// Re-enqueueing the same redacted spec must not create a second row.
	require.NoError(t, EnqueueTenantCostProjection(spec, fmt.Errorf("same transient error")))
	var count int64
	require.NoError(t, db.Model(&TenantCostProjectionRetry{}).Count(&count).Error)
	assert.EqualValues(t, 1, count)
	rows, err := ClaimTenantCostProjectionRetries(10, time.Now().UTC().UnixNano())
	require.NoError(t, err)
	require.Len(t, rows, 1)
	decoded, err := DecodeTenantCostProjectionRetry(rows[0])
	require.NoError(t, err)
	assert.Equal(t, spec.EventID, decoded.EventID)
	require.NoError(t, MarkTenantCostProjectionRetrySent(spec.EventID, time.Now().UTC().UnixNano()))
	var row TenantCostProjectionRetry
	require.NoError(t, db.Where("event_id = ?", spec.EventID).First(&row).Error)
	assert.Equal(t, TenantCostProjectionSent, row.State)

	conflict := spec
	conflict.GatewayChargeQuota++
	assert.ErrorIs(t, EnqueueTenantCostProjection(conflict, nil), ErrTenantCostConflict)
}

func TestDecodeTenantCostProjectionRetryRejectsCrossTenantPayload(t *testing.T) {
	db := setupTenantCostTestDB(t)
	second := &TenantCostDownstream{TenantID: "retry-payload-tenant", DisplayName: "Retry Payload Tenant", ConfigKey: "retry_payload_tenant", Status: TenantCostDownstreamActive}
	require.NoError(t, UpsertTenantCostDownstream(second))

	rowSpec := tenantCostTestSpec("cross-tenant-payload")
	require.NoError(t, EnqueueTenantCostProjection(rowSpec, fmt.Errorf("temporary")))
	var row TenantCostProjectionRetry
	require.NoError(t, db.Where("tenant_id = ? AND event_id = ?", DownstreamTenantID, rowSpec.EventID).First(&row).Error)

	// Simulate a tampered durable row whose payload and hash are internally
	// consistent for another registered tenant.  The row namespace must still
	// win and fail closed before any append/replay decision is made.
	forged := rowSpec
	forged.TenantID = second.TenantID
	normalized, err := forged.normalize()
	require.NoError(t, err)
	forgedPayload := tenantCostSemanticPayload(normalized)
	forgedHash, _, err := tenantCostPayloadHash(forgedPayload)
	require.NoError(t, err)
	forgedJSON, err := common.Marshal(normalized)
	require.NoError(t, err)
	require.NoError(t, db.Model(&TenantCostProjectionRetry{}).Where("id = ?", row.ID).Updates(map[string]interface{}{
		"spec_json": string(forgedJSON), "spec_hash": forgedHash,
	}).Error)

	row.SpecJSON = string(forgedJSON)
	row.SpecHash = forgedHash
	_, err = DecodeTenantCostProjectionRetry(row)
	assert.ErrorIs(t, err, ErrTenantCostScope)
	var events int64
	require.NoError(t, db.Model(&TenantCostEvent{}).Count(&events).Error)
	assert.Zero(t, events)
}

func TestTenantReportingTokenExpiryAndRevocation(t *testing.T) {
	setupTenantCostTestDB(t)
	now := common.GetTimestamp()
	row, secret, err := CreateTenantReportingToken(DownstreamTenantID, TenantCostReportingPermission, now+60)
	require.NoError(t, err)
	got, err := LookupTenantReportingToken(secret)
	require.NoError(t, err)
	assert.Equal(t, row.TokenID, got.TokenID)
	// Expiry and revocation are checked against persisted state, not a client
	// supplied tenant header.
	require.NoError(t, DB.Model(&TenantReportingToken{}).Where("id = ?", row.ID).Updates(map[string]interface{}{"expires_at": now - 1}).Error)
	_, err = LookupTenantReportingToken(secret)
	assert.ErrorIs(t, err, ErrTenantCostReportingToken)
	require.NoError(t, DB.Model(&TenantReportingToken{}).Where("id = ?", row.ID).Updates(map[string]interface{}{"expires_at": now + 60, "revoked_at": now}).Error)
	_, err = LookupTenantReportingToken(secret)
	assert.ErrorIs(t, err, ErrTenantCostReportingToken)
}
