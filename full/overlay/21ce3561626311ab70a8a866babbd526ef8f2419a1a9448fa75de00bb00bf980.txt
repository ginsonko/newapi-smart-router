func setupRouteAnalyticsTestDB(t *testing.T) *gorm.DB {
	t.Helper()
	previousDB := DB
	previousLogDB := LOG_DB
	dsn := fmt.Sprintf("file:route-analytics-%d?mode=memory&cache=shared&_busy_timeout=30000", time.Now().UnixNano())
	db, err := gorm.Open(sqlite.Open(dsn), &gorm.Config{})
	require.NoError(t, err)
	require.NoError(t, db.AutoMigrate(
		&RouteAnalyticsDimensionV1{},
		&RouteAnalyticsObservationV1{},
		&RouteAnalyticsBucketV1{},
		&KeyUsageAnalyticsBucketV1{},
		&RouteAnalyticsHistoryV1{},
		&RouteAnalyticsWorkerStateV1{},
	))
	DB = db
	LOG_DB = db
	common.SetDatabaseTypes(common.DatabaseTypeSQLite, common.DatabaseTypeSQLite)
	t.Cleanup(func() {
		DB = previousDB
		LOG_DB = previousLogDB
		common.SetDatabaseTypes(common.DatabaseTypeSQLite, common.DatabaseTypeSQLite)
	})
	return db
}

func analyticsTestDigest(value string) string {
	digest := sha256.Sum256([]byte(value))
	return fmt.Sprintf("%x", digest[:])
}

func TestApplyRouteAnalyticsProjectionIsIdempotentAndCorrectable(t *testing.T) {
	db := setupRouteAnalyticsTestDB(t)
	eventTimeMS := int64(1_800_000)
	projection := analyticsTestProjection(eventTimeMS, "observation-1", "projection-1")

	applied, err := ApplyRouteAnalyticsProjectionsV1(context.Background(), []RouteAnalyticsProjectionV1{projection})
	require.NoError(t, err)
	assert.Equal(t, 1, applied)
	applied, err = ApplyRouteAnalyticsProjectionsV1(context.Background(), []RouteAnalyticsProjectionV1{projection})
	require.NoError(t, err)
	assert.Zero(t, applied)

	var routeBucket RouteAnalyticsBucketV1
	require.NoError(t, db.First(&routeBucket, "bucket_key = ?", projection.RouteBucket.BucketKey).Error)
	assert.Equal(t, int64(1), routeBucket.LogicalRequests)
	assert.Equal(t, int64(1), routeBucket.FinalSuccesses)
	assert.Equal(t, int64(1), routeBucket.TTFTCount)
	assert.Equal(t, int64(1_200), routeBucket.TTFTSumMS)
	assert.Equal(t, int64(250), routeBucket.PredictedInputMicros)
	assert.JSONEq(t, `{"price-v1":1}`, routeBucket.PriceRevisionCountsJSON)

	var keyBucket KeyUsageAnalyticsBucketV1
	require.NoError(t, db.First(&keyBucket, "bucket_key = ?", projection.KeyBucket.BucketKey).Error)
	assert.Equal(t, int64(1), keyBucket.LogicalRequests)
	assert.Equal(t, int64(1), keyBucket.Successes)
	assert.Equal(t, int64(200), keyBucket.PositiveQuota)

	corrected := projection
	corrected.ProjectionHash = analyticsTestDigest("projection-2")
	corrected.IngestedAtMS++
	corrected.RouteFingerprintDigest = ""
	corrected.UserDigest = ""
	corrected.TokenID = 0
	corrected.LegacyIdentity = false
	corrected.RouteBucket = nil
	corrected.KeyBucket = nil
	corrected.RouteMeasures = RouteAnalyticsMeasuresV1{}
	corrected.KeyMeasures = KeyUsageAnalyticsMeasuresV1{}
	applied, err = ApplyRouteAnalyticsProjectionsV1(context.Background(), []RouteAnalyticsProjectionV1{corrected})
	require.NoError(t, err)
	assert.Equal(t, 1, applied)

	require.NoError(t, db.First(&routeBucket, "bucket_key = ?", projection.RouteBucket.BucketKey).Error)
	assert.Zero(t, routeBucket.LogicalRequests)
	assert.Zero(t, routeBucket.FinalSuccesses)
	assert.Zero(t, routeBucket.TTFTCount)
	assert.Zero(t, routeBucket.TTFTSumMS)
	assert.Zero(t, routeBucket.PredictedInputMicros)
	assert.JSONEq(t, `{}`, routeBucket.PriceRevisionCountsJSON)

	require.NoError(t, db.First(&keyBucket, "bucket_key = ?", projection.KeyBucket.BucketKey).Error)
	assert.Zero(t, keyBucket.LogicalRequests)
	assert.Zero(t, keyBucket.Successes)
	assert.Zero(t, keyBucket.PositiveQuota)

	var observation RouteAnalyticsObservationV1
	require.NoError(t, db.First(&observation, "observation_digest = ?", projection.ObservationDigest).Error)
	assert.Equal(t, corrected.ProjectionHash, observation.ProjectionHash)
	assert.Empty(t, observation.RouteFingerprintDigest)
	assert.Empty(t, observation.UserDigest)
	assert.Zero(t, observation.TokenID)
	assert.False(t, observation.LegacyIdentity)
	assert.Empty(t, observation.RouteBucketKey)
	assert.Empty(t, observation.KeyBucketKey)
	assert.JSONEq(t, `{}`, observation.RouteMeasuresJSON)
	assert.JSONEq(t, `{}`, observation.KeyMeasuresJSON)
}

func TestRouteAnalyticsConcurrentReplayCountsOnce(t *testing.T) {
	db := setupRouteAnalyticsTestDB(t)
	sqlDB, err := db.DB()
	require.NoError(t, err)
	sqlDB.SetMaxOpenConns(1)
	projection := analyticsTestProjection(7_200_000, "concurrent", "concurrent-v1")

	results := make(chan int, 2)
	errorsByWorker := make(chan error, 2)
	var workers sync.WaitGroup
	for range 2 {
		workers.Add(1)
		go func() {
			defer workers.Done()
			applied, applyErr := ApplyRouteAnalyticsProjectionsV1(context.Background(), []RouteAnalyticsProjectionV1{projection})
			results <- applied
			errorsByWorker <- applyErr
		}()
	}
	workers.Wait()
	close(results)
	close(errorsByWorker)

	totalApplied := 0
	for applyErr := range errorsByWorker {
		require.NoError(t, applyErr)
	}
	for applied := range results {
		totalApplied += applied
	}
	assert.Equal(t, 1, totalApplied)

	var routeBucket RouteAnalyticsBucketV1
	require.NoError(t, db.First(&routeBucket, "bucket_key = ?", projection.RouteBucket.BucketKey).Error)
	assert.Equal(t, int64(1), routeBucket.LogicalRequests)
	assert.Equal(t, int64(1), routeBucket.FinalSuccesses)
}

func TestRouteAnalyticsWorkerLeaseIsExclusiveAndFenced(t *testing.T) {
	setupRouteAnalyticsTestDB(t)
	ctx := context.Background()
	state, acquired, err := AcquireRouteAnalyticsWorkerLeaseV1(ctx, "worker-a", 1_000, 2_000)
	require.NoError(t, err)
	assert.True(t, acquired)
	assert.Equal(t, "worker-a", state.LeaseOwner)

	_, acquired, err = AcquireRouteAnalyticsWorkerLeaseV1(ctx, "worker-b", 1_500, 2_500)
	require.NoError(t, err)
	assert.False(t, acquired)

	state, acquired, err = AcquireRouteAnalyticsWorkerLeaseV1(ctx, "worker-b", 2_001, 3_000)
	require.NoError(t, err)
	assert.True(t, acquired)
	assert.Equal(t, "worker-b", state.LeaseOwner)

	err = UpdateRouteAnalyticsWorkerStateV1(ctx, "worker-a", 2_100, 3_100, map[string]any{"freshness_at_ms": int64(2_100)})
	assert.ErrorIs(t, err, ErrRouteAnalyticsWorkerLeaseLost)
	require.NoError(t, UpdateRouteAnalyticsWorkerStateV1(
		ctx,
		"worker-b",
		2_100,
		3_100,
		map[string]any{"freshness_at_ms": int64(2_100), "cursor_created_at": int64(123)},
	))
	state, err = GetRouteAnalyticsWorkerStateV1(ctx)
	require.NoError(t, err)
	assert.Equal(t, int64(2_100), state.FreshnessAtMS)
	assert.Equal(t, int64(123), state.CursorCreatedAt)

	require.NoError(t, ReleaseRouteAnalyticsWorkerLeaseV1(ctx, "worker-b", 2_200))
	state, err = GetRouteAnalyticsWorkerStateV1(ctx)
	require.NoError(t, err)
	assert.Empty(t, state.LeaseOwner)
	assert.Zero(t, state.LeaseUntilMS)
}

func TestRouteAnalyticsTTFTPercentilesMergeHistogramsAndPreserveUnknown(t *testing.T) {
	p50, p95, total, capped, err := RouteAnalyticsTTFTPercentilesV1(nil)
	require.NoError(t, err)
	assert.Nil(t, p50)
	assert.Nil(t, p95)
	assert.Zero(t, total)
	assert.False(t, capped)

	first := make([]int64, len(ttftHistogramUpperBoundsMS))
	first[0] = 1
	first[4] = 8
	first[len(first)-1] = 1
	second := make([]int64, len(ttftHistogramUpperBoundsMS))
	second[4] = 10
	firstJSON, err := common.Marshal(first)
	require.NoError(t, err)
	secondJSON, err := common.Marshal(second)
	require.NoError(t, err)

	p50, p95, total, capped, err = RouteAnalyticsTTFTPercentilesV1([]string{string(firstJSON), string(secondJSON)})
	require.NoError(t, err)
	require.NotNil(t, p50)
	require.NotNil(t, p95)
	assert.Equal(t, int64(2_000), *p50)
	assert.Equal(t, int64(2_000), *p95)
	assert.Equal(t, int64(20), total)
	assert.False(t, capped)

	lastOnly := make([]int64, len(ttftHistogramUpperBoundsMS))
	lastOnly[len(lastOnly)-1] = 1
	lastJSON, err := common.Marshal(lastOnly)
	require.NoError(t, err)
	_, p95, total, capped, err = RouteAnalyticsTTFTPercentilesV1([]string{string(lastJSON)})
	require.NoError(t, err)
	require.NotNil(t, p95)
	assert.Equal(t, int64(300_000), *p95)
	assert.Equal(t, int64(1), total)
	assert.True(t, capped)
}
