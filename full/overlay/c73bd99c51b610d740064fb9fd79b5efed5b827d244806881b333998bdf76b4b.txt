import {
  BarChart3,
  CircleDollarSign,
  LockKeyhole,
  WalletCards,
} from 'lucide-react'
import { useTranslation } from 'react-i18next'

import { Skeleton } from '@/components/ui/skeleton'
import { formatQuota } from '@/lib/format'

import type { UserWalletData } from '../types'

interface WalletStatsCardProps {
  user: UserWalletData | null
  loading?: boolean
}

export function WalletStatsCard(props: WalletStatsCardProps) {
  const { t } = useTranslation()
  if (props.loading) {
    return (
      <div className='overflow-hidden rounded-lg border'>
        <div className='divide-border/60 grid grid-cols-2 divide-x sm:grid-cols-4'>
          {Array.from({ length: 4 }).map((_, i) => (
            <div key={i} className='px-3 py-3 sm:px-5 sm:py-4'>
              <Skeleton className='h-3.5 w-20' />
              <Skeleton className='mt-2 h-7 w-28' />
              <Skeleton className='mt-1.5 h-3.5 w-24' />
            </div>
          ))}
        </div>
      </div>
    )
  }

  const availableQuota = props.user?.quota ?? 0
  const activeReservedQuota = props.user?.active_reserved_quota ?? 0
  const stableAccountFunds = availableQuota + activeReservedQuota
  const stats = [
    {
      label: t('Current Balance'),
      value: formatQuota(stableAccountFunds),
      description: `${t('Available balance')} + ${t('Active reservation')}`,
      icon: WalletCards,
    },
    {
      label: t('Available balance'),
      value: formatQuota(availableQuota),
      description: t('Remaining quota'),
      icon: CircleDollarSign,
    },
    {
      label: t('Active reservation'),
      value: formatQuota(activeReservedQuota),
      description: t('Reserved quota units'),
      icon: LockKeyhole,
    },
    {
      label: t('Settled consumption'),
      value: formatQuota(props.user?.used_quota ?? 0),
      description: t('Total consumed quota'),
      icon: BarChart3,
    },
  ]

  return (
    <div className='overflow-hidden rounded-lg border'>
      <div className='divide-border/60 grid grid-cols-2 divide-x sm:grid-cols-4'>
        {stats.map((item) => (
          <div key={item.label} className='px-3 py-3 sm:px-5 sm:py-4'>
            <div className='flex items-center gap-2'>
              <item.icon className='text-muted-foreground/60 size-3.5 shrink-0' />
              <div className='text-muted-foreground truncate text-xs font-medium tracking-wider uppercase'>
                {item.label}
              </div>
            </div>

            <div className='text-foreground mt-1.5 font-mono text-base font-bold tracking-tight break-all tabular-nums sm:mt-2 sm:text-2xl'>
              {item.value}
            </div>
            <div className='text-muted-foreground/60 mt-1 hidden text-xs md:block'>
              {item.description}
            </div>
          </div>
        ))}
      </div>
    </div>
  )
}
