// decodeTenantCostAdminJSON rejects unknown fields and trailing JSON values.
// This is important for the provisioning endpoints: silently ignoring a
// client-supplied key/tenant field would make the HTTP contract appear safer
// than it is.
func decodeTenantCostAdminJSON(c *gin.Context, destination any) error {
	if c == nil || c.Request == nil || c.Request.Body == nil {
		return errors.New("request body is missing")
	}
	decoder := json.NewDecoder(c.Request.Body)
	decoder.DisallowUnknownFields()
	if err := decoder.Decode(destination); err != nil {
		return err
	}
	var trailing interface{}
	if err := decoder.Decode(&trailing); err != io.EOF {
		if err == nil {
			return errors.New("request contains multiple JSON values")
		}
		return err
	}
	return nil
}

// AdminProvisionTenantCostSlot performs the HTTP-side portion of a Root-only
// slot/profile provisioning operation. All durable writes, binding identity
// checks, and cache invalidation remain in the model transaction.
func AdminProvisionTenantCostSlot(c *gin.Context) {
	setTenantCostNoStore(c)
	tenantID, ok := resolveTenantCostTenant(c)
	if !ok {
		return
	}
	var request tenantCostSlotProvisionRequest
	if err := decodeTenantCostAdminJSON(c, &request); err != nil {
		c.JSON(http.StatusBadRequest, gin.H{"success": false, "message": "请求格式无效"})
		return
	}
	request.RequestID = strings.TrimSpace(request.RequestID)
	if request.RequestID == "" {
		c.JSON(http.StatusBadRequest, gin.H{"success": false, "message": "必须提供 request_id"})
		return
	}
	tokenID := 0
	if request.TokenID != nil {
		tokenID = *request.TokenID
		if tokenID <= 0 {
			c.JSON(http.StatusBadRequest, gin.H{"success": false, "message": "token_id 无效"})
			return
		}
	}
	result, err := model.ProvisionTenantCostSlot(model.TenantCostSlotProvisionSpec{
		RequestID: request.RequestID, TenantID: tenantID, ProductPoolID: request.ProductPoolID, ProductFamily: request.ProductFamily,
		ServiceTier: request.ServiceTier, PublicLabel: request.PublicLabel, SortOrder: request.SortOrder,
		ProfileID: request.ProfileID, ProfileRevision: request.ProfileRevision, ModelSelector: request.ModelSelector,
		MinRatioPPM: request.MinRatioPPM, MaxRatioPPM: request.MaxRatioPPM, ProviderClass: request.ProviderClass,
		CandidateSpec: request.CandidateSpec, CapabilitySpec: request.CapabilitySpec,
		ProfileStatus: request.ProfileStatus, DisabledReason: request.DisabledReason, TokenID: tokenID,
		ModelCatalogRevision: request.ModelCatalogRevision, BindingStatus: request.BindingStatus,
	}, c.GetInt("id"))
	if err != nil {
		tenantCostAdminError(c, err, "配置租户服务槽位失败")
		return
	}
	markAuditLogged(c)
	data := gin.H{
		"slot":            tenantCostProvisionSlotViewOf(result.Slot),
		"profile":         tenantCostProvisionProfileViewOf(result.Profile),
		"binding":         tenantCostProvisionBindingViewOf(result.Binding),
		"binding_created": result.BindingCreated,
		"replayed":        result.Replayed,
	}
	// result.Secret is populated only for a newly-created binding by the model;
	// an idempotent/repeated provisioning operation therefore cannot recover or
	// re-disclose an old key through this endpoint.
	if result.BindingCreated && result.Secret != "" {
		data["secret"] = result.Secret
	}
	c.JSON(http.StatusOK, gin.H{"success": true, "data": data})
}

// AdminRotateTenantCostBindingKey exposes the one-time secret returned by a
// first successful rotation. Model-level receipt replay sets AlreadyRotated;
// in that case this handler deliberately emits an empty secret.
func AdminRotateTenantCostBindingKey(c *gin.Context) {
	setTenantCostNoStore(c)
	tenantID, ok := resolveTenantCostTenant(c)
	if !ok {
		return
	}
	bindingID, err := strconv.ParseInt(strings.TrimSpace(c.Param("binding_id")), 10, 64)
	if err != nil || bindingID <= 0 {
		c.JSON(http.StatusBadRequest, gin.H{"success": false, "message": "绑定 ID 无效"})
		return
	}
	var request tenantCostKeyRotationRequest
	if err := decodeTenantCostAdminJSON(c, &request); err != nil {
		c.JSON(http.StatusBadRequest, gin.H{"success": false, "message": "请求格式无效"})
		return
	}
	request.RequestID = strings.TrimSpace(request.RequestID)
	if request.RequestID == "" {
		c.JSON(http.StatusBadRequest, gin.H{"success": false, "message": "必须提供 request_id"})
		return
	}
	result, err := model.RotateTenantCostBindingKeyForTenant(tenantID, bindingID, request.RequestID, c.GetInt("id"))
	if err != nil {
		tenantCostAdminError(c, err, "轮换租户服务 Key 失败")
		return
	}
	markAuditLogged(c)
	data := gin.H{
		"binding":         tenantCostProvisionBindingViewOf(&result.Binding),
		"already_rotated": result.AlreadyRotated,
	}
	if !result.AlreadyRotated {
		data["secret"] = result.Secret
	}
	if result.CacheWarning != "" {
		data["cache_warning"] = result.CacheWarning
	}
	c.JSON(http.StatusOK, gin.H{"success": true, "data": data})
}

func tenantCostAdminError(c *gin.Context, err error, fallback string) {
	if err == nil {
		return
	}
	status := http.StatusBadRequest
	if errors.Is(err, gorm.ErrRecordNotFound) {
		status = http.StatusNotFound
	} else if errors.Is(err, model.ErrTenantCostConflict) {
		status = http.StatusConflict
	} else if errors.Is(err, model.ErrTenantCostScope) {
		status = http.StatusForbidden
	} else if errors.Is(err, model.ErrTenantCostBindingUnavailable) {
		status = http.StatusServiceUnavailable
	} else if strings.Contains(strings.ToLower(err.Error()), "database") || strings.Contains(strings.ToLower(err.Error()), "sql") {
		status = http.StatusInternalServerError
	}
	c.JSON(status, gin.H{"success": false, "message": fallback})
}

func AdminUpsertTenantCostDownstream(c *gin.Context) {
	var request tenantCostDownstreamRequest
	if err := c.ShouldBindJSON(&request); err != nil {
		c.JSON(http.StatusBadRequest, gin.H{"success": false, "message": "请求格式无效"})
		return
	}
	request.TenantID = strings.TrimSpace(request.TenantID)
	request.DisplayName = strings.TrimSpace(request.DisplayName)
	request.ConfigKey = strings.TrimSpace(request.ConfigKey)
	if request.Status != nil {
		status := strings.TrimSpace(*request.Status)
		request.Status = &status
	}
	if request.ConfigKey == "" {
		request.ConfigKey = model.DefaultTenantCostConfigKey(request.TenantID)
	}
	status := model.TenantCostDownstreamActive
	if request.Status != nil {
		status = *request.Status
	}
	// The model transaction owns existing-row state preservation while holding
	// the row lock. Avoid a second pre-read whose failure could otherwise be
	// ignored and split status semantics between controller and model.
	row := &model.TenantCostDownstream{TenantID: request.TenantID, DisplayName: request.DisplayName,
		ConfigKey: request.ConfigKey, Status: status}
	if err := model.UpsertTenantCostDownstreamWithAudit(row, c.GetInt("id")); err != nil {
		tenantCostAdminError(c, err, "保存下游配置失败")
		return
	}
	markAuditLogged(c)
	c.JSON(http.StatusOK, gin.H{"success": true, "data": row})
}

func AdminCreateTenantReportingToken(c *gin.Context) {
	setTenantCostNoStore(c)
	var request tenantReportingTokenRequest
	if c.Request.ContentLength > 0 {
		if err := c.ShouldBindJSON(&request); err != nil {
			c.JSON(http.StatusBadRequest, gin.H{"success": false, "message": "请求格式无效"})
			return
		}
	}
	request.RequestID = strings.TrimSpace(request.RequestID)
	if request.RequestID == "" {
		c.JSON(http.StatusBadRequest, gin.H{"success": false, "message": "必须提供 request_id"})
		return
	}
	tenantID, ok := resolveTenantCostTenant(c)
	if !ok {
		return
	}
	mutation, err := model.CreateTenantReportingTokenWithAuditRequest(tenantID,
		model.TenantCostReportingPermission, request.ExpiresAt, request.RequestID, c.GetInt("id"))
	if err != nil {
		tenantCostAdminError(c, err, "创建 reporting token 失败")
		return
	}
	markAuditLogged(c)
	row := mutation.Token
	views := model.TenantReportingTokenView{ID: row.ID, TenantID: row.TenantID, TokenID: row.TokenID,
		Permission: row.Permission, ExpiresAt: row.ExpiresAt, RevokedAt: row.RevokedAt,
		LastUsedAt: row.LastUsedAt, CreatedAt: row.CreatedAt}
	// The secret is intentionally present only in this creation response. It is
	// never stored in the view, audit details, or subsequent list responses.
	data := gin.H{"token": views, "replayed": mutation.Replayed}
	if mutation.Secret != "" {
		data["secret"] = mutation.Secret
	}
	c.JSON(http.StatusCreated, gin.H{"success": true, "data": data})
}
