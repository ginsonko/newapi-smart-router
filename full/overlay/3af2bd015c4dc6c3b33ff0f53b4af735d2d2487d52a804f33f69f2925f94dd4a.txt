func setupImageRouteJobTest(t *testing.T) *gorm.DB {
	t.Helper()
	previousDB := DB
	dsn := fmt.Sprintf("file:image-route-job-%s?mode=memory&cache=shared", strings.ReplaceAll(t.Name(), "/", "_"))
	database, err := gorm.Open(sqlite.Open(dsn), &gorm.Config{})
	require.NoError(t, err)
	connection, err := database.DB()
	require.NoError(t, err)
	connection.SetMaxOpenConns(1)
	DB = database
	t.Cleanup(func() {
		DB = previousDB
		require.NoError(t, connection.Close())
	})
	require.NoError(t, database.AutoMigrate(&Task{}, &TaskBillingReceipt{}, &ImageRouteJob{}))
	return database
}

func TestImageRouteJobChannelIntentRejectsReuseAndCompetingCAS(t *testing.T) {
	database := setupImageRouteJobTest(t)
	job, _ := createImageRouteJobTest(t)
	require.ErrorIs(t, RecordImageRouteJobChannel(job.ID, 73), ErrImageRouteJobConflict)
	_, won, err := ClaimImageRouteJob(job.ID, job.Version)
	require.NoError(t, err)
	require.True(t, won)
	require.NoError(t, RecordImageRouteJobChannel(job.ID, 73))
	first := reloadImageRouteJobTest(t, job)
	require.ErrorIs(t, RecordImageRouteJobChannel(job.ID, 73), ErrImageRouteJobConflict)
	assert.Equal(t, first.Version, reloadImageRouteJobTest(t, job).Version)
	require.NoError(t, RecordImageRouteJobChannel(job.ID, 74))
	assert.Equal(t, "[73,74]", reloadImageRouteJobTest(t, job).AttemptedChannelsJSON)
	injected := false
	require.NoError(t, database.Callback().Query().After("gorm:query").Register("image_route_competing_channel", func(tx *gorm.DB) {
		if injected || tx.Statement.Schema == nil || tx.Statement.Schema.Table != "image_route_jobs" {
			return
		}
		injected = true
		err := tx.Session(&gorm.Session{NewDB: true}).Model(&ImageRouteJob{}).Where("id = ?", job.ID).
			Updates(map[string]any{"attempted_channels_json": "[73,74,75]", "version": gorm.Expr("version + 1")}).Error
		tx.AddError(err)
	}))
	require.ErrorIs(t, RecordImageRouteJobChannel(job.ID, 76), ErrImageRouteJobConflict)
	require.NoError(t, database.Callback().Query().Remove("image_route_competing_channel"))
	assert.True(t, injected)
	assert.Equal(t, "[73,74]", reloadImageRouteJobTest(t, job).AttemptedChannelsJSON)
}

		{"wrong owner", func(child *Task, receipt *TaskBillingReceipt) { child.UserId++ }},
		{"wrong token", func(child *Task, receipt *TaskBillingReceipt) { child.PrivateData.TokenId++ }},

		{"receipt applied charged", func(child *Task, receipt *TaskBillingReceipt) { receipt.AppliedQuota = 1 }},

func TestImageRouteJobCompletionTerminalCASAndSafeProjection(t *testing.T) {
	for _, terminal := range []struct {
		state  string
		status TaskStatus
	}{
		{ImageRouteJobSucceeded, TaskStatusSuccess},
		{ImageRouteJobFailed, TaskStatusFailure},
		{ImageRouteJobUnknown, TaskStatusUnknown},
	} {
		t.Run(terminal.state, func(t *testing.T) {
			setupImageRouteJobTest(t)
			job, root, _, _ := waitingImageRouteJobTest(t)
			payload := []byte(`{"created":123,"data":[{"url":"https://images.example/output.png","revised_prompt":"private prompt"}],"metadata":{"key":"secret"}}`)
			completed, err := CompleteImageRouteJob(job.ID, job.Version-1, terminal.state, payload, "safe failure")
			require.NoError(t, err)
			assert.False(t, completed)
			completed, err = CompleteImageRouteJob(job.ID, job.Version, terminal.state, payload, "safe failure")
			require.NoError(t, err)
			require.True(t, completed)
			finished := reloadImageRouteJobTest(t, job)
			assert.Equal(t, job.Version+1, finished.Version)
			assert.Equal(t, terminal.state, finished.State)
			var task Task
			require.NoError(t, DB.First(&task, root.ID).Error)
			assert.Equal(t, terminal.status, task.Status)
			assert.Zero(t, task.Quota)
			assert.Equal(t, "b64_json", task.PrivateData.ImageTask.ResponseFormat)
			assert.EqualValues(t, 253402300799, task.NextPollAt)
			if terminal.state == ImageRouteJobSucceeded {
				assert.JSONEq(t, `{"created":123,"data":[{"url":"https://images.example/output.png"}]}`, string(task.Data))
				assert.Empty(t, task.FailReason)
			} else {
				assert.JSONEq(t, `{}`, string(task.Data))
				assert.Equal(t, "safe failure", task.FailReason)
			}
			for _, version := range []int64{job.Version, finished.Version} {
				for _, competing := range []string{ImageRouteJobFailed, ImageRouteJobSucceeded} {
					completed, err := CompleteImageRouteJob(job.ID, version, competing, payload, "competing result")
					require.NoError(t, err)
					assert.False(t, completed)
				}
			}
			resumed, err := ResumeImageRouteJob(job.ID, finished.Version)
			require.NoError(t, err)
			assert.False(t, resumed)
		})
	}
}

func TestImageRouteJobCompletionRollsBackWhenRootIsUnavailable(t *testing.T) {
	for _, rootState := range []string{"missing", "already succeeded", "duplicate"} {
		t.Run(rootState, func(t *testing.T) {
			setupImageRouteJobTest(t)
			job, root := createImageRouteJobTest(t)
			switch rootState {
			case "missing":
				require.NoError(t, DB.Delete(root).Error)
			case "already succeeded":
				require.NoError(t, DB.Model(root).Update("status", TaskStatusSuccess).Error)
			case "duplicate":
				duplicate := *root
				duplicate.ID = 0
				require.NoError(t, DB.Create(&duplicate).Error)
			}
			completed, err := CompleteImageRouteJob(job.ID, job.Version, ImageRouteJobFailed, nil, "safe failure")
			require.ErrorIs(t, err, ErrImageRouteJobConflict)
			assert.False(t, completed)
			assert.Equal(t, job.Version, reloadImageRouteJobTest(t, job).Version)
			assert.Equal(t, ImageRouteJobReady, reloadImageRouteJobTest(t, job).State)
			if rootState != "missing" {
				var persisted Task
				require.NoError(t, DB.First(&persisted, root.ID).Error)
				expectedStatus := TaskStatusSubmitted
				if rootState == "already succeeded" {
					expectedStatus = TaskStatusSuccess
				}
				assert.EqualValues(t, expectedStatus, persisted.Status)
				assert.JSONEq(t, `{}`, string(persisted.Data))
				assert.Zero(t, persisted.FinishTime)
			}
		})
	}
}

func TestImageRouteJobCompletionRootWriteFailureRollsBackBothRows(t *testing.T) {
	database := setupImageRouteJobTest(t)
	job, root := createImageRouteJobTest(t)
	sentinel := errors.New("root completion storage failed")
	require.NoError(t, database.Callback().Update().After("gorm:update").Register("image_route_root_completion_failure", func(tx *gorm.DB) {
		if tx.Statement.Schema != nil && tx.Statement.Schema.Table == "tasks" {
			tx.AddError(sentinel)
		}
	}))
	completed, err := CompleteImageRouteJob(job.ID, job.Version, ImageRouteJobFailed, nil, "safe failure")
	require.ErrorIs(t, err, sentinel)
	assert.False(t, completed)
	loaded := reloadImageRouteJobTest(t, job)
	assert.Equal(t, job.State, loaded.State)
	assert.Equal(t, job.Version, loaded.Version)
	var persisted Task
	require.NoError(t, database.First(&persisted, root.ID).Error)
	assert.EqualValues(t, TaskStatusSubmitted, persisted.Status)
	assert.Equal(t, "5%", persisted.Progress)
	assert.Zero(t, persisted.FinishTime)
	assert.Empty(t, persisted.FailReason)
}
