func validTenantCostPublicIdentifier(value string, maxLength int) bool {
	value = strings.TrimSpace(value)
	lowerValue := strings.ToLower(value)
	if value == "" || len(value) > maxLength || containsTenantCostSecretMarker(lowerValue) ||
		strings.Contains(lowerValue, "://") {
		return false
	}
	for _, r := range value {
		if r >= 'a' && r <= 'z' || r >= 'A' && r <= 'Z' || r >= '0' && r <= '9' || r == '_' || r == '-' || r == '.' || r == ':' {
			continue
		}
		return false
	}
	return true
}

// containsTenantCostSecretMarker rejects API-key-shaped identifiers while
// allowing ordinary IDs such as "task-123", where "sk-" only occurs across
// letters inside a larger word.
func containsTenantCostSecretMarker(lowerValue string) bool {
	for offset := 0; offset < len(lowerValue); {
		index := strings.Index(lowerValue[offset:], "sk-")
		if index < 0 {
			return false
		}
		index += offset
		if index == 0 || !((lowerValue[index-1] >= 'a' && lowerValue[index-1] <= 'z') ||
			(lowerValue[index-1] >= '0' && lowerValue[index-1] <= '9')) {
			return true
		}
		offset = index + len("sk-")
	}
	return false
}

func validTenantCostKeySuffix(value string) bool {
	value = strings.TrimSpace(value)
	if len(value) < 4 || len(value) > 8 {
		return false
	}
	for _, r := range value {
		if (r >= 'a' && r <= 'z') || (r >= 'A' && r <= 'Z') || (r >= '0' && r <= '9') {
			continue
		}
		return false
	}
	return true
}

func validTenantBindingStatus(value string) bool {
	switch strings.TrimSpace(value) {
	case TenantBindingPending, TenantBindingEnabled, TenantBindingDisabled, TenantBindingRevoked:
		return true
	default:
		return false
	}
}

func (s TenantCostEventSpec) normalize() (TenantCostEventSpec, error) {
	if err := validateTenantParameterCosts(s); err != nil {return s, err}
	s.TenantID = strings.TrimSpace(s.TenantID)
	s.EventID = strings.TrimSpace(s.EventID)
	s.EventType = strings.TrimSpace(s.EventType)
	s.ProductPoolID = strings.TrimSpace(s.ProductPoolID)
	s.ServiceTier = strings.TrimSpace(s.ServiceTier)
	s.RouteProfileID = strings.TrimSpace(s.RouteProfileID)
	s.ProfileRevision = strings.TrimSpace(s.ProfileRevision)
	s.KeyID = strings.TrimSpace(s.KeyID)
	s.RequestedModel = strings.TrimSpace(s.RequestedModel)
	s.CanonicalModel = strings.TrimSpace(s.CanonicalModel)
	s.BillingMode = strings.TrimSpace(s.BillingMode)
	s.BillingUnit = strings.TrimSpace(s.BillingUnit)
	s.Evidence = strings.TrimSpace(s.Evidence)
	s.Status = strings.TrimSpace(s.Status)
	if s.TenantID == "" {
		// A missing tenant is an identity failure, not a request for the first
		// registered downstream.  Falling back here would silently attribute a
		// malformed or forged projection to Downstream.  Compatibility callers
		// must choose the default explicitly before entering this normalizer.
		return s, ErrTenantCostScope
	}
	if s.EventID == "" {
		s.EventID = "uce_" + uuid.New().String()
	}
	if s.OccurredAt.IsZero() {
		s.OccurredAt = time.Now().UTC()
	}
	if s.FinalizedAt.IsZero() {
		s.FinalizedAt = s.OccurredAt
	}
	if s.Status == "" {
		s.Status = TenantCostStatusFinal
	}
	if s.Currency == "" {
		s.Currency = "USD"
	}
	if s.ProcurementCurrency == "" {
		s.ProcurementCurrency = s.Currency
	}
	if !IsValidTenantCostTenantID(s.TenantID) || s.EventType == "" || s.ProductPoolID == "" ||
		s.ServiceTier == "" || s.RouteProfileID == "" || s.ProfileRevision == "" ||
		!validTenantCostPublicIdentifier(s.KeyID, 128) || s.CanonicalModel == "" || s.BillingMode == "" || s.BillingUnit == "" ||
		s.GatewayChargeAmount == "" || s.Evidence == "" {
		return s, ErrTenantCostInvalidInput
	}
	if !validTenantCostEventType(s.EventType) || !validTenantCostEvidence(s.Evidence) {
		return s, ErrTenantCostInvalidInput
	}
	if s.EventType == TenantCostEventSettlement || s.EventType == TenantCostEventChargedFailure ||
		s.EventType == TenantCostEventObservedZero {
		if s.GatewayChargeQuota < 0 {
			return s, ErrTenantCostInvalidInput
		}
	}
	if s.EventType == TenantCostEventObservedZero && s.GatewayChargeQuota != 0 {
		return s, ErrTenantCostInvalidInput
	}
	if s.EventType == TenantCostEventRefund || s.EventType == TenantCostEventReversal || s.EventType == TenantCostEventCorrection {
		if s.ParentEventID == nil || strings.TrimSpace(*s.ParentEventID) == "" {
			return s, ErrTenantCostInvalidInput
		}
	}
	if s.Evidence == TenantCostEvidenceUnknown {
		if s.ProcurementAmount != nil || strings.TrimSpace(s.UnknownReason) == "" {
			return s, ErrTenantCostInvalidInput
		}
	} else if s.ProcurementAmount == nil {
		return s, ErrTenantCostInvalidInput
	}
	return s, nil
}

func validTenantCostEventType(value string) bool {
	switch value {
	case TenantCostEventSettlement, TenantCostEventChargedFailure, TenantCostEventRefund,
		TenantCostEventReversal, TenantCostEventSupplement, TenantCostEventCorrection,
		TenantCostEventObservedZero:
		return true
	default:
		return false
	}
}

func validTenantCostEvidence(value string) bool {
	switch value {

	default:
		return false
	}
}

func tenantCostPayloadHash(payload TenantCostEventPayload) (string, []byte, error) {
	// Hash only semantic fields; sequence/publish time are assigned by the
	// append transaction, so retries with the same source fact remain idempotent.
	payload.PayloadHash = ""
	b, err := common.Marshal(payload)
	if err != nil {
		return "", nil, err
	}
	sum := sha256.Sum256(b)
	hash := "sha256:" + hex.EncodeToString(sum[:])
	payload.PayloadHash = hash
	finalBytes, err := common.Marshal(payload)
	return hash, finalBytes, err
}

// tenantCostEventIdentityHash is used only to decide whether a retry with an
// already-used event ID is the same business fact.  Sequence and publication
// time are assigned by the append transaction and are intentionally absent;
// occurred/finalized timestamps are also excluded because the existing relay
// and task paths may reconstruct the same terminal fact after a process
// restart.  All financial, correlation, binding, model, evidence and parent
// fields remain part of the identity, so a changed charge or route cannot be
// silently accepted as a replay.
func tenantCostEventIdentityHash(payload TenantCostEventPayload) (string, error) {
	identity := payload
	identity.Sequence = 0
	identity.OccurredAt = ""
	identity.FinalizedAt = ""
	identity.PublishedAt = ""
	identity.PayloadHash = ""
	b, err := common.Marshal(identity)
	if err != nil {
		return "", err
	}
	sum := sha256.Sum256(b)
	return "sha256:" + hex.EncodeToString(sum[:]), nil
}

func tenantCostCorrelationQuery(tx *gorm.DB, spec TenantCostEventSpec) *gorm.DB {
	conditions := make([]string, 0, 4)
	args := make([]interface{}, 0, 4)
	for _, item := range []struct {
		column string
		value  string
	}{
		{"downstream_request_id", spec.DownstreamRequestId},
		{"gateway_request_id", spec.GatewayRequestId},
		{"reservation_id", spec.ReservationID},
		{"task_public_id", spec.TaskPublicID},
	} {
		if value := strings.TrimSpace(item.value); value != "" {
			conditions = append(conditions, item.column+" = ?")
			args = append(args, value)
		}
	}
	if len(conditions) == 0 {
		return tx.Where("1 = 0")
	}
	return tx.Where("("+strings.Join(conditions, " OR ")+")", args...)
}

// rejectTerminalCorrelationConflict prevents an asynchronous refund/zero
// projection from being followed by a second positive charge for the same
// durable request after a process restart. The billing authority remains the
// source of truth; this guard turns an impossible state into an auditable
// projection conflict instead of silently double-counting it.
func rejectTerminalCorrelationConflict(tx *gorm.DB, spec TenantCostEventSpec) error {
	var existing TenantCostEvent
	positiveTypes := []string{TenantCostEventSettlement, TenantCostEventChargedFailure, TenantCostEventSupplement}
	zeroTypes := []string{TenantCostEventObservedZero}
	var types []string
	if spec.EventType == TenantCostEventObservedZero {
		types = positiveTypes
	} else if spec.EventType == TenantCostEventSettlement || spec.EventType == TenantCostEventChargedFailure || spec.EventType == TenantCostEventSupplement {
		types = zeroTypes
	} else {
		return nil
	}
	query := tx.Where("tenant_id = ? AND status = ? AND event_type IN ?", spec.TenantID, TenantCostStatusFinal, types)
	query = tenantCostCorrelationQuery(query, spec)
	err := query.Order("sequence desc").First(&existing).Error
	if errors.Is(err, gorm.ErrRecordNotFound) {
		return nil
	}
	if err != nil {
		return err
	}
	return ErrTenantCostConflict
}

func nextTenantCostSequence(tx *gorm.DB, tenantID string, now int64) (int64, error) {
	var row TenantCostSequence
	err := lockForUpdate(tx).Where("tenant_id = ?", tenantID).First(&row).Error
	if errors.Is(err, gorm.ErrRecordNotFound) {
		row = TenantCostSequence{TenantID: tenantID, NextSequence: 2, UpdatedAt: now}
		if err := tx.Create(&row).Error; err != nil {
			// A concurrent initializer can win. Re-read under the transaction
			// lock; the unique tenant key prevents two cursors.
			if retryErr := lockForUpdate(tx).Where("tenant_id = ?", tenantID).First(&row).Error; retryErr != nil {
				return 0, retryErr
			}
			seq := row.NextSequence
			row.NextSequence++
			row.UpdatedAt = now
			return seq, tx.Save(&row).Error
		}
		return 1, nil
	}
	if err != nil {
		return 0, err
	}
	sequence := row.NextSequence
	row.NextSequence++
	row.UpdatedAt = now
	return sequence, tx.Save(&row).Error
}

// AppendTenantCostEvent appends exactly one immutable event and one outbox
// row. Replaying the same semantic event ID/hash is a successful no-op.
func AppendTenantCostEvent(spec TenantCostEventSpec) (TenantCostAppendResult, error) {
	if DB == nil {
		return TenantCostAppendResult{}, errors.New("database is not initialized")
	}
	var result TenantCostAppendResult
	err := DB.Transaction(func(tx *gorm.DB) error {
		var err error
		result, err = AppendTenantCostEventTx(tx, spec)
		return err
	})
	return result, err
}

// AppendTenantCostEventTx is the transaction-aware variant used by an
// existing authoritative settlement transaction when one is available. It
// never starts or commits a transaction itself.
func AppendTenantCostEventTx(tx *gorm.DB, spec TenantCostEventSpec) (TenantCostAppendResult, error) {
	var result TenantCostAppendResult
	normalized, err := spec.normalize()
	if err != nil {
		return result, err
	}
	semanticPayload := tenantCostSemanticPayload(normalized)
	semanticHash, _, err := tenantCostPayloadHash(semanticPayload)
	if err != nil {
		return result, err
	}
	if tx == nil {
		return result, errors.New("database is not initialized")
	}
	// A tenant namespace must be registered before an immutable fact can be
	// projected. Disabled tenants remain readable and can receive historical
	// reconciliation, but unknown IDs cannot be manufactured by an internal
	// caller or a forged request context.
	if _, err := requireTenantCostDownstreamTx(tx, normalized.TenantID, false); err != nil {
		return result, err
	}
	err = func() error {
		var existing TenantCostEvent
		lookupErr := tx.Where("tenant_id = ? AND event_id = ?", normalized.TenantID, normalized.EventID).First(&existing).Error
		if lookupErr == nil {
			var existingPayload TenantCostEventPayload
			if err := common.Unmarshal([]byte(existing.Payload), &existingPayload); err != nil {
				return err
			}
			incomingPayload := tenantCostSemanticPayload(normalized)
			existingIdentity, err := tenantCostEventIdentityHash(existingPayload)
			if err != nil {
				return err
			}
			incomingIdentity, err := tenantCostEventIdentityHash(incomingPayload)
			if err != nil {
				return err
			}
			if existingIdentity != incomingIdentity {
				return ErrTenantCostConflict
			}
			result.Event = existing
			result.Payload = existingPayload
			result.Replayed = true
			return nil
		}
		if !errors.Is(lookupErr, gorm.ErrRecordNotFound) {
			return lookupErr
		}
		if err := rejectTerminalCorrelationConflict(tx, normalized); err != nil {
			return err
		}
		now := time.Now().UTC()
		sequence, seqErr := nextTenantCostSequence(tx, normalized.TenantID, now.UnixNano())
		if seqErr != nil {
			return seqErr
		}
		payload := tenantCostSemanticPayload(normalized)
		payload.Sequence = sequence
		payload.PublishedAt = now.Format(time.RFC3339Nano)
		payloadHash, payloadBytes, hashErr := tenantCostPayloadHash(payload)
		if hashErr != nil {
			return hashErr
		}
		// The semantic hash is stable across retries. The final JSON includes
		// the assigned sequence and publication time for consumers.
		payload.PayloadHash = semanticHash
		payloadBytes, hashErr = common.Marshal(payload)
		if hashErr != nil {
			return hashErr
		}
		_ = payloadHash
		parent := normalized.ParentEventID
		row := TenantCostEvent{
			TenantID: normalized.TenantID, EventID: normalized.EventID, Sequence: sequence,
			EventType: normalized.EventType, ParentEventID: parent,
			OccurredAt: normalized.OccurredAt.UnixNano(), FinalizedAt: normalized.FinalizedAt.UnixNano(),
			PublishedAt: now.UnixNano(), DownstreamRequestId: normalized.DownstreamRequestId,
			GatewayRequestId: normalized.GatewayRequestId, ReservationID: normalized.ReservationID,
			RouteReceiptID: normalized.RouteReceiptID, UpstreamAttemptID: normalized.UpstreamAttemptID,
			TaskPublicID: normalized.TaskPublicID, ProductPoolID: normalized.ProductPoolID,
			ServiceTier: normalized.ServiceTier, RouteProfileID: normalized.RouteProfileID,
			ProfileRevision: normalized.ProfileRevision, KeyID: normalized.KeyID,
			RequestedModel: normalized.RequestedModel, CanonicalModel: normalized.CanonicalModel,
			UpstreamModel: normalized.UpstreamModel, MappingRevision: normalized.MappingRevision,
			BillingMode: normalized.BillingMode, BillingUnit: normalized.BillingUnit,
			GatewayChargeQuota: normalized.GatewayChargeQuota, GatewayChargeAmount: normalized.GatewayChargeAmount,
			Currency: normalized.Currency, ProcurementAmount: normalized.ProcurementAmount,
			ProcurementCurrency: normalized.ProcurementCurrency, Evidence: normalized.Evidence,
			BasisKind: normalized.BasisKind, BasisRevision: normalized.BasisRevision,
			Formula: normalized.Formula, UnknownReason: normalized.UnknownReason, Status: normalized.Status,
			PayloadHash: semanticHash, Payload: string(payloadBytes), CreatedAt: now.UnixNano(),
		}
		if err := tx.Create(&row).Error; err != nil {
			return err
		}
		outbox := TenantCostOutbox{TenantID: row.TenantID, EventID: row.EventID, State: TenantOutboxReady,
			NextAttemptAt: now.UnixNano(), CreatedAt: now.UnixNano(), UpdatedAt: now.UnixNano()}
		if err := tx.Create(&outbox).Error; err != nil {
			return err
		}
		result.Event = row
		result.Payload = payload
		return nil
	}()
	return result, err
}

func UpsertTenantCredentialBinding(binding *TenantCredentialBinding) error {
	if binding == nil || binding.TokenID <= 0 || strings.TrimSpace(binding.TenantID) == "" ||
		strings.TrimSpace(binding.ProductPoolID) == "" || strings.TrimSpace(binding.ServiceTier) == "" ||
		strings.TrimSpace(binding.RouteProfileID) == "" || strings.TrimSpace(binding.ProfileRevision) == "" ||
		strings.TrimSpace(binding.ModelCatalogRevision) == "" ||
		!validTenantCostPublicIdentifier(binding.KeyID, 128) || !validTenantCostKeySuffix(binding.KeySuffix) {
		return ErrTenantCostInvalidInput
	}
	if !IsValidTenantCostTenantID(binding.TenantID) || binding.CredentialPurpose != TenantCredentialPurposeDataPlane {
		return ErrTenantCostScope
	}
	if binding.BindingStatus == "" {
		binding.BindingStatus = TenantBindingPending
	}
	if !validTenantBindingStatus(binding.BindingStatus) {
		return ErrTenantCostInvalidInput
	}
	if DB == nil {
		return errors.New("database is not initialized")
	}
	now := common.GetTimestamp()
	if binding.CreatedAt == 0 {
		binding.CreatedAt = now
	}
	binding.UpdatedAt = now
	return DB.Transaction(func(tx *gorm.DB) error {
		if _, err := requireTenantCostDownstreamTx(tx, binding.TenantID, binding.BindingStatus == TenantBindingEnabled); err != nil {
			return err
		}
		var profile TenantRouteProfile
		if err := lockForUpdate(tx).Where("tenant_id = ? AND profile_id = ?", binding.TenantID, binding.RouteProfileID).First(&profile).Error; err != nil {
			if errors.Is(err, gorm.ErrRecordNotFound) {
				return ErrTenantCostBindingUnavailable
			}
			return err
		}
		if profile.ProductPoolID != binding.ProductPoolID || profile.ServiceTier != binding.ServiceTier ||
			profile.Revision != binding.ProfileRevision {
			return ErrTenantCostInvalidInput
		}
		if err := ValidateTenantRouteProfile(&profile); err != nil {
			return err
		}
		if binding.BindingStatus == TenantBindingEnabled && profile.Status != TenantBindingEnabled {
			return fmt.Errorf("route profile is not enabled")
		}
		var existing TenantCredentialBinding
		err := lockForUpdate(tx).Where("token_id = ?", binding.TokenID).First(&existing).Error
		if errors.Is(err, gorm.ErrRecordNotFound) {
			return tx.Create(binding).Error
		}
		if err != nil {
			return err
		}
		if existing.TenantID != binding.TenantID || existing.CredentialPurpose != binding.CredentialPurpose {
			return ErrTenantCostConflict
		}
		if existing.BindingStatus == TenantBindingRevoked && binding.BindingStatus != TenantBindingRevoked {
			return fmt.Errorf("revoked binding cannot be re-enabled")
		}
		if existing.BindingStatus == TenantBindingEnabled &&
			(existing.KeyID != binding.KeyID || existing.KeySuffix != binding.KeySuffix ||
				existing.ProductPoolID != binding.ProductPoolID || existing.ServiceTier != binding.ServiceTier ||
				existing.RouteProfileID != binding.RouteProfileID || existing.ProfileRevision != binding.ProfileRevision ||
				existing.ModelCatalogRevision != binding.ModelCatalogRevision) {
			return ErrTenantCostConflict
		}
		binding.ID = existing.ID
		return tx.Save(binding).Error
	})
}

// FindTenantCostChargeEvent locates the immutable positive-charge fact for a
// refund/correction parent. Correlation fields are all server-owned; callers
// cannot select another tenant. The query is intentionally bounded to charge
// event types and newest sequence so a late refund never points at an older
// unrelated request.
func FindTenantCostChargeEvent(tenantID, downstreamRequestID, gatewayRequestID, reservationID, taskPublicID string) (*TenantCostEvent, error) {
	if DB == nil {
		return nil, ErrTenantCostScope
	}
	tenantID = strings.TrimSpace(tenantID)
	if _, err := RequireRegisteredTenantCostDownstream(tenantID); err != nil {
		return nil, err
	}
	conditions := make([]string, 0, 4)
	args := make([]interface{}, 0, 4)
	for _, item := range []struct {
		column string
		value  string
	}{
		{"downstream_request_id", downstreamRequestID},
		{"gateway_request_id", gatewayRequestID},
		{"reservation_id", reservationID},
		{"task_public_id", taskPublicID},
	} {
		if value := strings.TrimSpace(item.value); value != "" {
			conditions = append(conditions, item.column+" = ?")
			args = append(args, value)
		}
	}
	if len(conditions) == 0 {
		return nil, ErrTenantCostInvalidInput
	}
	query := DB.Where("tenant_id = ?", tenantID).Where(
		"event_type IN ?",
		[]string{TenantCostEventSettlement, TenantCostEventChargedFailure, TenantCostEventSupplement},
	).Where("status = ?", TenantCostStatusFinal)
	query = query.Where("("+strings.Join(conditions, " OR ")+")", args...)
	var row TenantCostEvent
	if err := query.Order("sequence desc").First(&row).Error; err != nil {
		if errors.Is(err, gorm.ErrRecordNotFound) {
			return nil, nil
		}
		return nil, err
	}
	return &row, nil
}

// ClaimTenantCostOutbox atomically moves a bounded due batch to pending. A
// stale pending row is reclaimable after 10 minutes, which makes process
// crashes recoverable without allowing concurrent workers to double-claim.
func ClaimTenantCostOutbox(limit int, now int64) ([]TenantCostOutbox, error) {
	if DB == nil {
		return nil, errors.New("database is not initialized")
	}
	if limit <= 0 || limit > 500 {
		limit = 100
	}
	if now <= 0 {
		now = time.Now().UTC().UnixNano()
	}
	claimed := make([]TenantCostOutbox, 0, limit)
	err := DB.Transaction(func(tx *gorm.DB) error {
		var rows []TenantCostOutbox
		staleBefore := now - int64(10*time.Minute)
		query := lockForUpdate(tx).Where("((state IN ? AND next_attempt_at <= ?) OR (state = ? AND updated_at <= ?))",
			[]string{TenantOutboxReady, TenantOutboxFailed}, now, TenantOutboxPending, staleBefore).Order("id asc").Limit(limit).Find(&rows)
		if query.Error != nil {
			return query.Error
		}
		for i := range rows {
			row := rows[i]
			row.State = TenantOutboxPending
			row.Attempts++
			row.UpdatedAt = now
			if err := tx.Model(&TenantCostOutbox{}).Where("id = ? AND tenant_id = ?", row.ID, row.TenantID).Updates(map[string]interface{}{
				"state": row.State, "attempts": row.Attempts, "updated_at": row.UpdatedAt,
			}).Error; err != nil {
				return err
			}
			claimed = append(claimed, row)
		}
		return nil
	})
	return claimed, err
}

func MarkTenantCostOutboxSentForTenant(tenantID, eventID string, now int64) error {
	if DB == nil || !IsValidTenantCostTenantID(strings.TrimSpace(tenantID)) || strings.TrimSpace(eventID) == "" {
		return ErrTenantCostInvalidInput
	}
	if now <= 0 {
		now = time.Now().UTC().UnixNano()
	}
	return DB.Model(&TenantCostOutbox{}).Where("tenant_id = ? AND event_id = ? AND state = ?", strings.TrimSpace(tenantID), strings.TrimSpace(eventID), TenantOutboxPending).
		Updates(map[string]interface{}{"state": TenantOutboxSent, "published_at": now, "updated_at": now}).Error
}

func MarkTenantCostOutboxFailed(eventID string, cause error, now int64) error {
	return MarkTenantCostOutboxFailedForTenant(DefaultTenantCostTenantID, eventID, cause, now)
}

func MarkTenantCostOutboxFailedForTenant(tenantID, eventID string, cause error, now int64) error {
	if DB == nil || !IsValidTenantCostTenantID(strings.TrimSpace(tenantID)) || strings.TrimSpace(eventID) == "" {
		return ErrTenantCostInvalidInput
	}
	if now <= 0 {
		now = time.Now().UTC().UnixNano()
	}
	message := "unknown error"
	if cause != nil {
		message = cause.Error()
	}
	if len(message) > 2048 {
		message = message[:2048]
	}
	var row TenantCostOutbox
	if err := DB.Where("tenant_id = ? AND event_id = ?", strings.TrimSpace(tenantID), strings.TrimSpace(eventID)).First(&row).Error; err != nil {
		return err
	}
	backoff := time.Second * time.Duration(1<<minInt(row.Attempts, 10))
	if backoff > time.Hour {
		backoff = time.Hour
	}
	return DB.Model(&TenantCostOutbox{}).Where("id = ? AND tenant_id = ? AND state = ?", row.ID, strings.TrimSpace(tenantID), TenantOutboxPending).
		Updates(map[string]interface{}{"state": TenantOutboxFailed, "last_error": message, "next_attempt_at": now + backoff.Nanoseconds(), "updated_at": now}).Error
}

}

func minInt(a, b int) int {
	if a < b {
		return a
	}
	return b
}

func GetTenantCostReconciliation(tenantID string) (TenantCostReconciliation, error) {
	tenantID = strings.TrimSpace(tenantID)
	result := TenantCostReconciliation{TenantID: tenantID, GeneratedAt: time.Now().UTC().UnixNano()}
	if _, err := RequireRegisteredTenantCostDownstream(tenantID); err != nil {
		return result, err
	}
	if err := DB.Model(&TenantCostEvent{}).Where("tenant_id = ?", tenantID).Count(&result.ExpectedEvents).Error; err != nil {
		return result, err
	}
	// An event is durable as soon as it is appended, but it is not published
	// to the partner sink until its outbox row reaches sent.  Keep these two
	// notions separate so the reconciliation endpoint exposes real lag.
	if err := DB.Model(&TenantCostOutbox{}).Where("tenant_id = ? AND state = ?", tenantID, TenantOutboxSent).Count(&result.PublishedEvents).Error; err != nil {
		return result, err
	}
	for evidence, target := range map[string]*int64{
		TenantCostEvidenceExact: &result.ExactCount, TenantCostEvidenceObserved: &result.ObservedCount,
		TenantCostEvidenceEstimated: &result.EstimatedCount, TenantCostEvidenceFallback: &result.FallbackCount,
		TenantCostEvidenceUnknown: &result.UnknownCount,
	} {
		var count int64
		if err := DB.Model(&TenantCostEvent{}).Where("tenant_id = ? AND evidence = ?", tenantID, evidence).Count(&count).Error; err != nil {
			return result, err
		}
		*target = count
	}
	if err := DB.Model(&TenantCostOutbox{}).Where("tenant_id = ? AND state IN ?", tenantID, []string{TenantOutboxReady, TenantOutboxPending, TenantOutboxFailed}).Count(&result.PendingOutbox).Error; err != nil {
		return result, err
	}
	var oldest TenantCostOutbox
	if err := DB.Where("tenant_id = ? AND state IN ?", tenantID, []string{TenantOutboxReady, TenantOutboxPending, TenantOutboxFailed}).Order("created_at asc").First(&oldest).Error; err == nil {
		result.OldestPendingAt = oldest.CreatedAt
	}
	var latest TenantCostEvent
	if err := DB.Where("tenant_id = ?", tenantID).Order("sequence desc").First(&latest).Error; err == nil {
		result.LastSequence = latest.Sequence
	}
	if err := DB.Model(&TenantCostProjectionRetry{}).Where("tenant_id = ? AND state IN ?", tenantID,
		[]string{TenantCostProjectionPending, TenantCostProjectionFailed}).Count(&result.PendingProjectionRetries).Error; err != nil {
		return result, err
	}
	var oldestProjection TenantCostProjectionRetry
	if err := DB.Where("tenant_id = ? AND state IN ?", tenantID,
		[]string{TenantCostProjectionPending, TenantCostProjectionFailed}).Order("created_at asc").First(&oldestProjection).Error; err == nil {
		result.OldestProjectionRetryAt = oldestProjection.CreatedAt
	}
	if err := DB.Model(&TenantCostEvent{}).Where("tenant_id = ? AND status = ?", tenantID, TenantCostStatusReview).Count(&result.ConflictCount).Error; err != nil {
		return result, err
	}
	if result.PendingOutbox > 0 {
		result.Blocker = "outbox_pending"
	} else if result.PendingProjectionRetries > 0 {
		result.Blocker = "projection_retry_pending"
	}
	return result, nil
}

func HashTenantReportingToken(secret string) string {
	sum := sha256.Sum256([]byte(strings.TrimSpace(secret)))
	return hex.EncodeToString(sum[:])
}

func GenerateTenantReportingToken() (string, error) {
	buf := make([]byte, 32)
	if _, err := rand.Read(buf); err != nil {
		return "", err
	}
	return "tvr_" + hex.EncodeToString(buf), nil
}

func CreateTenantReportingToken(tenantID, permission string, expiresAt int64) (*TenantReportingToken, string, error) {
	tenantID = strings.TrimSpace(tenantID)
	if strings.TrimSpace(permission) != TenantCostReportingPermission {
		return nil, "", ErrTenantCostScope
	}
	if _, err := RequireActiveTenantCostDownstream(tenantID); err != nil {
		return nil, "", err
	}
	if DB == nil {
		return nil, "", errors.New("database is not initialized")
	}
	secret, err := GenerateTenantReportingToken()
	if err != nil {
		return nil, "", err
	}
	now := common.GetTimestamp()
	row := &TenantReportingToken{TenantID: tenantID, TokenID: "trt_" + uuid.New().String(), TokenHash: HashTenantReportingToken(secret), Permission: permission, ExpiresAt: expiresAt, CreatedAt: now}
	if err := DB.Create(row).Error; err != nil {
		return nil, "", err
	}
	return row, secret, nil
}

func LookupTenantReportingToken(secret string) (*TenantReportingToken, error) {
	secret = strings.TrimSpace(secret)
	if secret == "" || DB == nil {
		return nil, ErrTenantCostReportingToken
	}
	var row TenantReportingToken
	if err := DB.Where("token_hash = ?", HashTenantReportingToken(secret)).First(&row).Error; err != nil {
		return nil, ErrTenantCostReportingToken
	}
	now := common.GetTimestamp()
	if _, scopeErr := RequireActiveTenantCostDownstream(row.TenantID); scopeErr != nil || row.RevokedAt > 0 || (row.ExpiresAt > 0 && row.ExpiresAt <= now) {
		return nil, ErrTenantCostReportingToken
	}
	// Last-used is telemetry only; failure to update it must not turn a valid
	// read into a 5xx response.
	_ = DB.Model(&TenantReportingToken{}).Where("id = ?", row.ID).Update("last_used_at", now).Error
	return &row, nil
}

func UpsertTenantRouteProfile(profile *TenantRouteProfile) error {
	if err := ValidateTenantRouteProfile(profile); err != nil {
		return err
	}
	if DB == nil {
		return errors.New("database is not initialized")
	}
	now := common.GetTimestamp()
	if profile.CreatedAt == 0 {
		profile.CreatedAt = now
	}
	profile.UpdatedAt = now
	return DB.Transaction(func(tx *gorm.DB) error {
		if _, err := requireTenantCostDownstreamTx(tx, profile.TenantID, profile.Status == TenantBindingEnabled); err != nil {
			return err
		}
		// Revalidate inside the transaction so a concurrent profile mutation
		// cannot make an enabled binding point at a mismatched pool/tier or an
		// invalid anonymous contract after the initial read.
		if err := ValidateTenantRouteProfile(profile); err != nil {
			return err
		}
		var existing TenantRouteProfile
		err := lockForUpdate(tx).Where("tenant_id = ? AND profile_id = ?", profile.TenantID, profile.ProfileID).First(&existing).Error
		if errors.Is(err, gorm.ErrRecordNotFound) {
			return tx.Create(profile).Error
		}
		if err != nil {
			return err
		}
		if existing.ID != 0 {
			var enabledBindings int64
			if err := tx.Model(&TenantCredentialBinding{}).Where("tenant_id = ? AND route_profile_id = ? AND binding_status = ?", profile.TenantID, profile.ProfileID, TenantBindingEnabled).Count(&enabledBindings).Error; err != nil {
				return err
			}
			if enabledBindings > 0 {
				if profile.Status != TenantBindingEnabled || existing.ProductPoolID != profile.ProductPoolID ||
					existing.ServiceTier != profile.ServiceTier || existing.Revision != profile.Revision ||
					existing.ModelSelector != profile.ModelSelector || existing.MinRatioPPM != profile.MinRatioPPM ||
					existing.MaxRatioPPM != profile.MaxRatioPPM || existing.ProviderClass != profile.ProviderClass ||
					existing.CandidateSpec != profile.CandidateSpec || existing.CapabilitySpec != profile.CapabilitySpec {
					return fmt.Errorf("enabled bindings require an immutable enabled route profile revision")
				}
			}
		}
		profile.ID = existing.ID
		return tx.Save(profile).Error
	})
}

// TenantCostCatalogSlot is the deliberately anonymous catalog representation
// exposed to a reporting consumer. Physical candidates and provider details
// remain in TenantRouteProfile.CandidateSpec/ProviderClass and never cross
// this boundary.
type TenantCostCatalogSlot struct {
	ProductPoolID        string `json:"product_pool_id"`
	ProductFamily        string `json:"product_family"`
	ServiceTier          string `json:"service_tier"`
	PublicLabel          string `json:"public_label"`
	Status               string `json:"status"`
	KeyID                string `json:"key_id,omitempty"`
	KeySuffix            string `json:"key_suffix,omitempty"`
	RouteProfileID       string `json:"route_profile_id,omitempty"`
	ProfileRevision      string `json:"profile_revision,omitempty"`
	ModelCatalogRevision string `json:"model_catalog_revision,omitempty"`
	DisabledReason       string `json:"disabled_reason,omitempty"`
}

// BuildTenantCostCatalog returns every explicitly registered public slot even
// when it has not yet been bound. Unbound slots are pending rather than
// fabricated as available.
func BuildTenantCostCatalog() ([]TenantCostCatalogSlot, error) {
	return BuildTenantCostCatalogForTenant(DefaultTenantCostTenantID)
}

func BuildTenantCostCatalogForTenant(tenantID string) ([]TenantCostCatalogSlot, error) {
	tenantID = strings.TrimSpace(tenantID)
	if _, err := RequireRegisteredTenantCostDownstream(tenantID); err != nil {
		return nil, err
	}
	slots, err := ListTenantCostProductSlotsForTenant(tenantID, 500)
	if err != nil {
		return nil, err
	}
	bindings, err := ListTenantCredentialBindingsForTenant(tenantID, 500)
	if err != nil {
		return nil, err
	}
	profiles, err := ListTenantRouteProfilesForTenant(tenantID, 500)
	if err != nil {
		return nil, err
	}
	type bindingView struct {
		keyID, suffix, profileRevision, catalogRevision, profileID, status string
	}
	bindingsBySlot := make(map[string]bindingView, len(bindings))
	for _, binding := range bindings {
		if binding.CredentialPurpose != TenantCredentialPurposeDataPlane {
			continue
		}
		key := binding.ProductPoolID + "\x00" + binding.ServiceTier
		bindingsBySlot[key] = bindingView{binding.KeyID, binding.KeySuffix, binding.ProfileRevision,
			binding.ModelCatalogRevision, binding.RouteProfileID, binding.BindingStatus}
	}
	profilesByID := make(map[string]TenantRouteProfile, len(profiles))
	for _, profile := range profiles {
		profilesByID[profile.ProfileID] = profile
	}
	result := make([]TenantCostCatalogSlot, 0, len(slots))
	for _, registered := range slots {
		validated := registered
		if validated.TenantID != tenantID || normalizeTenantCostProductSlot(&validated) != nil {
			return nil, fmt.Errorf("%w: malformed registered product slot", ErrTenantCostInvalidInput)
		}
		pool, tier := validated.ProductPoolID, validated.ServiceTier
		slot := TenantCostCatalogSlot{
			ProductPoolID: pool, ProductFamily: validated.ProductFamily, ServiceTier: tier,
			PublicLabel: validated.PublicLabel, Status: TenantBindingPending, DisabledReason: "尚未绑定服务凭据",
		}
		key := pool + "\x00" + tier
		if binding, ok := bindingsBySlot[key]; ok {
			// Validate every server-owned identity before copying any status into
			// the public projection.  In particular, never let a later assignment
			// overwrite the disabled result for a malformed key/profile.
			identifiersValid := validTenantCostPublicIdentifier(binding.keyID, 128) &&
				validTenantCostKeySuffix(binding.suffix) &&
				validTenantCostPublicIdentifier(binding.profileID, 128) &&
				validTenantCostPublicIdentifier(binding.profileRevision, 96) &&
				validTenantCostPublicIdentifier(binding.catalogRevision, 96) &&
				validTenantBindingStatus(binding.status)
			profile, profileExists := profilesByID[binding.profileID]
			profileValid := profileExists && profile.TenantID == tenantID &&
				profile.ProductPoolID == pool && profile.ServiceTier == tier &&
				profile.PublicLabel == validated.PublicLabel &&
				profile.Revision == binding.profileRevision && ValidateTenantRouteProfile(&profile) == nil
			if !identifiersValid {
				slot.Status = TenantBindingDisabled
				slot.DisabledReason = "服务标识或绑定状态未通过校验"
			} else if !profileExists {
				slot.Status = TenantBindingDisabled
				slot.DisabledReason = "绑定的路由档位不存在"
			} else if !profileValid {
				slot.Status = TenantBindingDisabled
				slot.DisabledReason = "路由档位与产品/档位合同不一致"
			} else {
				slot.KeyID, slot.KeySuffix = binding.keyID, binding.suffix
				slot.ProfileRevision, slot.ModelCatalogRevision = binding.profileRevision, binding.catalogRevision
				slot.RouteProfileID = binding.profileID
				effectiveStatus := binding.status
				profileStatus := strings.TrimSpace(profile.Status)
				if profileStatus == "" {
					profileStatus = TenantBindingPending
				}
				if binding.status == TenantBindingEnabled && profileStatus != TenantBindingEnabled {
					effectiveStatus = TenantBindingDisabled
					slot.DisabledReason = "服务绑定未满足已启用路由档位"
				} else if profileStatus == TenantBindingDisabled || profileStatus == TenantBindingRevoked {
					effectiveStatus = profileStatus
					slot.DisabledReason = profile.DisabledReason
				} else if binding.status != TenantBindingEnabled {
					slot.DisabledReason = profile.DisabledReason
					if slot.DisabledReason == "" {
						slot.DisabledReason = "服务凭据尚未启用"
					}
				}
				slot.Status = effectiveStatus
				if slot.Status == TenantBindingEnabled {
					slot.DisabledReason = ""
				}
			}
		}
		result = append(result, slot)
	}
	return result, nil
}

type TenantCostOperationSummary struct {
	EventCount       int64            `json:"event_count"`
	GatewayChargeQuota int64            `json:"gateway_charge_quota"`
	ByProductPool    map[string]int64 `json:"by_product_pool"`
	ByServiceTier    map[string]int64 `json:"by_service_tier"`
	ByEvidence       map[string]int64 `json:"by_evidence"`
	GeneratedAt      int64            `json:"generated_at"`
}

// SummarizeTenantCostEvents is intentionally bounded. It is an operator
// overview, not a replacement for the sequence API or a full-table report.
func SummarizeTenantCostEvents(tenantID string, since int64, limit int) (TenantCostOperationSummary, error) {
	tenantID = strings.TrimSpace(tenantID)
	result := TenantCostOperationSummary{ByProductPool: map[string]int64{}, ByServiceTier: map[string]int64{}, ByEvidence: map[string]int64{}, GeneratedAt: time.Now().UTC().UnixNano()}
	if _, err := RequireRegisteredTenantCostDownstream(tenantID); err != nil {
		return result, err
	}
	if limit <= 0 || limit > 5000 {
		limit = 5000
	}
	query := DB.Where("tenant_id = ?", tenantID)
	if since > 0 {
		query = query.Where("occurred_at >= ?", since)
	}
	var rows []TenantCostEvent
	if err := query.Order("sequence asc").Limit(limit).Find(&rows).Error; err != nil {
		return result, err
	}
	for _, row := range rows {
		result.EventCount++
		result.GatewayChargeQuota += row.GatewayChargeQuota
		result.ByProductPool[row.ProductPoolID]++
		result.ByServiceTier[row.ServiceTier]++
		result.ByEvidence[row.Evidence]++
	}
	return result, nil
}
