package media

import (
	"os"

	"github.com/QuantumNous/new-api/model"
	mediasetting "github.com/QuantumNous/new-api/setting/media_setting"

	"github.com/stretchr/testify/require"
	"gorm.io/gorm"
)

func preparePhysicalStorageTest(t *testing.T) (string, *gorm.DB) {
	t.Helper()
	previousDB := model.DB
	db, err := gorm.Open(sqlite.Open("file:"+t.Name()+"?mode=memory&cache=shared"), &gorm.Config{})
	require.NoError(t, err)
	model.DB = db
	t.Cleanup(func() { model.DB = previousDB })
	require.NoError(t, db.AutoMigrate(
		&model.MediaBlob{}, &model.TaskArtifact{}, &model.MediaStorageState{},
		&model.MediaStorageReservation{}, &model.MediaCleanupRun{}, &model.MediaModerationEvent{},
	))

	previousSetting := mediasetting.Get()
	root := t.TempDir()
	next := previousSetting
	next.StoragePath = root
	next.GlobalMaxBytes = 10 * 1024 * 1024 * 1024
	next.PerUserMaxBytes = 2 * 1024 * 1024 * 1024
	next.ReservedFreeBytes = 1
	require.NoError(t, mediasetting.Replace(next))
	t.Cleanup(func() { require.NoError(t, mediasetting.Replace(previousSetting)) })
	require.NoError(t, EnsureStorage())
	return root, db
}

func TestReconcileStorageStateCountsTrackedResidualAndTemporaryBytes(t *testing.T) {
	root, db := preparePhysicalStorageTest(t)
	trackedKey := filepath.Join("objects", "tracked.bin")
	require.NoError(t, os.WriteFile(filepath.Join(root, trackedKey), make([]byte, 11), 0o600))
	require.NoError(t, os.WriteFile(filepath.Join(root, "objects", "residual.bin"), make([]byte, 23), 0o600))
	require.NoError(t, os.WriteFile(filepath.Join(root, "tmp", "upload.tmp"), make([]byte, 7), 0o600))
	require.NoError(t, db.Create(&model.MediaBlob{
		PublicID: "tracked", UserID: 1, StorageBackend: model.MediaStorageBackendLocal,
		ObjectKey: trackedKey, SizeBytes: 11, Status: model.MediaBlobStatusReady,
	}).Error)

	require.NoError(t, ReconcileStorageState())
	var state model.MediaStorageState
	require.NoError(t, db.First(&state, 1).Error)
	assert.EqualValues(t, 11, state.UsedBytes)
	assert.EqualValues(t, 41, state.PhysicalUsedBytes)
	assert.EqualValues(t, 23, state.UntrackedBytes)
	assert.EqualValues(t, 7, state.TemporaryBytes)
	assert.EqualValues(t, 41, effectiveMediaUsedBytes(state))
	assert.NotZero(t, state.LastScannedAt)
}

func TestDeleteBlobMissingLocalFilePreservesReadyMetadata(t *testing.T) {
	_, db := preparePhysicalStorageTest(t)
	blob := &model.MediaBlob{
		PublicID: "missing-local", UserID: 1, StorageBackend: model.MediaStorageBackendLocal,
		ObjectKey: filepath.Join("objects", "missing.bin"), SizeBytes: 19,
		Status: model.MediaBlobStatusReady, ModerationStatus: model.MediaModerationApproved,
	}
	require.NoError(t, db.Create(blob).Error)

	err := DeleteBlob(blob.PublicID, "test-owner", false)
	require.Error(t, err)
	require.NoError(t, db.First(blob, blob.ID).Error)
	assert.Zero(t, blob.DeletedAt)
	assert.Equal(t, model.MediaBlobStatusReady, blob.Status)
	assert.Equal(t, filepath.Join("objects", "missing.bin"), blob.ObjectKey)
}

func TestReserveStorageUsesPhysicalResidualAgainstGlobalLimit(t *testing.T) {
	root, _ := preparePhysicalStorageTest(t)
	setting := mediasetting.Get()
	setting.GlobalMaxBytes = 50
	setting.PerUserMaxBytes = 50
	require.NoError(t, mediasetting.Replace(setting))
	require.NoError(t, os.WriteFile(
		filepath.Join(root, "objects", "residual-capacity.bin"),
		make([]byte, 45),
		0o600,
	))
	require.NoError(t, ReconcileStorageState())

	_, err := reserveStorage(1, 10)
	assert.ErrorIs(t, err, ErrMediaCapacity)
}

func TestRequestSpoolReservationGrowsByActualBytesAndReleasesOnce(t *testing.T) {
	_, db := preparePhysicalStorageTest(t)
	setting := mediasetting.Get()
	setting.GlobalMaxBytes = 64 * 1024 * 1024
	setting.PerUserMaxBytes = 64 * 1024 * 1024
	setting.MaxFileBytes = 32 * 1024 * 1024
	setting.MaxRequestBytes = 32 * 1024 * 1024
	require.NoError(t, mediasetting.Replace(setting))

	reservation, err := BeginRequestSpoolReservation(setting.MaxRequestBytes, 0)
	require.NoError(t, err)
	var state model.MediaStorageState
	require.NoError(t, db.First(&state, 1).Error)
	assert.EqualValues(t, 8*1024*1024, state.ReservedBytes)

	require.NoError(t, reservation.GrowTo(8*1024*1024+1))
	require.NoError(t, db.First(&state, 1).Error)
	assert.EqualValues(t, 16*1024*1024, state.ReservedBytes)

	reservation.Release()
	reservation.Release()
	require.NoError(t, db.First(&state, 1).Error)
	assert.Zero(t, state.ReservedBytes)
	assert.Error(t, reservation.GrowTo(1))
	var row model.MediaStorageReservation
	require.NoError(t, db.Where("token <> ''").First(&row).Error)
	assert.Equal(t, "released", row.State)
}
