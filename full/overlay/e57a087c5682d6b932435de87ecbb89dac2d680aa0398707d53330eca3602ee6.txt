type staticCatalogSource struct {
	mu        sync.Mutex
	abilities []model.Ability
	channels  []model.Channel
	blocks    []model.SmartRouteBlock
	loads     int
	err       error
}

func (source *staticCatalogSource) LoadActiveSmartRouteBlocks(_ context.Context, now time.Time) ([]model.SmartRouteBlock, error) {
	source.mu.Lock()
	defer source.mu.Unlock()
	result := make([]model.SmartRouteBlock, 0, len(source.blocks))
	for _, block := range source.blocks {
		if block.ExpiresAt == 0 || block.ExpiresAt > now.Unix() {
			result = append(result, block)
		}
	}
	return result, nil
}

func (source *staticCatalogSource) LoadEnabledRoutingState(context.Context) ([]model.Ability, []model.Channel, error) {
	source.mu.Lock()
	defer source.mu.Unlock()
	source.loads++
	if source.err != nil {
		return nil, nil, source.err
	}
	return append([]model.Ability(nil), source.abilities...), append([]model.Channel(nil), source.channels...), nil
}

func (source *staticCatalogSource) loadCount() int {
	source.mu.Lock()
	defer source.mu.Unlock()
	return source.loads
}

func onlyContract(t *testing.T, catalog smartrouter.CertifiedRouteCatalog) smartrouter.ContractCatalog {
	t.Helper()
	require.Len(t, catalog.Contracts, 1)
	for _, contract := range catalog.Contracts {
		return contract
	}
	return smartrouter.ContractCatalog{}
}

func contractByEndpoint(t *testing.T, catalog smartrouter.CertifiedRouteCatalog, endpoint string) smartrouter.ContractCatalog {
	t.Helper()
	for _, contract := range catalog.Contracts {
		if contract.Contract.Endpoint == endpoint {
			return contract
		}
	}
	require.Failf(t, "missing contract", "endpoint %q was not published", endpoint)
	return smartrouter.ContractCatalog{}
}

func publishedEndpointSet(catalog smartrouter.CertifiedRouteCatalog) map[string]bool {
	endpoints := make(map[string]bool, len(catalog.Contracts))
	for _, contract := range catalog.Contracts {
		endpoints[contract.Contract.Endpoint] = true
	}
	return endpoints
}

	firstRoutes := firstContract.Pools[smartrouter.AutomaticPoolID].Candidates

		if route.ChannelID == 1 {
			firstRoute = route
		}

	var secondRoute smartrouter.CertifiedRoute
	for _, route := range secondRoutes {
		if route.ChannelID == 1 {
			secondRoute = route
		}
	}
	require.NotEmpty(t, secondRoute.CacheNamespaceRevision)
	assert.Equal(t, firstRoute.CacheDomain, secondRoute.CacheDomain)
	assert.NotEqual(t, firstRoute.CacheNamespaceRevision, secondRoute.CacheNamespaceRevision)
}

	setting.Discovery.AllowedTextEndpoints = []constant.EndpointType{
		constant.EndpointTypeOpenAI,
		constant.EndpointTypeOpenAIResponse,
		constant.EndpointTypeAnthropic,
	}
	setting.ChannelProfiles = map[string]smartroutersetting.ChannelProfileSetting{
		"25": {EndpointAllow: []constant.EndpointType{
			constant.EndpointTypeAnthropic,
			constant.EndpointTypeOpenAI,
			constant.EndpointTypeOpenAIResponse,
		}},
	}
	priority := int64(10)
	source := &staticCatalogSource{
		abilities: []model.Ability{
			{Group: "economy", Model: "gpt-5.6-sol", ChannelId: 50, Enabled: true, Priority: &priority},
			{Group: "economy", Model: "gpt-5.6-sol", ChannelId: 25, Enabled: true, Priority: &priority},
			{Group: "fallback", Model: "gpt-5.6-sol", ChannelId: 58, Enabled: true, Priority: &priority},

			{Id: 50, Type: constant.ChannelTypeOpenAI, Status: common.ChannelStatusEnabled},
			{Id: 25, Type: constant.ChannelTypeAnthropic, Status: common.ChannelStatusEnabled},
			{Id: 58, Type: constant.ChannelTypeOpenAI, Status: common.ChannelStatusEnabled},
		},
	}

	catalog, _, err := BuildCertifiedRouteCatalog(context.Background(), setting, source)
	require.NoError(t, err)

	mapping, err := common.Marshal(map[string]string{"gpt-test": "upstream-a", "upstream-a": "upstream-final"})

	mappingJSON := string(mapping)
	source.channels[0].ModelMapping = &mappingJSON

	catalog, report, err := BuildCertifiedRouteCatalog(context.Background(), setting, source)
	require.NoError(t, err)
	require.NoError(t, catalog.Validate())

	assert.Equal(t, requestFingerprint, contract.Contract.CapabilityFingerprint)
	balanced := contract.Pools["balanced-v1"].Candidates
	require.Len(t, balanced, 2)
	assert.Equal(t, 1, balanced[0].ChannelID)
	assert.Equal(t, "upstream-final", balanced[0].UpstreamModel)
	assert.False(t, balanced[0].StableFallback)
	assert.Equal(t, 2, balanced[1].ChannelID)
	assert.False(t, balanced[1].StableFallback)
	assert.Equal(t, "channel:1", balanced[0].CacheDomain)
	assert.Equal(t, "channel:1", balanced[0].CapacityDomain)
	assert.NotEmpty(t, balanced[0].ChannelGeneration)
	assert.NotContains(t, balanced[0].ChannelGeneration, "upstream")
	assert.NotEqual(t, balanced[0].RouteID, balanced[1].RouteID)
	_, unsafePoolPublished := contract.Pools["stable-v1"]
	assert.False(t, unsafePoolPublished, "a single-channel pool must not be exposed to smart keys")
}

func TestAutomaticCatalogKeepsCanonicalModelsIsolatedAcrossExactMappings(t *testing.T) {
	setting := smartCatalogTestSetting(constant.EndpointTypeOpenAI)
	priority := int64(1)
	mappingOne, err := common.Marshal(map[string]string{"gpt-5.6-sol": "provider-a-sol"})
	require.NoError(t, err)
	mappingTwo, err := common.Marshal(map[string]string{"gpt-5.6-sol": "provider-b-sol"})
	require.NoError(t, err)
	mappingNear, err := common.Marshal(map[string]string{"gpt-5.6-sol-preview": "provider-preview"})
	require.NoError(t, err)
	mappingClaude, err := common.Marshal(map[string]string{"claude-opus-4-6": "provider-claude"})
	require.NoError(t, err)
	mappingOneJSON := string(mappingOne)
	mappingTwoJSON := string(mappingTwo)
	mappingNearJSON := string(mappingNear)
	mappingClaudeJSON := string(mappingClaude)
	source := &staticCatalogSource{
		abilities: []model.Ability{
			{Group: "economy", Model: "gpt-5.6-sol", ChannelId: 1, Enabled: true, Priority: &priority},
			{Group: "fallback", Model: "gpt-5.6-sol", ChannelId: 2, Enabled: true, Priority: &priority},
			{Group: "economy", Model: "gpt-5.6-sol-preview", ChannelId: 3, Enabled: true, Priority: &priority},
			{Group: "fallback", Model: "claude-opus-4-6", ChannelId: 4, Enabled: true, Priority: &priority},
		},
		channels: []model.Channel{
			{Id: 1, Type: constant.ChannelTypeOpenAI, Status: common.ChannelStatusEnabled, ModelMapping: &mappingOneJSON},
			{Id: 2, Type: constant.ChannelTypeOpenAI, Status: common.ChannelStatusEnabled, ModelMapping: &mappingTwoJSON},
			{Id: 3, Type: constant.ChannelTypeOpenAI, Status: common.ChannelStatusEnabled, ModelMapping: &mappingNearJSON},
			{Id: 4, Type: constant.ChannelTypeOpenAI, Status: common.ChannelStatusEnabled, ModelMapping: &mappingClaudeJSON},
		},
	}

	catalog, _, err := BuildCertifiedRouteCatalog(context.Background(), setting, source)
	require.NoError(t, err)
	contractID, _, _, err := SmartContractIdentity("gpt-5.6-sol", "/v1/chat/completions")
	require.NoError(t, err)
	contract, exists := catalog.Contracts[contractID]
	require.True(t, exists)
	candidates := contract.Pools[SmartRouteAutoDiscoveredPool].Candidates
	require.Len(t, candidates, 2)
	assert.ElementsMatch(t, []int{1, 2}, []int{candidates[0].ChannelID, candidates[1].ChannelID})
	assert.ElementsMatch(t, []string{"provider-a-sol", "provider-b-sol"}, []string{candidates[0].UpstreamModel, candidates[1].UpstreamModel})
	for _, candidate := range candidates {
		assert.NotEqual(t, 3, candidate.ChannelID, "a near model name must remain a separate contract")
		assert.NotEqual(t, 4, candidate.ChannelID, "a different model family must remain a separate contract")
	}
}

func TestAutomaticCatalogAppliesDurableBlocksAtPreciseScopes(t *testing.T) {
	setting := smartCatalogTestSetting(constant.EndpointTypeOpenAI)
	source := smartCatalogFixture("gpt-test", constant.ChannelTypeOpenAI)
	source.blocks = []model.SmartRouteBlock{{
		Id: 1, Scope: SmartRouteBlockScopeGroupModel, Group: "economy",
		ModelPattern: "gpt-*", Source: SmartRouteBlockSourceManual,
	}}

	catalog, report, err := BuildCertifiedRouteCatalog(context.Background(), setting, source)
	require.NoError(t, err)
	assert.Equal(t, 2, report.BlacklistedRoutes, "chat and completions routes in the blocked group are excluded")
	chat := contractByEndpoint(t, catalog, SmartEndpointChat)
	require.Len(t, chat.Pools["balanced-v1"].Candidates, 1)
	assert.Equal(t, "fallback", chat.Pools["balanced-v1"].Candidates[0].Group)

	// An expired manual block is no longer authoritative at the next catalog
	// refresh and the previously excluded route is discovered again.
	source.blocks[0].ExpiresAt = time.Now().Add(-time.Minute).Unix()
	refreshed, report, err := BuildCertifiedRouteCatalog(context.Background(), setting, source)
	require.NoError(t, err)
	assert.Zero(t, report.BlacklistedRoutes)
	assert.Len(t, contractByEndpoint(t, refreshed, SmartEndpointChat).Pools["balanced-v1"].Candidates, 2)
}

func TestCatalogIgnoresLegacyAutomaticBlocksButKeepsManualAuthority(t *testing.T) {
	source := smartCatalogFixture("gpt-5.6", constant.ChannelTypeOpenAI)
	setting := smartCatalogTestSetting(constant.EndpointTypeOpenAI)

	initial, _, err := BuildCertifiedRouteCatalog(context.Background(), setting, source)
	require.NoError(t, err)
	var routes []smartrouter.CertifiedRoute
	var contracts []smartrouter.RouteContract
	for _, entry := range initial.Contracts {
		pool := entry.Pools[SmartRouteAutoDiscoveredPool]
		for _, route := range pool.Candidates {
			routes = append(routes, route)
			contracts = append(contracts, entry.Contract)
		}
	}
	require.NotEmpty(t, routes)
	source.blocks = make([]model.SmartRouteBlock, 0, len(routes))
	for index, route := range routes {
		contract := contracts[index]
		source.blocks = append(source.blocks, model.SmartRouteBlock{
			Scope: SmartRouteBlockScopeRoute, RouteId: route.RouteID,
			Group: route.Group, ChannelId: route.ChannelID,
			ContractId: contract.ContractID, Endpoint: contract.Endpoint,
			Source: SmartRouteBlockSourceAutoHealth, CreatedAt: 1_725_000_000, Epoch: 4,
		})
	}

	blockedCatalog, report, err := BuildCertifiedRouteCatalog(context.Background(), setting, source)
	require.NoError(t, err)
	assert.NotEmpty(t, blockedCatalog.Contracts)
	assert.Empty(t, report.BlockedRoutes)
	assert.Zero(t, report.BlacklistedRoutes)

	encoded, err := json.Marshal(report)
	require.NoError(t, err)
	// The report is safe to expose to the admin UI: internal route/channel
	// identifiers are retained only in the in-process bridge used by users'
	// redacted status endpoint.
	for _, route := range routes {
		assert.NotContains(t, string(encoded), route.RouteID)
	}
	assert.NotContains(t, string(encoded), "blocked_routes")
}

func TestCatalogManagerRetainsLastValidatedCatalogOnTransientSourceFailure(t *testing.T) {
	setting := smartCatalogTestSetting(constant.EndpointTypeOpenAI)
	source := smartCatalogFixture("gpt-test", constant.ChannelTypeOpenAI)
	manager := NewCatalogManager(source, func() smartroutersetting.Setting { return setting })

	first, err := manager.Refresh(context.Background())
	require.NoError(t, err)
	source.mu.Lock()
	source.err = errors.New("temporary database failure")
	source.mu.Unlock()
	manager.Invalidate()
	_, err = manager.Refresh(context.Background())
	require.Error(t, err)

	current, exists := manager.Current()
	require.True(t, exists)
	assert.Same(t, first, current)

	source.mu.Lock()
	source.err = nil
	source.channels = nil
	source.mu.Unlock()
	_, err = manager.Refresh(context.Background())
	require.ErrorIs(t, err, ErrNoCertifiedSmartRoutes)
	_, exists = manager.Current()
	assert.False(t, exists, "authoritative revocation must not retain the old catalog")

	setting.MaxAttempts = 0
	_, err = manager.Refresh(context.Background())
	require.Error(t, err)
	_, exists = manager.Current()
	assert.False(t, exists)

	setting.Enabled = false
	_, err = manager.Refresh(context.Background())
	require.ErrorIs(t, err, ErrSmartRouterDisabled)
	_, exists = manager.Current()
	assert.False(t, exists)
}

func TestGlobalSettingPublicationInvalidatesDerivedStateAndDisableStopsImmediately(t *testing.T) {
	original := smartroutersetting.GetSmartRouterSettingSnapshot()
	defer func() { require.NoError(t, smartroutersetting.ReplaceSmartRouterSetting(original)) }()
	setting := smartCatalogTestSetting(constant.EndpointTypeOpenAI)
	require.NoError(t, smartroutersetting.ReplaceSmartRouterSetting(setting))

	catalogs := NewCatalogManager(smartCatalogFixture("gpt-test", constant.ChannelTypeOpenAI), nil)
	defer catalogs.Close()
	_, err := catalogs.Refresh(context.Background())
	require.NoError(t, err)
	ratios := &fakeSmartRatioSource{
		revision: 1,
		ratios: map[string]map[string]int64{
			"standard": {"economy": 20_000, "fallback": 50_000},
		},
	}
	prices := NewPriceSnapshotManager(catalogs, ratios)
	defer prices.Close()
	_, err = prices.Refresh(context.Background(), "standard")
	require.NoError(t, err)
	_, exists := prices.Current("standard")
	require.True(t, exists)

	setting.DefaultForNewKeys = true
	require.NoError(t, smartroutersetting.ReplaceSmartRouterSetting(setting))

	_, exists = catalogs.Current()

}
