export const smartRouterFormSchema = z
  .object({
    setting_base: z.custom<SmartRouterSetting>(),
    default_policy: z.custom<SmartRouterPolicy>(),
    enabled: z.boolean(),
    default_for_new_keys: z.boolean(),
    default_max_effective_ratio: z
      .number()
      .finite()
      .positive('Default group ratio ceiling must be positive')
      .max(MAX_SMART_ROUTER_EFFECTIVE_RATIO),
    actual_input_cost_ranking: z.boolean(),
    absolute_max_effective_ratio: z
      .number()
      .finite()
      .positive('Absolute group ratio ceiling must be positive')
      .max(MAX_SMART_ROUTER_EFFECTIVE_RATIO),
    attempt_limit_enabled: z.boolean(),
    recovery_fallback_enabled: z.boolean(),
    site_max_attempts: z.number().int().min(1).max(8),
    max_failure_threshold: z.number().int().min(1).max(100),
    hard_failure_window_seconds: z.number().int().min(60).max(86400),
    catalog_refresh_seconds: z.number().int().min(10).max(3600),
    probe_initial_backoff_seconds: z.number().int().min(1).max(3600),
    probe_max_backoff_seconds: z.number().int().min(1).max(86400),
    max_probe_failures: z.number().int().min(1).max(100),
    max_outage_seconds: z.number().int().min(60).max(604800),
    probe_budget_per_minute: z.number().int().min(1).max(10000),
    probe_max_concurrency: z.number().int().min(1).max(100),
    probe_timeout_seconds: z.number().int().positive().finite(),
    probe_completion_grace_seconds: z.number().int().positive().finite(),
    probe_scan_interval_ms: z.number().int().min(100).max(60000),
    probe_queue_capacity: z.number().int().min(1).max(100000),
    probe_lease_ttl_seconds: z.number().int().min(1).max(172815),
    recovery_fast_interval_seconds: z.number().int().min(1).max(86400),
    recovery_fast_window_seconds: z.number().int().min(1).max(86400),
    recovery_sweep_interval_seconds: z.number().int().min(5).max(3600),
    recovery_canary_initial_traffic_percent: z.number().min(0.0001).max(100),
    recovery_canary_healthy_samples: z.number().int().min(1).max(100),
    recovery_canary_fast_history_samples: z.number().int().min(1).max(100000),
    recovery_canary_fast_reliability_percent: z.number().min(0).max(100),
    recovery_coverage_target: z.number().int().min(1).max(8),
    recovery_backup_evidence_ttl_seconds: z.number().int().min(60).max(604800),
    cold_start_wave_batch_size: z.number().int().min(1).max(100),
    cold_start_wave_max_concurrency: z.number().int().min(1).max(100),
    cold_start_probe_budget_per_minute: z.number().int().min(1).max(10000),
    cold_start_wait_ms: z.number().int().min(1).max(300000),
    cold_start_poll_interval_ms: z.number().int().min(10).max(5000),
    background_probe_enabled: z.boolean(),
    max_queue_wait_ms: z.number().int().min(0).max(60000),
  })
  .superRefine((values, ctx) => {
    if (!values.enabled && values.default_for_new_keys) {
      ctx.addIssue({
        code: 'custom',
        path: ['default_for_new_keys'],
        message: 'Enable smart routing before making it the new-key default',
      })
    }
    if (
      values.default_max_effective_ratio > values.absolute_max_effective_ratio
    ) {
      ctx.addIssue({
        code: 'custom',
        path: ['absolute_max_effective_ratio'],
        message: 'Absolute group ratio ceiling must cover the default ceiling',
      })
    }
    if (
      values.probe_max_backoff_seconds < values.probe_initial_backoff_seconds
    ) {
      ctx.addIssue({
        code: 'custom',
        path: ['probe_max_backoff_seconds'],
        message:
          'Maximum probe backoff cannot be shorter than the initial backoff',
      })
    }
    if (values.max_outage_seconds < values.probe_initial_backoff_seconds) {
      ctx.addIssue({
        code: 'custom',
        path: ['max_outage_seconds'],
        message: 'Maximum outage cannot be shorter than the initial backoff',
      })
    }
    if (
      values.recovery_fast_window_seconds <
      values.recovery_fast_interval_seconds
    ) {
      ctx.addIssue({
        code: 'custom',
        path: ['recovery_fast_window_seconds'],
        message: 'Fast recovery window cannot be shorter than its interval',
      })
    }
    if (
      values.cold_start_wave_max_concurrency > values.cold_start_wave_batch_size
    ) {
      ctx.addIssue({
        code: 'custom',
        path: ['cold_start_wave_max_concurrency'],
        message: 'Cold-start concurrency cannot exceed its batch size',
      })
    }
    if (values.cold_start_poll_interval_ms > values.cold_start_wait_ms) {
      ctx.addIssue({
        code: 'custom',
        path: ['cold_start_poll_interval_ms'],
        message: 'Cold-start polling cannot exceed its wait window',
      })
    }
  })

export type SmartRouterFormValues = z.infer<typeof smartRouterFormSchema>

export function createEmptySmartRouterFormValues(): SmartRouterFormValues {
  return smartRouterSettingToForm(createDefaultSetting())
}

function normalizePolicy(setting: SmartRouterSetting) {
  const parsed = parseSmartRouterPolicy(
    JSON.stringify(setting.default_policy ?? fallbackPolicy)
  )
  return parsed ?? structuredClone(fallbackPolicy)
}

export function smartRouterSettingToForm(
  setting: SmartRouterSetting
): SmartRouterFormValues {
  const defaults = createDefaultSetting()
  const policy = normalizePolicy(setting)
  const defaultRatioPPM =
    setting.default_max_effective_ratio_ppm ??
    (policy.max_effective_ratio_ppm > 0
      ? policy.max_effective_ratio_ppm
      : (defaults.default_max_effective_ratio_ppm ?? 1_000_000))
  const absoluteRatioPPM = Math.max(
    defaultRatioPPM,
    setting.absolute_max_effective_ratio_ppm ??
      defaults.absolute_max_effective_ratio_ppm ??
      10_000_000
  )

  return {
    setting_base: structuredClone(setting),
    default_policy: policy,
    enabled: setting.enabled === true,
    default_for_new_keys: setting.default_for_new_keys === true,
    default_max_effective_ratio: defaultRatioPPM / SMART_ROUTER_RATIO_SCALE,
    absolute_max_effective_ratio: absoluteRatioPPM / SMART_ROUTER_RATIO_SCALE,
    actual_input_cost_ranking: policy.actual_input_cost_ranking !== false,
    attempt_limit_enabled: setting.attempt_limit_enabled === true,
    recovery_fallback_enabled: setting.recovery_fallback_enabled ?? true,
    site_max_attempts: setting.max_attempts || defaults.max_attempts,
    max_failure_threshold:
      setting.max_failure_threshold ?? defaults.max_failure_threshold ?? 10,
    hard_failure_window_seconds:
      setting.hard_failure_window_seconds ??
      defaults.hard_failure_window_seconds ??
      600,
    catalog_refresh_seconds:
      setting.catalog_refresh_seconds || defaults.catalog_refresh_seconds,
    probe_initial_backoff_seconds:
      setting.probe_initial_backoff_seconds ??
      defaults.probe_initial_backoff_seconds ??
      60,
    probe_max_backoff_seconds:
      setting.probe_max_backoff_seconds ??
      defaults.probe_max_backoff_seconds ??
      60,
    max_probe_failures:
      setting.max_probe_failures ?? defaults.max_probe_failures ?? 8,
    max_outage_seconds:
      setting.max_outage_seconds ?? defaults.max_outage_seconds ?? 86400,
    probe_budget_per_minute:
      setting.probe_budget_per_minute ?? defaults.probe_budget_per_minute ?? 30,
    probe_max_concurrency:
      setting.probe_max_concurrency ?? defaults.probe_max_concurrency ?? 3,
    probe_timeout_seconds:
      setting.probe_timeout_seconds ?? defaults.probe_timeout_seconds ?? 45,
    probe_completion_grace_seconds:
      setting.probe_completion_grace_seconds ??
      defaults.probe_completion_grace_seconds ??
      15,
    probe_scan_interval_ms:
      setting.probe_scan_interval_ms ?? defaults.probe_scan_interval_ms ?? 5000,
    probe_queue_capacity:
      setting.probe_queue_capacity ?? defaults.probe_queue_capacity ?? 2048,
    probe_lease_ttl_seconds:
      setting.probe_lease_ttl_seconds ?? defaults.probe_lease_ttl_seconds ?? 75,
    recovery_fast_interval_seconds:
      setting.recovery_fast_interval_seconds ??
      defaults.recovery_fast_interval_seconds ??
      60,
    recovery_fast_window_seconds:
      setting.recovery_fast_window_seconds ??
      defaults.recovery_fast_window_seconds ??
      3600,
    recovery_sweep_interval_seconds:
      setting.recovery_sweep_interval_seconds ??
      defaults.recovery_sweep_interval_seconds ??
      30,
    recovery_canary_initial_traffic_percent:
      (setting.recovery_canary_initial_traffic_ppm ??
        defaults.recovery_canary_initial_traffic_ppm ??
        250_000) / 10_000,
    recovery_canary_healthy_samples:
      setting.recovery_canary_healthy_samples ??
      defaults.recovery_canary_healthy_samples ??
      3,
    recovery_canary_fast_history_samples:
      setting.recovery_canary_fast_history_samples ??
      defaults.recovery_canary_fast_history_samples ??
      5,
    recovery_canary_fast_reliability_percent:
      (setting.recovery_canary_fast_reliability_ppm ??
        defaults.recovery_canary_fast_reliability_ppm ??
        800_000) / 10_000,
    recovery_coverage_target:
      setting.recovery_coverage_target ??
      defaults.recovery_coverage_target ??
      3,
    recovery_backup_evidence_ttl_seconds:
      setting.recovery_backup_evidence_ttl_seconds ??
      defaults.recovery_backup_evidence_ttl_seconds ??
      3600,
    cold_start_wave_batch_size:
      setting.cold_start_wave_batch_size ??
      defaults.cold_start_wave_batch_size ??
      2,
    cold_start_wave_max_concurrency:
      setting.cold_start_wave_max_concurrency ??
      defaults.cold_start_wave_max_concurrency ??
      2,
    cold_start_probe_budget_per_minute:
      setting.cold_start_probe_budget_per_minute ??
      defaults.cold_start_probe_budget_per_minute ??
      60,
    cold_start_wait_ms:
      setting.cold_start_wait_ms ?? defaults.cold_start_wait_ms ?? 60000,
    cold_start_poll_interval_ms:
      setting.cold_start_poll_interval_ms ??
      defaults.cold_start_poll_interval_ms ??
      100,
    background_probe_enabled:
      setting.background_probe_enabled ??
      defaults.background_probe_enabled ??
      true,
    max_queue_wait_ms:
      setting.max_queue_wait_ms ?? defaults.max_queue_wait_ms ?? 1000,
  }
}

export function smartRouterFormToSetting(
  values: SmartRouterFormValues
): SmartRouterSetting {
  const defaultRatioPPM = Math.round(
    values.default_max_effective_ratio * SMART_ROUTER_RATIO_SCALE
  )
  const absoluteRatioPPM = Math.round(
    values.absolute_max_effective_ratio * SMART_ROUTER_RATIO_SCALE
  )
  const policy: SmartRouterPolicy = {
    ...values.default_policy,
    version: 4,
    enabled: values.enabled,
    strategy: values.default_policy.strategy || 'price',
    excluded_groups: values.default_policy.excluded_groups ?? [],
    allow_future_groups: values.default_policy.allow_future_groups !== false,
    manual_group_order: values.default_policy.manual_group_order ?? [],
    max_attempts: Math.min(
      values.default_policy.max_attempts || 3,
      values.site_max_attempts
    ),
    failure_threshold: Math.min(
      values.default_policy.failure_threshold || 3,
      values.max_failure_threshold
    ),
    max_effective_ratio_ppm: 0,
    actual_input_cost_ranking: values.actual_input_cost_ranking,
  }
  const settingBase = structuredClone(values.setting_base)
  delete settingBase.certified_groups
  delete settingBase.text_contracts

  return {
    ...settingBase,
    enabled: values.enabled,
    default_for_new_keys: values.enabled && values.default_for_new_keys,
    default_policy: policy,
    default_max_effective_ratio_ppm: defaultRatioPPM,
    absolute_max_effective_ratio_ppm: absoluteRatioPPM,
    attempt_limit_enabled: values.attempt_limit_enabled,
    recovery_fallback_enabled: values.recovery_fallback_enabled,
    max_attempts: values.site_max_attempts,
    max_failure_threshold: values.max_failure_threshold,
    hard_failure_window_seconds: values.hard_failure_window_seconds,
    catalog_refresh_seconds: values.catalog_refresh_seconds,
    probe_initial_backoff_seconds: values.probe_initial_backoff_seconds,
    probe_max_backoff_seconds: values.probe_max_backoff_seconds,
    max_probe_failures: values.max_probe_failures,
    max_outage_seconds: values.max_outage_seconds,
    probe_budget_per_minute: values.probe_budget_per_minute,
    probe_max_concurrency: values.probe_max_concurrency,
    probe_timeout_seconds: values.probe_timeout_seconds,
    probe_completion_grace_seconds: values.probe_completion_grace_seconds,
    probe_scan_interval_ms: values.probe_scan_interval_ms,
    probe_queue_capacity: values.probe_queue_capacity,
    probe_lease_ttl_seconds: values.probe_lease_ttl_seconds,
    recovery_fast_interval_seconds: values.recovery_fast_interval_seconds,
    recovery_fast_window_seconds: values.recovery_fast_window_seconds,
    recovery_sweep_interval_seconds: values.recovery_sweep_interval_seconds,
    recovery_canary_initial_traffic_ppm: Math.round(
      values.recovery_canary_initial_traffic_percent * 10_000
    ),
    recovery_canary_healthy_samples: values.recovery_canary_healthy_samples,
    recovery_canary_fast_history_samples:
      values.recovery_canary_fast_history_samples,
    recovery_canary_fast_reliability_ppm: Math.round(
      values.recovery_canary_fast_reliability_percent * 10_000
    ),
    recovery_coverage_target: values.recovery_coverage_target,
    recovery_backup_evidence_ttl_seconds:
      values.recovery_backup_evidence_ttl_seconds,
    cold_start_wave_batch_size: values.cold_start_wave_batch_size,
    cold_start_wave_max_concurrency: values.cold_start_wave_max_concurrency,
    cold_start_probe_budget_per_minute:
      values.cold_start_probe_budget_per_minute,
    cold_start_wait_ms: values.cold_start_wait_ms,
    cold_start_poll_interval_ms: values.cold_start_poll_interval_ms,
    background_probe_enabled: values.background_probe_enabled,
    max_queue_wait_ms: values.max_queue_wait_ms,
    kill_switch_mode: 'normal',
  }
}
