func (fixture *imageRouteReconcileFixture) observeJobTransitions(test *testing.T) *[]string {
	test.Helper()
	transitions := []string{}
	err := fixture.database.Callback().Update().After("gorm:update").Register("observe_image_route_state", func(transaction *gorm.DB) {
		if transaction.Error != nil || transaction.RowsAffected != 1 || transaction.Statement.Schema == nil || transaction.Statement.Schema.Table != "image_route_jobs" {
			return
		}
		updates, ok := transaction.Statement.Dest.(map[string]interface{})
		if !ok {
			return
		}
		if state, ok := updates["state"].(string); ok {
			transitions = append(transitions, state)
		}
	})
	require.NoError(test, err)
	return &transitions
}

func TestReconcileImageRouteJobRefundMustBeAppliedAndZero(test *testing.T) {
	cases := []struct {
		name          string
		state         string
		targetQuota   int
		appliedQuota  int
		usageQuota    int
		childQuota    int
		targetStatus  model.TaskStatus
		removeReceipt bool
	}{
		{name: "refund_pending", state: model.TaskBillingReceiptTerminalPending, appliedQuota: 100, usageQuota: 100},
		{name: "pending_even_when_all_amounts_are_zero", state: model.TaskBillingReceiptTerminalPending},
		{name: "submission_pending", state: model.TaskBillingReceiptSubmissionPending},
		{name: "tracking", state: model.TaskBillingReceiptTracking},
		{name: "manual_review", state: model.TaskBillingReceiptReview},
		{name: "missing_receipt", removeReceipt: true},
		{name: "applied_nonzero_target", state: model.TaskBillingReceiptApplied, targetQuota: 1},
		{name: "applied_nonzero_funding", state: model.TaskBillingReceiptApplied, appliedQuota: 1},
		{name: "applied_nonzero_usage", state: model.TaskBillingReceiptApplied, usageQuota: 1},
		{name: "applied_nonzero_child", state: model.TaskBillingReceiptApplied, childQuota: 1},
		{name: "applied_wrong_terminal_target", state: model.TaskBillingReceiptApplied, targetStatus: model.TaskStatusSuccess},
	}
	for _, testCase := range cases {
		test.Run(testCase.name, func(test *testing.T) {
			fixture := newImageRouteReconcileFixture(test)
			if testCase.targetStatus == "" {
				testCase.targetStatus = model.TaskStatusFailure
			}
			fixture.receipt.State, fixture.receipt.TerminalTargetStatus = testCase.state, testCase.targetStatus
			fixture.receipt.TargetQuota, fixture.receipt.AppliedQuota, fixture.receipt.UsageAppliedQuota = testCase.targetQuota, testCase.appliedQuota, testCase.usageQuota
			require.NoError(test, fixture.database.Save(fixture.receipt).Error)
			if testCase.removeReceipt {
				require.NoError(test, fixture.database.Delete(fixture.receipt).Error)
			}
			require.NoError(test, fixture.database.Model(fixture.child).Update("quota", testCase.childQuota).Error)
			before := fixture.accounting(test)
			transitions := fixture.observeJobTransitions(test)
			reconcileImageRouteJob(context.Background(), fixture.job)
			reconcileImageRouteJob(context.Background(), fixture.loadJob(test))
			assert.Equal(test, fixture.job, fixture.loadJob(test))
			assert.Empty(test, *transitions)
			assert.EqualValues(test, model.TaskStatusSubmitted, fixture.loadRoot(test).Status)
			assert.Equal(test, before, fixture.accounting(test))
			assert.FileExists(test, fixture.job.BodyPath)
		})
	}
}

type imageRouteLocalOnlyTransport struct {
	baseURL   string
	transport http.RoundTripper
	blocked   atomic.Int64
}

func (transport *imageRouteLocalOnlyTransport) RoundTrip(request *http.Request) (*http.Response, error) {
	if request.URL.Scheme+"://"+request.URL.Host != transport.baseURL {
		transport.blocked.Add(1)
		return nil, fmt.Errorf("image route test refuses non-fixture URL: %s", request.URL.Redacted())
	}
	return transport.transport.RoundTrip(request)
}

func TestReconcileImageRouteJobExecuteAsyncFailureRefundThenFallbackSuccess(test *testing.T) {
	var postsA, postsB, pollsA, pollsB atomic.Int64
	var providerURL string
	fixture, provider, guard := newImageRouteLocalUpstreamFixture(test, func(fixture *imageRouteReconcileFixture) http.Handler {
		return http.HandlerFunc(func(writer http.ResponseWriter, request *http.Request) {
			writer.Header().Set("Content-Type", "application/json")
			switch request.Method + " " + request.URL.Path {
			case "POST /v1/images/generations":
				var payload map[string]interface{}
				if !assert.NoError(test, common.DecodeJson(request.Body, &payload)) {
					writer.WriteHeader(http.StatusBadRequest)
					return
				}
				assert.Equal(test, "gpt-image-2", payload["model"])
				assert.Equal(test, "local fixture", payload["prompt"])
				assert.Equal(test, "url", payload["response_format"])
				assert.Equal(test, false, payload["stream"])
				switch request.Header.Get("Authorization") {
				case "Bearer local-channel-a":
					postsA.Add(1)
					writer.WriteHeader(http.StatusAccepted)
					_, _ = writer.Write([]byte(`{"task_id":"provider-a","status":"pending","data":[]}`))
				case "Bearer local-channel-b":
					postsB.Add(1)
					var failedReceipt model.TaskBillingReceipt
					assert.NoError(test, fixture.database.Where("channel_id = ?", 703).First(&failedReceipt).Error)
					assert.Equal(test, model.TaskBillingReceiptApplied, failedReceipt.State)
					assert.Zero(test, failedReceipt.AppliedQuota)
					assert.Zero(test, failedReceipt.UsageAppliedQuota)
					assert.Zero(test, failedReceipt.TargetQuota)
					writer.WriteHeader(http.StatusAccepted)
					_, _ = writer.Write([]byte(`{"task_id":"provider-b","status":"pending","data":[]}`))
				default:
					test.Errorf("unexpected local upstream credential")
					writer.WriteHeader(http.StatusUnauthorized)
				}
			case "GET /v1/images/generations/provider-a":
				pollsA.Add(1)
				_, _ = writer.Write([]byte(`{"task_id":"provider-a","status":"failed","message":"local provider A failed"}`))
			case "GET /v1/images/generations/provider-b":
				pollsB.Add(1)
				_, _ = fmt.Fprintf(writer, `{"task_id":"provider-b","status":"completed","created":123,"data":[{"url":%q}]}`, providerURL+"/result.png")
			default:
				test.Errorf("unexpected local upstream request %s %s", request.Method, request.URL.Path)
				writer.WriteHeader(http.StatusNotFound)
			}
		})
	})
	providerURL = provider.URL
	transitions := fixture.observeJobTransitions(test)
	executeImageRouteJob(context.Background(), fixture.job)
	waitingA := fixture.loadJob(test)
	require.Equal(test, model.ImageRouteJobWaiting, waitingA.State, waitingA.LastError)
	require.EqualValues(test, 1, postsA.Load())
	require.Zero(test, postsB.Load())
	childA, found, err := model.GetByTaskId(fixture.job.UserID, waitingA.CurrentTaskID)
	require.NoError(test, err)
	require.True(test, found)
	require.Equal(test, "provider-a", childA.PrivateData.UpstreamTaskID)
	receiptA, err := model.GetTaskBillingReceipt(childA.ID)
	require.NoError(test, err)
	require.Equal(test, model.TaskBillingReceiptTracking, receiptA.State)
	require.Equal(test, 100, receiptA.AppliedQuota)
	assert.Equal(test, 900, fixture.accounting(test).User.Quota)
	reconcileImageRouteJob(context.Background(), waitingA)
	assert.Equal(test, waitingA, fixture.loadJob(test))
	require.NoError(test, service.PollImageTask(context.Background(), childA))
	receiptA, err = model.GetTaskBillingReceipt(childA.ID)
	require.NoError(test, err)
	require.EqualValues(test, model.TaskStatusFailure, childA.Status)
	require.True(test, childA.PrivateData.ImageTask.RetryableFailure)
	require.Equal(test, model.TaskBillingReceiptApplied, receiptA.State)
	require.Zero(test, receiptA.AppliedQuota)
	require.Zero(test, receiptA.UsageAppliedQuota)
	require.Zero(test, receiptA.TargetQuota)
	afterRefund := fixture.accounting(test)
	require.Equal(test, 1000, afterRefund.User.Quota)
	require.Equal(test, 1000, afterRefund.Token.RemainQuota)
	require.Len(test, afterRefund.Reservations, 1)
	assert.Equal(test, model.QuotaReservationStatusSettled, afterRefund.Reservations[0].Status)
	assert.Zero(test, afterRefund.Reservations[0].FinalQuota)
	var refundLogs []model.Log
	require.NoError(test, fixture.database.Where("type = ?", model.LogTypeRefund).Find(&refundLogs).Error)
	require.Len(test, refundLogs, 1)
	assert.Equal(test, 100, refundLogs[0].Quota)
	assert.Equal(test, childA.ChannelId, refundLogs[0].ChannelId)
	assert.Contains(test, refundLogs[0].Other, childA.TaskID)
	reconcileImageRouteJob(context.Background(), waitingA)
	readyB := fixture.loadJob(test)
	require.Equal(test, model.ImageRouteJobReady, readyB.State)
	require.Equal(test, "[703]", readyB.AttemptedChannelsJSON)
	executeImageRouteJob(context.Background(), readyB)
	waitingB := fixture.loadJob(test)
	require.Equal(test, model.ImageRouteJobWaiting, waitingB.State, waitingB.LastError)
	require.NotEqual(test, waitingA.CurrentTaskID, waitingB.CurrentTaskID)
	require.EqualValues(test, 1, postsA.Load())
	require.EqualValues(test, 1, postsB.Load())
	childB, found, err := model.GetByTaskId(fixture.job.UserID, waitingB.CurrentTaskID)
	require.NoError(test, err)
	require.True(test, found)
	require.Equal(test, "provider-b", childB.PrivateData.UpstreamTaskID)
	require.Equal(test, 704, childB.ChannelId)
	require.NoError(test, service.PollImageTask(context.Background(), childB))
	require.EqualValues(test, model.TaskStatusSuccess, childB.Status)
	reconcileImageRouteJob(context.Background(), waitingB)
	completed := fixture.loadJob(test)
	require.Equal(test, model.ImageRouteJobSucceeded, completed.State, completed.LastError)
	root := fixture.loadRoot(test)
	assert.Equal(test, fixture.job.PublicTaskID, root.TaskID)
	assert.EqualValues(test, model.TaskStatusSuccess, root.Status)
	assert.Zero(test, root.Quota)
	assert.Contains(test, string(root.Data), provider.URL+"/result.png")
	afterSuccess := fixture.accounting(test)
	require.Len(test, afterSuccess.Receipts, 2)
	assert.Equal(test, model.TaskBillingReceiptApplied, afterSuccess.Receipts[1].State)
	assert.Equal(test, 200, afterSuccess.Receipts[1].AppliedQuota)
	assert.Equal(test, 800, afterSuccess.User.Quota)
	assert.Equal(test, 200, afterSuccess.User.UsedQuota)
	assert.Equal(test, 2, afterSuccess.User.RequestCount)
	assert.Equal(test, 800, afterSuccess.Token.RemainQuota)
	assert.Equal(test, 200, afterSuccess.Token.UsedQuota)
	require.Len(test, afterSuccess.Reservations, 2)
	assert.NotEqual(test, afterSuccess.Reservations[0].RequestId, afterSuccess.Reservations[1].RequestId)
	assert.Equal(test, 200, afterSuccess.Reservations[1].FinalQuota)
	assert.Zero(test, afterSuccess.Channels[0].UsedQuota)
	assert.EqualValues(test, 200, afterSuccess.Channels[1].UsedQuota)
	assert.Equal(test, "[703,704]", completed.AttemptedChannelsJSON)
	assert.Equal(test, []string{model.ImageRouteJobDispatching, model.ImageRouteJobWaiting, model.ImageRouteJobReady,
		model.ImageRouteJobDispatching, model.ImageRouteJobWaiting, model.ImageRouteJobSucceeded}, *transitions)
	transitionCount := len(*transitions)
	for _, stale := range []*model.ImageRouteJob{fixture.job, waitingA, readyB, waitingB, completed} {
		reconcileImageRouteJob(context.Background(), stale)
		executeImageRouteJob(context.Background(), stale)
	}
	require.NoError(test, service.PollImageTask(context.Background(), childA))
	require.NoError(test, service.PollImageTask(context.Background(), childB))
	assert.Equal(test, completed, fixture.loadJob(test))
	assert.Equal(test, afterSuccess, fixture.accounting(test))
	assert.Len(test, *transitions, transitionCount)
	assert.EqualValues(test, 1, postsA.Load())
	assert.EqualValues(test, 1, postsB.Load())
	assert.EqualValues(test, 1, pollsA.Load())
	assert.EqualValues(test, 1, pollsB.Load())
	assert.Zero(test, guard.blocked.Load())
	response, err := imageRouteJobResponse(completed)
	require.NoError(test, err)
	assert.Contains(test, string(response), provider.URL+"/result.png")
	assert.NotContains(test, string(response), "provider-b")
}

func TestReconcileImageRouteJobSyncDeliverySurvivesPostResponseLookupFailure(test *testing.T) {
	const pngBase64 = "iVBORw0KGgoAAAANSUhEUgAAAAEAAAABCAQAAAC1HAwCAAAAC0lEQVR42mP8/x8AAwMCAO+aWQAAAABJRU5ErkJggg=="
	wireResponse := fmt.Sprintf(`{"created":123,"data":[{"b64_json":%q}]}`, pngBase64)
	var posts, lookupFailures atomic.Int64
	fixture, _, guard := newImageRouteLocalUpstreamFixture(test, func(fixture *imageRouteReconcileFixture) http.Handler {
		return http.HandlerFunc(func(writer http.ResponseWriter, request *http.Request) {
			assert.Equal(test, http.MethodPost, request.Method)
			assert.Equal(test, "/v1/images/generations", request.URL.Path)
			assert.Equal(test, "Bearer local-channel-a", request.Header.Get("Authorization"))
			var beforeSend model.ImageRouteJob
			if assert.NoError(test, fixture.database.First(&beforeSend, fixture.job.ID).Error) {
				assert.Equal(test, model.ImageRouteJobDispatching, beforeSend.State)
				assert.NotEmpty(test, beforeSend.ResultReservation)
				assert.Equal(test, ".json", filepath.Ext(beforeSend.ResultPath))
				assert.NoFileExists(test, beforeSend.ResultPath)
			}
			posts.Add(1)
			writer.Header().Set("Content-Type", "application/json")
			_, _ = writer.Write([]byte(wireResponse))
		})
	})
	var channels []model.Channel
	require.NoError(test, fixture.database.Order("id").Find(&channels).Error)
	for index := range channels {
		channels[index].SetOtherSettings(dto.ChannelOtherSettings{ImageAdapter: &dto.ImageAdapterConfig{Protocol: "generation_references"}})
		require.NoError(test, fixture.database.Save(&channels[index]).Error)
	}
	engine, err := service.GetDefaultSmartRouterEngine()
	require.NoError(test, err)
	engine.Catalogs.Invalidate()
	engine.Prices.InvalidateAll()
	_, err = engine.Catalogs.Refresh(context.Background())
	require.NoError(test, err)
	injected := errors.New("local test: job lookup unavailable after synchronized delivery")
	require.NoError(test, fixture.database.Callback().Query().After("gorm:query").Register("fail_image_route_lookup_after_delivery", func(transaction *gorm.DB) {
		job, ok := transaction.Statement.Dest.(*model.ImageRouteJob)
		if !ok || transaction.Error != nil || job.ID != fixture.job.ID || posts.Load() != 1 || lookupFailures.Load() != 0 {
			return
		}
		if _, err := os.Stat(job.ResultPath); err == nil && lookupFailures.CompareAndSwap(0, 1) {
			transaction.AddError(injected)
		}
	}))
	test.Cleanup(func() {
		require.NoError(test, fixture.database.Callback().Query().Remove("fail_image_route_lookup_after_delivery"))
	})
	transitions := fixture.observeJobTransitions(test)
	executeImageRouteJob(context.Background(), fixture.job)
	require.EqualValues(test, 1, posts.Load())
	require.EqualValues(test, 1, lookupFailures.Load())
	pending := fixture.loadJob(test)
	require.Equal(test, model.ImageRouteJobDispatching, pending.State)
	require.Empty(test, pending.CurrentTaskID)
	require.Equal(test, ".json", filepath.Ext(pending.ResultPath))
	require.FileExists(test, pending.ResultPath)
	assert.NoFileExists(test, pending.ResultPath[:len(pending.ResultPath)-len(".json")])
	raw, err := os.ReadFile(pending.ResultPath)
	require.NoError(test, err)
	assert.JSONEq(test, wireResponse, string(raw))
	assert.EqualValues(test, model.TaskStatusSubmitted, fixture.loadRoot(test).Status)
	var reservation model.MediaStorageReservation
	require.NoError(test, fixture.database.Where("token = ?", pending.ResultReservation).First(&reservation).Error)
	assert.GreaterOrEqual(test, reservation.Bytes, int64(len(raw)))
	afterPost := fixture.accounting(test)
	assert.Equal(test, 900, afterPost.User.Quota)
	assert.Equal(test, 900, afterPost.Token.RemainQuota)
	reconcileImageRouteDeliveries(context.Background())
	completed := fixture.loadJob(test)
	require.Equal(test, model.ImageRouteJobSucceeded, completed.State)
	assert.Equal(test, pending.ResultPath, completed.ResultPath)
	root := fixture.loadRoot(test)
	assert.Equal(test, fixture.job.PublicTaskID, root.TaskID)
	assert.EqualValues(test, model.TaskStatusSuccess, root.Status)
	assert.Zero(test, root.Quota)
	assert.NotContains(test, string(root.Data), pngBase64)
	delivered, err := imageRouteJobResponse(completed)
	require.NoError(test, err)
	assert.JSONEq(test, wireResponse, string(delivered))
	for replay := 0; replay < 3; replay++ {
		reconcileImageRouteDeliveries(context.Background())
		reconcileImageRouteJob(context.Background(), pending)
		executeImageRouteJob(context.Background(), fixture.job)
		executeImageRouteJob(context.Background(), pending)
	}
	assert.Equal(test, completed, fixture.loadJob(test))
	assert.Equal(test, afterPost, fixture.accounting(test))
	assert.Equal(test, []string{model.ImageRouteJobDispatching, model.ImageRouteJobSucceeded}, *transitions)
	assert.EqualValues(test, 1, posts.Load())
	assert.EqualValues(test, 1, lookupFailures.Load())
	assert.Zero(test, guard.blocked.Load())
	assert.FileExists(test, completed.ResultPath)
}
