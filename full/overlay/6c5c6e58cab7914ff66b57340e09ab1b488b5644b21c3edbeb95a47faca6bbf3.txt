func cancellationReservation(t *testing.T) (*gorm.DB, *relaycommon.RelayInfo) {
	t.Helper()
	db, err := gorm.Open(sqlite.Open(":memory:"), &gorm.Config{})
	require.NoError(t, err)
	sqlDB, err := db.DB()
	require.NoError(t, err)
	sqlDB.SetMaxOpenConns(1)
	previous := model.DB
	model.DB = db
	t.Cleanup(func() { model.DB = previous; _ = sqlDB.Close() })
	require.NoError(t, db.AutoMigrate(&model.QuotaReservation{}))
	require.NoError(t, db.Create(&model.QuotaReservation{ReservationId: "cancel-reservation", RequestId: "cancel-request", UserId: 9201, TokenId: 9202, Status: model.QuotaReservationStatusActive}).Error)
	return db, &relaycommon.RelayInfo{
		RequestId: "cancel-request", UserId: 9201, TokenId: 9202, StartTime: time.Now(),
		ChannelMeta:   &relaycommon.ChannelMeta{ChannelId: 99},
		TaskRelayInfo: &relaycommon.TaskRelayInfo{PublicTaskID: "task-cancel"},
		Billing:       &cancellationBilling{},
		PriceData: types.PriceData{Quota: 100, ParameterPricing: &types.ParameterPriceSnapshot{
			SchemaVersion: 1, Revision: "test-v1", RuleID: "test-rule", Group: "default", Model: "cancel-model", ChannelID: 99,
			Unit: "per_request", Currency: "USD", UnitPrice: "0.0002", Quantity: "1", Amount: "0.0002", Quota: 100, Parameters: map[string]string{"count": "1"},
		}},
	}
}

func TestMediaCancellationTaskPreparationAndDispatchFence(t *testing.T) {
	gin.SetMode(gin.TestMode)
	for _, phase := range []string{"before_prepare", "during_body", "during_persist", "live_worker"} {
		t.Run(phase, func(t *testing.T) {
			db, info := cancellationReservation(t)
			execution, cancel := context.WithCancel(context.Background())
			defer cancel()
			c, _ := gin.CreateTestContext(httptest.NewRecorder())
			c.Request = httptest.NewRequest(http.MethodPost, "/v1/videos", nil).WithContext(execution)
			adaptor := &cancellationTaskAdaptor{}
			if phase == "before_prepare" {
				cancel()
			}
			if phase == "during_body" {
				adaptor.build = cancel
			}
			if phase == "during_persist" {
				require.NoError(t, db.Callback().Update().After("gorm:update").Register("cancel_after_snapshot_update", func(tx *gorm.DB) {
					if tx.Statement.Table == "quota_reservations" && tx.Error == nil {
						cancel()
					}
				}))
			}
			preparation := &TaskSubmitPreparation{adaptor: adaptor}
			result, taskErr := RelayTaskSubmitPrepared(c, info, preparation)
			var row model.QuotaReservation
			require.NoError(t, db.First(&row).Error)
			if phase == "live_worker" {
				require.Nil(t, taskErr)
				require.Equal(t, "upstream-local-test", result.UpstreamTaskID)
				require.Equal(t, 1, adaptor.sends)
				return
			}
			require.Nil(t, result)
			require.NotNil(t, taskErr)
			require.True(t, taskErr.LocalError, "cancellation cannot authorize candidate retry")
			require.ErrorIs(t, taskErr.Error, context.Canceled)
			require.Zero(t, adaptor.sends)
			if phase == "before_prepare" {
				require.Zero(t, adaptor.builds)
			}
			if phase != "during_persist" {
				require.Nil(t, row.ParameterCostContext, "canceled preparation must not acquire a send fence")
				return
			}
			require.ErrorIs(t, taskErr.Error, service.ErrSmartMediaSubmissionAmbiguous)
			require.NotNil(t, row.ParameterCostContext)
			var frozen model.ParameterReservationCostContext
			require.NoError(t, common.UnmarshalJsonStr(*row.ParameterCostContext, &frozen))
			require.Equal(t, "dispatching", frozen.SubmissionState)
			require.Equal(t, info.PublicTaskID, frozen.TaskPublicID)
			require.Equal(t, info.RequestId, frozen.RequestID)
			c.Request = c.Request.WithContext(context.Background())
			_, replayErr := RelayTaskSubmitPrepared(c, info, preparation)
			require.NotNil(t, replayErr)
			require.ErrorIs(t, replayErr.Error, model.ErrParameterTaskSubmissionPending)
			require.Zero(t, adaptor.sends, "rebuilding the original request must not reopen dispatch")
		})
	}
}

func TestMediaCancellationImageBeforeAndDuringPersistence(t *testing.T) {
	previousPricing := billing_setting.GetParameterPricingJSONString()
	t.Cleanup(func() {
		_, _, err := billing_setting.UpdateParameterPricingFromJSONString(previousPricing)
		require.NoError(t, err)
	})
	_, _, err := billing_setting.UpdateParameterPricingFromJSONString(`{"schema_version":1,"rules":[{"id":"cancel-rule","group":"default","model":"cancel-model","channel_id":99,"unit":"per_request","unit_price":"0.0002","currency":"USD"}]}`)
	require.NoError(t, err)
	if service.GetHttpClient() == nil {
		service.InitHttpClient()
	}
	client := service.GetHttpClient()
	previousClient := *client
	t.Cleanup(func() { *client = previousClient })
	for _, phase := range []string{"before_prepare", "during_persist"} {
		t.Run(phase, func(t *testing.T) {
			db, info := cancellationReservation(t)
			info.TaskRelayInfo = nil
			info.OriginModelName, info.UsingGroup = "cancel-model", "default"
			info.RelayMode = relayconstant.RelayModeImagesGenerations
			info.Request = &dto.ImageRequest{Model: "cancel-model", Prompt: "local cancellation test"}
			execution, cancel := context.WithCancel(context.Background())
			defer cancel()
			c, _ := gin.CreateTestContext(httptest.NewRecorder())
			c.Request = httptest.NewRequest(http.MethodPost, "/v1/images/generations", nil).WithContext(execution)
			common.SetContextKey(c, constant.ContextKeyChannelType, constant.ChannelTypeOpenAI)
			common.SetContextKey(c, constant.ContextKeyChannelId, 99)
			common.SetContextKey(c, constant.ContextKeyChannelBaseUrl, "https://media-cancellation.invalid")
			common.SetContextKey(c, constant.ContextKeyOriginalModel, "cancel-model")
			info.InitChannelMeta(c)
			quote, err := helper.ResolveRequestParameterPrice(c, info)
			require.NoError(t, err)
			require.NotNil(t, quote)
			info.PriceData.ParameterPricing = quote
			if phase == "before_prepare" {
				cancel()
			} else {
				require.NoError(t, db.Callback().Update().After("gorm:update").Register("cancel_after_image_snapshot", func(tx *gorm.DB) {
					if tx.Statement.Table == "quota_reservations" && tx.Error == nil {
						cancel()
					}
				}))
			}
			sends := 0
			client.Transport = cancellationImageTransport(func(req *http.Request) (*http.Response, error) {
				sends++
				_ = req.Body.Close()
				return &http.Response{StatusCode: http.StatusBadRequest, Header: http.Header{}, Body: io.NopCloser(strings.NewReader(`{"error":{"message":"local test rejection"}}`))}, nil
			})
			apiErr := ImageHelper(c, info)
			require.NotNil(t, apiErr)
			require.ErrorIs(t, apiErr, context.Canceled)
			require.True(t, types.IsSkipRetryError(apiErr))
			require.Zero(t, sends)
			var row model.QuotaReservation
			require.NoError(t, db.First(&row).Error)
			if phase == "before_prepare" {
				require.Nil(t, row.ParameterCostContext)
			} else {
				require.NotNil(t, row.ParameterCostContext)
			}
		})
	}
}
