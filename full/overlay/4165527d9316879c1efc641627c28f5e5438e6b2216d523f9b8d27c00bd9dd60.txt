type smartRouteTestClock struct {
	mu  sync.Mutex
	now time.Time
}

func newSmartRouteTestClock() *smartRouteTestClock {
	return &smartRouteTestClock{now: time.Unix(1_700_000_000, 0)}
}

func (clock *smartRouteTestClock) Now() time.Time {
	clock.mu.Lock()
	defer clock.mu.Unlock()
	return clock.now
}

func (clock *smartRouteTestClock) Advance(duration time.Duration) {
	clock.mu.Lock()
	clock.now = clock.now.Add(duration)
	clock.mu.Unlock()
}

func newTestMemorySmartRouteStateStore(t *testing.T, clock *smartRouteTestClock) *MemorySmartRouteStateStore {
	t.Helper()
	config := DefaultSmartRouteStateConfig()
	config.Clock = clock.Now
	store, err := NewMemorySmartRouteStateStore(config)
	require.NoError(t, err)
	return store
}

func TestSmartRouteRecentMetricsUseBoundedRecencyWeightedRealTraffic(t *testing.T) {
	nowMS := time.Unix(1_700_000_000, 0).UnixMilli()
	metrics := calculateSmartRouteRecentMetrics(
		[]smartRouteTTFTSample{
			{AtMS: nowMS - int64(61*time.Minute/time.Millisecond), MS: 50},
			{AtMS: nowMS - int64(50*time.Minute/time.Millisecond), MS: 100},
			{AtMS: nowMS, MS: 1_000},
			{AtMS: nowMS, MS: 1, Synthetic: true},
		},
		[]smartRouteOutcomeSample{
			{AtMS: nowMS - int64(61*time.Minute/time.Millisecond), Success: false},
			{AtMS: nowMS - int64(50*time.Minute/time.Millisecond), Success: false},
			{AtMS: nowMS, Success: true},
		},
		smartRouteRecentHistory{},
		nowMS,
	)
	assert.Equal(t, 2, metrics.TTFTSamples)
	assert.Equal(t, int64(784), metrics.TTFTRecentAverageMS)
	assert.Equal(t, int64(9_816), metrics.TTFTAverageMS,
		"two real observations contribute exactly 2% before a route history batch exists")
	assert.Equal(t, int64(20_000), metrics.TTFTConfidencePPM)
	assert.Equal(t, 2, metrics.StabilitySamples)
	assert.Equal(t, 1, metrics.StabilitySuccesses)
	assert.Equal(t, 1, metrics.StabilityFailures)
	assert.InDelta(t, 0.98050935, metrics.StabilityRate, 0.000001)
	assert.Equal(t, int64(20_000), metrics.StabilityConfidencePPM)
	assert.Equal(t, nowMS, metrics.LastObservedAtMS)
}

func TestSmartRouteRecentMetricsHistoryMixMatrix(t *testing.T) {
	nowMS := time.Unix(1_700_000_000, 0).UnixMilli()
	history := smartRouteRecentHistory{
		TTFT: smartRouteMetricHistory{
			Version: "route-history-v1", Value: 2_000, P50: 1_800, P95: 3_000,
			EffectiveSamples: 200, TotalSamples: 200, LastEventMS: nowMS - 1,
		},
		Stability: smartRouteMetricHistory{
			Version: "route-history-v1", Value: 0.25,
			EffectiveSamples: 200, TotalSamples: 200, LastEventMS: nowMS - 1,
		},
	}
	for _, sampleCount := range []int{0, 1, 9, 10, 20, 99, 100} {
		t.Run(fmt.Sprintf("samples_%d", sampleCount), func(t *testing.T) {
			ttft := make([]smartRouteTTFTSample, 0, sampleCount)
			outcomes := make([]smartRouteOutcomeSample, 0, sampleCount)
			for index := 0; index < sampleCount; index++ {
				ttft = append(ttft, smartRouteTTFTSample{AtMS: nowMS, MS: 1_000})
				outcomes = append(outcomes, smartRouteOutcomeSample{AtMS: nowMS, Success: true})
			}
			metrics := calculateSmartRouteRecentMetrics(ttft, outcomes, history, nowMS)
			share := float64(sampleCount) / 100
			assert.Equal(t, int64(math.Round(1_000*share+2_000*(1-share))), metrics.TTFTAverageMS)
			assert.InDelta(t, share+0.25*(1-share), metrics.StabilityRate, 1e-12)
			assert.Equal(t, int64(sampleCount)*10_000, metrics.TTFTConfidencePPM)
			assert.Equal(t, int64(sampleCount)*10_000, metrics.StabilityConfidencePPM)
			assert.Equal(t, int64(200), metrics.TTFTHistoricalSamples)
			assert.Equal(t, int64(200), metrics.StabilityHistoricalSamples)
			if sampleCount == 0 {
				assert.Equal(t, "route_history", metrics.TTFTSource)
				assert.Equal(t, "route_history", metrics.StabilitySource)
			} else if sampleCount == 100 {
				assert.Equal(t, "recent", metrics.TTFTSource)
				assert.Equal(t, "recent", metrics.StabilitySource)
			} else {
				assert.Equal(t, "recent_route_history", metrics.TTFTSource)
				assert.Equal(t, "recent_route_history", metrics.StabilitySource)
			}
		})
	}
}

func TestSmartRouteHistoryBatchesTrimTTFTButNeverStabilityFailures(t *testing.T) {
	ttftHistory := smartRouteMetricHistory{}
	stabilityHistory := smartRouteMetricHistory{}
	for index := 0; index < smartRouteLeaderboardHistoryBatchSize; index++ {
		value := 1_000.0
		if index == 0 {
			value = 0
		} else if index == smartRouteLeaderboardHistoryBatchSize-1 {
			value = 100_000
		}
		ttftHistory = updateSmartRouteMetricHistory(ttftHistory, value, int64(1_000+index), false)
		success := 1.0
		if index == 0 {
			success = 0
		}
		stabilityHistory = updateSmartRouteMetricHistory(stabilityHistory, success, int64(1_000+index), true)
	}
	assert.Equal(t, int64(20), ttftHistory.TotalSamples)
	assert.InDelta(t, 1_000, ttftHistory.Value, 1e-9)
	assert.Equal(t, int64(20), stabilityHistory.TotalSamples)
	assert.InDelta(t, 0.95, stabilityHistory.Value, 1e-6,
		"the one confirmed failure remains in the history batch")
}

func TestMemorySmartRouteProbeIsSingleFlightAcrossConcurrentCallers(t *testing.T) {
	clock := newSmartRouteTestClock()
	store := newTestMemorySmartRouteStateStore(t, clock)
	const callers = 100
	start := make(chan struct{})
	results := make(chan *SmartRouteLease, callers)
	errorsSeen := make(chan error, callers)
	var wait sync.WaitGroup
	wait.Add(callers)
	for index := 0; index < callers; index++ {
		go func() {
			defer wait.Done()
			<-start
			lease, acquired, err := store.TryAcquireProbe(context.Background(), SmartRouteProbeRequest{
				RouteID: "cheap-route",
				TTL:     time.Minute,
			})
			if err != nil {
				errorsSeen <- err
				return
			}
			if acquired {
				results <- lease
			}
		}()
	}
	close(start)
	wait.Wait()
	close(results)
	close(errorsSeen)
	require.Empty(t, errorsSeen)
	var leases []*SmartRouteLease
	for lease := range results {
		leases = append(leases, lease)
	}
	require.Len(t, leases, 1)

	snapshot, err := store.Snapshot(context.Background(), SmartRouteStateSnapshotRequest{Routes: []SmartRouteStateRoute{{
		RouteID: "cheap-route", CapacityDomain: "cheap-capacity",
	}}})
	require.NoError(t, err)
	assert.Equal(t, smartrouter.QualityHalfOpen, snapshot.Quality.Routes["cheap-route"].Phase)
}

func TestMemorySmartRouteProbeLeaseIsSingleFlightAcrossRouteAliases(t *testing.T) {
	clock := newSmartRouteTestClock()
	config := DefaultSmartRouteStateConfig()
	config.Clock = clock.Now
	config.ProbeBudgetLimit = 100
	config.ProbeMaxConcurrency = 10
	store, err := NewMemorySmartRouteStateStore(config)
	require.NoError(t, err)

	first, acquired, err := store.TryAcquireProbe(context.Background(), SmartRouteProbeRequest{
		RouteID: "group-cheap", LeaseKey: "channel:7:model:gpt:contract:responses",
	})
	require.NoError(t, err)
	require.True(t, acquired)
	require.NoError(t, store.RecordAttempt(context.Background(), SmartRouteAttemptRecord{
		RouteID: "group-cheap", Outcome: smartrouter.OutcomeSuccess,
		Admission: smartrouter.AdmissionProbe, RouteEpoch: first.Epoch, ProbeLeaseID: first.ID,
	}))
	snapshot, err := store.Snapshot(context.Background(), SmartRouteStateSnapshotRequest{Routes: []SmartRouteStateRoute{
		{RouteID: "group-cheap", CapacityDomain: "cheap-capacity"},
		{RouteID: "group-pro", CapacityDomain: "pro-capacity"},
	}})
	require.NoError(t, err)
	assert.NotEqual(t, smartrouter.QualityUnknown, snapshot.Quality.Routes["group-cheap"].Phase)
	assert.Equal(t, smartrouter.QualityUnknown, snapshot.Quality.Routes["group-pro"].Phase,
		"sharing a physical probe lease must not merge route-local health evidence")

	second, acquired, err := store.TryAcquireProbe(context.Background(), SmartRouteProbeRequest{
		RouteID: "group-pro", LeaseKey: "channel:7:model:gpt:contract:responses",
	})
	require.NoError(t, err)
	assert.False(t, acquired)
	assert.Nil(t, second)

	require.NoError(t, first.Release(context.Background()))
	second, acquired, err = store.TryAcquireProbe(context.Background(), SmartRouteProbeRequest{
		RouteID: "group-pro", LeaseKey: "channel:7:model:gpt:contract:responses",
	})
	require.NoError(t, err)
	require.True(t, acquired)
	require.NoError(t, second.Release(context.Background()))
}

func TestMemoryProbeBudgetsAreIndependentForRecoveryAndColdStart(t *testing.T) {
	config := DefaultSmartRouteStateConfig()
	config.ProbeBudgetLimit = 1
	config.ProbeMaxConcurrency = 1
	store, err := NewMemorySmartRouteStateStore(config)
	require.NoError(t, err)
	acquire := func(routeID, namespace string, budget, concurrency int) (*SmartRouteLease, bool) {
		lease, acquired, acquireErr := store.TryAcquireProbe(context.Background(), SmartRouteProbeRequest{
			RouteID: routeID, SubjectKey: routeID, BudgetNamespace: namespace,
			BudgetLimit: budget, MaxConcurrency: concurrency,
		})
		require.NoError(t, acquireErr)
		return lease, acquired
	}
	recovery, acquired := acquire("recovery-1", SmartRouteProbeBudgetRecovery, 1, 1)
	require.True(t, acquired)
	coldOne, acquired := acquire("cold-1", SmartRouteProbeBudgetColdStart, 60, 2)
	require.True(t, acquired, "cold-start does not wait behind the ordinary recovery namespace")
	_, acquired = acquire("recovery-2", SmartRouteProbeBudgetRecovery, 1, 1)
	assert.False(t, acquired)
	coldTwo, acquired := acquire("cold-2", SmartRouteProbeBudgetColdStart, 60, 2)
	require.True(t, acquired)
	_, acquired = acquire("cold-3", SmartRouteProbeBudgetColdStart, 60, 2)
	assert.False(t, acquired)
	require.NoError(t, recovery.Release(context.Background()))
	require.NoError(t, coldOne.Release(context.Background()))
	require.NoError(t, coldTwo.Release(context.Background()))
}

	}, smartrouter.DefaultSharedOpenFailureThreshold)

		RouteID: routeID, CapacityDomain: "recovering-capacity",
	}}})
	require.NoError(t, err)
	opened := snapshot.Quality.Routes[routeID]

	errorsSeen := make(chan error, callers)

		go func() {
			defer wait.Done()

		RouteEpoch: leases[0].Epoch, ProbeLeaseID: leases[0].ID,
	}))

	snapshot, err = store.Snapshot(context.Background(), SmartRouteStateSnapshotRequest{Routes: []SmartRouteStateRoute{{
		RouteID: routeID, CapacityDomain: "recovering-capacity",
	}}})

	assert.Equal(t, smartrouter.QualityHealthy, recovered.Phase)
	assert.Zero(t, recovered.NextProbeAtMS)
	assert.Zero(t, recovered.SampleDueAtMS)

	assert.Equal(t, updated.Quality.OpenBackoffMS, redisStore.runtimeConfigSnapshot().Quality.OpenBackoffMS)
	assert.Equal(t, updated.MaxProbeFailures, redisStore.runtimeConfigSnapshot().MaxProbeFailures)
	assert.Equal(t, updated.MaxOutage, redisStore.fallback.runtimeConfigSnapshot().MaxOutage)
}
