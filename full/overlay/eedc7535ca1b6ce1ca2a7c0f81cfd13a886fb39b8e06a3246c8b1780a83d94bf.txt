import {
  downloadMediaAssetArchive,
  getAdminMediaAsset,
  getMediaAsset,
} from './api'
import { assetDeliveryUrl } from './lib'
import type { MediaAsset } from './types'

const refreshes = new Map<string, Promise<MediaAsset>>()

export function refreshMediaAsset(asset: MediaAsset, admin = false) {
  if (asset.availability_status === 'unavailable') {
    return Promise.reject(new Error('asset is unavailable'))
  }
  if (asset.id.startsWith('ephemeral-')) return Promise.resolve(asset)
  const key = `${admin ? 'admin' : 'user'}:${asset.id}`
  const active = refreshes.get(key)
  if (active) return active
  const refresh = (
    admin ? getAdminMediaAsset(asset.id) : getMediaAsset(asset.id)
  ).finally(() => refreshes.delete(key))
  refreshes.set(key, refresh)
  return refresh
}

export async function freshAssetDeliveryUrl(asset: MediaAsset, admin = false) {
  const current = assetDeliveryUrl(asset)
  if (
    !current ||
    current.startsWith('blob:') ||
    current.startsWith('data:') ||
    asset.id.startsWith('ephemeral-')
  ) {
    return current
  }
  const refreshed = await refreshMediaAsset(asset, admin)
  return assetDeliveryUrl(refreshed)
}

export async function openMediaAsset(asset: MediaAsset, admin = false) {
  const popup = window.open('about:blank', '_blank')
  if (popup) popup.opener = null
  try {
    const url = await freshAssetDeliveryUrl(asset, admin)
    if (!url) throw new Error('asset delivery URL is unavailable')
    if (popup) {
      popup.location.replace(url)
      return
    }
    const link = document.createElement('a')
    link.href = url
    link.rel = 'noreferrer'
    link.target = '_blank'
    link.click()
  } catch (error) {
    popup?.close()
    throw error
  }
}

export async function downloadMediaAsset(asset: MediaAsset, admin = false) {
  const url = await freshAssetDeliveryUrl(asset, admin)
  if (!url) throw new Error('asset delivery URL is unavailable')
  const link = document.createElement('a')
  link.download = asset.filename || 'media-asset'
  link.href = url
  link.rel = 'noreferrer'
  document.body.append(link)
  link.click()
  link.remove()
}

export async function downloadMediaAssets(assets: MediaAsset[], admin = false) {
  if (admin) {
    throw new Error('administrator archive download is not available')
  }
  const archive = await downloadMediaAssetArchive(
    assets.map((asset) => asset.id)
  )
  const url = URL.createObjectURL(archive.blob)
  try {
    const link = document.createElement('a')
    link.download = archive.filename
    link.href = url
    document.body.append(link)
    link.click()
    link.remove()
  } finally {
    if (typeof window !== 'undefined') {
      window.setTimeout(() => URL.revokeObjectURL(url), 0)
    } else {
      URL.revokeObjectURL(url)
    }
  }
  return archive
}
