export async function getAllSmartRouterHealth() {
  const first = await getSmartRouterHealthPage({ page: 1, pageSize: 100 })
  if (!first.success) {
    throw new Error(first.message || 'Unable to load smart-routing health')
  }
  if (!first.data || Array.isArray(first.data)) return first

  const total = first.data.total ?? first.data.items.length
  const pageCount = Math.min(50, Math.ceil(total / 100))
  if (pageCount <= 1) return first

  const remaining = []
  for (let page = 2; page <= pageCount; page += 4) {
    const batch = await Promise.all(
      Array.from({ length: Math.min(4, pageCount - page + 1) }, (_, index) =>
        getSmartRouterHealthPage({ page: page + index, pageSize: 100 })
      )
    )
    remaining.push(...batch)
  }
  const failedPage = remaining.find((response) => !response.success)
  if (failedPage) {
    throw new Error(
      failedPage.message || 'Unable to load a smart-routing health page'
    )
  }

  const items = [
    ...first.data.items,
    ...remaining.flatMap((response) => {
      if (!response.success || !response.data) return []
      return Array.isArray(response.data) ? response.data : response.data.items
    }),
  ]
  return {
    ...first,
    data: {
      ...first.data,
      items,
      page: 1,
      page_size: items.length,
    },
  }
}

/**
 * Fetch the privacy-filtered route view for one of the current user's keys.
 * The endpoint is intentionally separate from the Root-only health console;
 * callers must never infer channel topology from this response.
 */
export async function getSmartRouterStatus(params: {
  tokenId?: number
  model?: string
  endpoint?: string
}) {
  const search = new URLSearchParams()
  if (params.tokenId !== undefined) {
    search.set('token_id', String(params.tokenId))
  }
  if (params.model) {
    search.set('model', params.model)
  }
  if (params.endpoint) {
    search.set('endpoint', params.endpoint)
  }
  const query = search.toString()
  const res = await api.get<SmartRouterStatusResponse>(
    `/api/smart-router/status${query ? `?${query}` : ''}`
  )
  return res.data
}

export async function getSmartRouterStatusModels(params: {

  const res = await api.get<SmartRouterStatusModelsResponse>(

        page_size: params.pageSize ?? 100,
      },
    }
  )
  return res.data

// The backend can return the complete privacy-safe index in one bounded
// snapshot. Keep this path single-request so a slow page can never leave the
// status screen waiting on a fan-out of 20 overlapping Redis/catalog reads.
export async function getAllSmartRouterStatusModels(tokenId: number) {
  const res = await getSmartRouterStatusModels({
    tokenId,
    page: 1,
    pageSize: 1000,
  })
  if (!res.success) {
    throw new Error(res.message || 'Unable to load smart-routing model index')
  }
  return res
}
