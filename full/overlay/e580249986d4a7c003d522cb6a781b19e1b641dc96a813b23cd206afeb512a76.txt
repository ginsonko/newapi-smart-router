package middleware

import (
	"errors"
	"net/http"
	"net/http/httptest"
	"strings"
	"testing"

	"github.com/QuantumNous/new-api/types"
	"github.com/gin-gonic/gin"
	"github.com/stretchr/testify/assert"
	"github.com/stretchr/testify/require"
)

func TestSystemPerformanceRejectionProtocolAndNoDownstreamWork(t *testing.T) {
	gin.SetMode(gin.TestMode)
	for _, path := range []string{"/v1/responses", "/v1/messages"} {
		t.Run(path, func(t *testing.T) {
			r := gin.New()
			called := false
			r.Use(systemPerformanceCheck(func() *types.NewAPIError {
				return types.NewErrorWithStatusCode(errors.New("服务器内存负载较高，请5秒后重试"), "system_memory_overloaded", 503)
			}))
			r.POST(path, func(c *gin.Context) { called = true; c.Status(200) })
			w := httptest.NewRecorder()
			r.ServeHTTP(w, httptest.NewRequest(http.MethodPost, path, strings.NewReader("{}")))
			require.Equal(t, 503, w.Code)
			assert.Equal(t, "5", w.Header().Get("Retry-After"))
			assert.Contains(t, w.Body.String(), "服务器内存负载较高")
			assert.Contains(t, w.Body.String(), "error")
			assert.False(t, called, "拒绝必须发生在业务/扣费之前")
		})
	}
}

func TestSystemPerformanceHealthyRequestStillRequiresAuthentication(t *testing.T) {
	gin.SetMode(gin.TestMode)
	r := gin.New()
	r.Use(systemPerformanceCheck(func() *types.NewAPIError { return nil }))
	r.POST("/v1/responses", func(c *gin.Context) { c.AbortWithStatus(http.StatusUnauthorized) })
	w := httptest.NewRecorder()
	r.ServeHTTP(w, httptest.NewRequest(http.MethodPost, "/v1/responses", nil))
	assert.Equal(t, http.StatusUnauthorized, w.Code)
	assert.Empty(t, w.Header().Get("Retry-After"))
}
