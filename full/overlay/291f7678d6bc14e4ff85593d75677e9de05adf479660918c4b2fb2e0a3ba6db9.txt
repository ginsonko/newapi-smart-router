func TestSmartRouteAttemptCommittedIgnoresOnlyCurrentAttemptKeepalive(t *testing.T) {
	recorder := httptest.NewRecorder()
	c, _ := gin.CreateTestContext(recorder)
	c.Request = httptest.NewRequest(http.MethodPost, "/v1/chat/completions", nil)

	writtenBefore := c.Writer.Written()
	sizeBefore := c.Writer.Size()
	require.NoError(t, relayhelper.PingData(c))
	assert.False(t, smartRouteAttemptCommitted(c, writtenBefore, sizeBefore), "SSE comments alone remain replaceable")

	// A new attempt begins after the prior keepalive. Gin's Writer remains in a
	// written state, so the decision must use only this attempt's byte delta.
	common.SetContextKey(c, constant.ContextKeySmartStreamPingBytes, 0)
	writtenBefore = c.Writer.Written()
	sizeBefore = c.Writer.Size()
	assert.False(t, smartRouteAttemptCommitted(c, writtenBefore, sizeBefore), "a prior route's keepalive must not leak into the next attempt")

	require.NoError(t, relayhelper.ObjectData(c, gin.H{"id": "payload"}))
	assert.True(t, smartRouteAttemptCommitted(c, writtenBefore, sizeBefore), "any protocol payload is committed")
}

func TestSmartRouteAttemptCommittedTreatsHeaderOnlyFlushAsCommitted(t *testing.T) {
	recorder := httptest.NewRecorder()
	c, _ := gin.CreateTestContext(recorder)

	assert.True(t, smartRouteAttemptCommitted(c, writtenBefore, sizeBefore))
}

func TestResetSmartChannelContextClearsPriorRouteIdentity(t *testing.T) {
	recorder := httptest.NewRecorder()
	c, _ := gin.CreateTestContext(recorder)
	c.Request = httptest.NewRequest(http.MethodPost, "/v1/responses", nil)
	common.SetContextKey(c, constant.ContextKeyChannelId, 99)
	common.SetContextKey(c, constant.ContextKeyChannelName, "api-b")

	common.SetContextKey(c, constant.ContextKeyChannelBaseUrl, "https://internal.example")
	common.SetContextKey(c, constant.ContextKeyChannelIsMultiKey, true)
	common.SetContextKey(c, constant.ContextKeyChannelMultiKeyIndex, 3)

func TestRelaySmartFaultInjection(t *testing.T) {

	t.Run("one transient failure retries the same physical route and never marks it red", func(t *testing.T) {
		h := newSmartRelayTestHarness(t, 1_000)
		ctx, _, routeRequest := h.newRequest(t, "same-route-confirmation-success")
		var channels []int

		resultErr := h.relay(ctx, routeRequest, func(c *gin.Context, _ types.RelayFormat, info *relaycommon.RelayInfo) *types.NewAPIError {
			channels = append(channels, common.GetContextKeyInt(c, constant.ContextKeyChannelId))
			if len(channels) == 1 {
				return smartRelayTestError(http.StatusServiceUnavailable, types.ErrorCodeBadResponseStatusCode, 0)
			}
			require.NoError(t, info.Billing.Settle(info.PriceData.QuotaToPreConsume))
			return nil
		})

		require.Nil(t, resultErr)
		require.Equal(t, []int{smartRelayTestCheapChannel, smartRelayTestCheapChannel}, channels)
		reservation := h.reservation(t, "same-route-confirmation-success")
		assert.Equal(t, model.QuotaReservationStatusSettled, reservation.Status)
		assert.Equal(t, 2, reservation.AttemptCount)

		contractID, _, err := service.BuildSmartContractIdentity(smartRelayTestModel, service.SmartEndpointChat)
		require.NoError(t, err)
		route := catalogRouteByContractGroup(t, h.catalog(t), contractID, "cheap")
		snapshot, err := h.state.Snapshot(context.Background(), service.SmartRouteStateSnapshotRequest{
			Routes: []service.SmartRouteStateRoute{{
				RouteID: route.RouteID, CapacityDomain: route.CapacityDomain,
				CredentialDomain: smartrouter.RouteCredentialDomain(route),
			}},
		})
		require.NoError(t, err)
		quality := snapshot.Quality.Routes[route.RouteID]
		assert.Equal(t, smartrouter.QualityHealthy, quality.Phase)
		assert.Zero(t, quality.RealHardFailures)
		assert.Zero(t, quality.ConsecutiveHardFailures)
	})

	t.Run("generic upstream 400 keeps evidence and fails over before semantic commit", func(t *testing.T) {
		h := newSmartRelayTestHarness(t, 1_000)
		ctx, _, routeRequest := h.newRequest(t, "provider-400-failover")
		var channels []int
		resultErr := h.relay(ctx, routeRequest, func(c *gin.Context, _ types.RelayFormat, info *relaycommon.RelayInfo) *types.NewAPIError {
			channelID := common.GetContextKeyInt(c, constant.ContextKeyChannelId)
			channels = append(channels, channelID)
			if channelID == smartRelayTestCheapChannel {
				return types.WithOpenAIError(types.OpenAIError{
					Message: "provider temporarily rejected this request", Type: "provider_error", Code: "vendor_busy",
				}, http.StatusBadRequest, types.ErrOptionWithUpstreamResponseReceived())
			}
			require.NoError(t, info.Billing.Settle(info.PriceData.QuotaToPreConsume))
			return nil
		})
		require.Nil(t, resultErr)
		require.Equal(t, []int{smartRelayTestCheapChannel, smartRelayTestCheapChannel, smartRelayTestPlusChannel}, channels)
		assert.NotContains(t, ctx.Errors.String(), "具体字段或参数")
		reservation := h.reservation(t, "provider-400-failover")
		assert.Equal(t, model.QuotaReservationStatusSettled, reservation.Status)
		assert.Equal(t, 3, reservation.AttemptCount)
	})

	for _, disconnected := range []bool{false, true} {
		t.Run(fmt.Sprintf("upstream cancellation downstream disconnected %v", disconnected), func(t *testing.T) {
			h := newSmartRelayTestHarness(t, 1_000)
			ctx, _, routeRequest := h.newRequest(t, fmt.Sprintf("cancel-live-%v", disconnected))
			original, cancel := context.WithCancel(ctx.Request.Context())
			defer cancel()
			ctx.Request = ctx.Request.WithContext(original)
			var channels []int
			resultErr := h.relay(ctx, routeRequest, func(c *gin.Context, _ types.RelayFormat, info *relaycommon.RelayInfo) *types.NewAPIError {
				id := common.GetContextKeyInt(c, constant.ContextKeyChannelId)
				channels = append(channels, id)
				if disconnected {
					cancel()
				}
				if id == smartRelayTestCheapChannel {
					return types.NewErrorWithStatusCode(fmt.Errorf("upstream: %w", context.Canceled), types.ErrorCodeDoRequestFailed, http.StatusBadGateway)
				}
				require.NoError(t, info.Billing.Settle(info.PriceData.QuotaToPreConsume))
				return nil
			})
			if disconnected {
				require.NotNil(t, resultErr)
				require.Equal(t, []int{smartRelayTestCheapChannel}, channels)
			} else {
				require.Nil(t, resultErr)
				require.Equal(t, []int{smartRelayTestCheapChannel, smartRelayTestCheapChannel, smartRelayTestPlusChannel}, channels)
			}
		})
	}

	t.Run("cheap 503 falls back to plus and settles the final group quote", func(t *testing.T) {
		h := newSmartRelayTestHarness(t, 1_000)
		ctx, recorder, routeRequest := h.newRequest(t, "cheap-503-plus-success")
		var channels []int
		var ratios []float64

		dispatch := func(c *gin.Context, _ types.RelayFormat, info *relaycommon.RelayInfo) *types.NewAPIError {
			channelID := common.GetContextKeyInt(c, constant.ContextKeyChannelId)
			channels = append(channels, channelID)
			ratios = append(ratios, info.PriceData.GroupRatioInfo.GroupRatio)
			if channelID == smartRelayTestCheapChannel {
				return smartRelayTestError(http.StatusServiceUnavailable, types.ErrorCodeBadResponseStatusCode, 0)
			}
			if err := info.Billing.Settle(info.PriceData.QuotaToPreConsume); err != nil {
				return types.NewError(err, types.ErrorCodeUpdateDataError)
			}
			return nil
		}

		resultErr := h.relay(ctx, routeRequest, dispatch)
		require.Nil(t, resultErr)
		assert.Equal(t, []int{smartRelayTestCheapChannel, smartRelayTestCheapChannel, smartRelayTestPlusChannel}, channels)
		require.Len(t, ratios, 3)
		assert.InDelta(t, 0.02, ratios[0], 0.000001)
		assert.InDelta(t, 0.02, ratios[1], 0.000001)
		assert.InDelta(t, 0.05, ratios[2], 0.000001)
		assert.False(t, recorder.Flushed)

		reservation := h.reservation(t, "cheap-503-plus-success")
		assert.Equal(t, model.QuotaReservationStatusSettled, reservation.Status)
		assert.False(t, reservation.SettlementPending)
		assert.Equal(t, smartRelayTestPlusChannel, reservation.ChannelId)
		assert.Equal(t, "plus", reservation.Group)
		assert.Equal(t, int64(50_000), reservation.RatioPPM)
		assert.Equal(t, reservation.QuotedQuota, reservation.FinalQuota)
		assert.Equal(t, 3, reservation.AttemptCount)
		assert.Greater(t, reservation.FinalQuota, 0)
		assert.Equal(t, h.quota-reservation.FinalQuota, h.userQuota(t))
	})

		require.Len(t, selectedKeys, 2)

		assert.Equal(t, "cheap-valid-key", selectedKeys[1])

		require.Eventually(t, func() bool {
			if err := h.db.First(&updated, smartRelayTestCheapChannel).Error; err != nil {
				return false

		nextKey, nextIndex, keyErr := updated.GetNextEnabledKey()
		require.Nil(t, keyErr)
		assert.Equal(t, "cheap-valid-key", nextKey)
		assert.Equal(t, 1, nextIndex)
	})

	settingProvider := func() smartroutersetting.Setting { return setting }
	catalogs := service.NewCatalogManager(service.NewGormCatalogSource(db), settingProvider)
	prices := service.NewPriceSnapshotManager(catalogs, nil)
	state, err := service.NewMemorySmartRouteStateStore(service.DefaultSmartRouteStateConfig())
	require.NoError(t, err)
	engine, err := service.NewSmartRouterEngine(
		catalogs, prices, state, service.NewMemorySmartRouteAffinityStore(nil, 100), settingProvider,
	)
	require.NoError(t, err)
	catalog, err := catalogs.Refresh(context.Background())
	require.NoError(t, err)
	contractID, _, err := service.BuildSmartContractIdentity(smartRelayTestModel, service.SmartEndpointChat)
	require.NoError(t, err)
	contract, exists := catalog.Contracts[contractID]
	require.True(t, exists)
	require.NotEmpty(t, contract.Pools["balanced-v1"].Candidates)
	priceSnapshot, err := prices.Refresh(context.Background(), "default")
	require.NoError(t, err)
	for _, route := range contract.Pools["balanced-v1"].Candidates {
		_, priced := priceSnapshot.RatiosPPM[route.RouteID]
		require.True(t, priced)
		if route.Group != "plus" {
			continue
		}
		// These controller cases exercise failover after the shared router has
		// already observed a usable alternative. V4 intentionally has no
		// administrator-declared stable fallback, so the fixture must establish
		// that runtime evidence explicitly instead of relying on group metadata.
		require.NoError(t, state.RecordAttempt(context.Background(), service.SmartRouteAttemptRecord{
			RouteID: route.RouteID, Group: route.Group, Model: smartRelayTestModel,
			ChannelID: route.ChannelID, ContractID: contract.Contract.ContractID,
			Endpoint: contract.Contract.Endpoint, CapacityDomain: route.CapacityDomain,
			CredentialDomain: smartrouter.RouteCredentialDomain(route),
			Outcome:          smartrouter.OutcomeSuccess, Admission: smartrouter.AdmissionNormal,
		}))
	}

	t.Cleanup(func() {
		prices.Close()
		catalogs.Close()
		sqlDB, sqlErr := db.DB()
		if sqlErr == nil {
			_ = sqlDB.Close()
		}
	})
	return &smartRelayTestHarness{db: db, engine: engine, state: state, setting: setting, quota: quota}
}

func catalogRouteByContractGroup(
	t *testing.T,
	catalog smartrouter.CertifiedRouteCatalog,
	contractID string,
	group string,
) smartrouter.CertifiedRoute {
	t.Helper()
	contract, exists := catalog.Contracts[contractID]
	require.Truef(t, exists, "catalog has no contract %q", contractID)
	for _, pool := range contract.Pools {
		for _, route := range pool.Candidates {
			if route.Group == group {
				return route
			}
		}
	}
	require.Failf(t, "missing route", "catalog contract %q has no route for group %q", contractID, group)
	return smartrouter.CertifiedRoute{}
}

func catalogStateRoutes(catalog smartrouter.CertifiedRouteCatalog) []service.SmartRouteStateRoute {
	seen := make(map[string]struct{})
	var routes []service.SmartRouteStateRoute
	for _, contract := range catalog.Contracts {
		for _, pool := range contract.Pools {
			for _, route := range pool.Candidates {
				if _, exists := seen[route.RouteID]; exists {
					continue
				}
				seen[route.RouteID] = struct{}{}
				routes = append(routes, service.SmartRouteStateRoute{
					RouteID: route.RouteID, CapacityDomain: route.CapacityDomain,
					StableFallback: route.StableFallback, MaxInflight: route.MaxInflight,
				})
			}
		}
	}
	return routes
}
