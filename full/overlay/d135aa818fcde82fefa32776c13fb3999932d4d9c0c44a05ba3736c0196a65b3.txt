func TestSmartRouteStreamGuard(t *testing.T) {
	gin.SetMode(gin.TestMode)

	newContext := func(smart bool) *gin.Context {
		ctx, _ := gin.CreateTestContext(httptest.NewRecorder())
		ctx.Request = httptest.NewRequest("POST", "/v1/chat/completions", nil)
		if smart {
			common.SetContextKey(ctx, constant.ContextKeyTokenRoutingPolicy, true)
		}
		return ctx
	}

	t.Run("normal completed stream passes", func(t *testing.T) {
		ctx := newContext(true)
		status := relaycommon.NewStreamStatus()
		status.SetEndReason(relaycommon.StreamEndReasonDone, nil)
		assert.Nil(t, smartRouteStreamGuard(ctx, &relaycommon.RelayInfo{IsStream: true, StreamStatus: status}))
	})

	t.Run("provider terminal followed by EOF passes", func(t *testing.T) {
		ctx := newContext(true)
		status := relaycommon.NewStreamStatus()
		status.MarkTerminal()
		status.SetEndReason(relaycommon.StreamEndReasonEOF, nil)
		assert.Nil(t, smartRouteStreamGuard(ctx, &relaycommon.RelayInfo{IsStream: true, StreamStatus: status}))
	})

	t.Run("provider terminal followed by client disconnect passes", func(t *testing.T) {
		ctx := newContext(true)
		status := relaycommon.NewStreamStatus()
		status.SetEndReason(relaycommon.StreamEndReasonClientGone, context.Canceled)
		status.MarkTerminalKind("chat_finish_reason")
		assert.Nil(t, smartRouteStreamGuard(ctx, &relaycommon.RelayInfo{IsStream: true, StreamStatus: status}))
	})

	t.Run("bare EOF is surfaced", func(t *testing.T) {
		ctx := newContext(true)
		status := relaycommon.NewStreamStatus()
		status.SetEndReason(relaycommon.StreamEndReasonEOF, nil)
		err := smartRouteStreamGuard(ctx, &relaycommon.RelayInfo{IsStream: true, StreamStatus: status})
		require.NotNil(t, err)
		assert.Equal(t, 502, err.StatusCode)
	})

	t.Run("scanner failure becomes retryable relay error", func(t *testing.T) {
		ctx := newContext(true)
		status := relaycommon.NewStreamStatus()
		status.SetEndReason(relaycommon.StreamEndReasonScannerErr, errors.New("connection reset"))
		err := smartRouteStreamGuard(ctx, &relaycommon.RelayInfo{IsStream: true, StreamStatus: status})
		require.NotNil(t, err)
		assert.Equal(t, 502, err.StatusCode)
		assert.Contains(t, err.Error(), "connection reset")
	})

	t.Run("soft stream errors are surfaced before billing", func(t *testing.T) {
		ctx := newContext(true)
		status := relaycommon.NewStreamStatus()
		status.RecordError("invalid event")
		status.SetEndReason(relaycommon.StreamEndReasonHandlerStop, nil)
		require.NotNil(t, smartRouteStreamGuard(ctx, &relaycommon.RelayInfo{IsStream: true, StreamStatus: status}))
	})

	t.Run("fixed key keeps legacy stream behavior", func(t *testing.T) {
		ctx := newContext(false)
		status := relaycommon.NewStreamStatus()
		status.SetEndReason(relaycommon.StreamEndReasonScannerErr, errors.New("connection reset"))
		assert.Nil(t, smartRouteStreamGuard(ctx, &relaycommon.RelayInfo{IsStream: true, StreamStatus: status}))
	})
}

func TestSmartResponsesDoneCompletedThenEOFPassesOuterGuard(t *testing.T) {
	gin.SetMode(gin.TestMode)
	recorder := httptest.NewRecorder()
	ctx, _ := gin.CreateTestContext(recorder)
	ctx.Request = httptest.NewRequest(http.MethodPost, "/v1/responses", nil)
	common.SetContextKey(ctx, constant.ContextKeyTokenRoutingPolicy, true)
	body := strings.Join([]string{
		`data: {"type":"response.created","response":{"id":"resp_done","status":"in_progress"}}`,
		`data: {"type":"response.output_text.delta","delta":"complete"}`,
		`data: {"type":"response.done","response":{"status":"completed","usage":{"input_tokens":1,"output_tokens":1,"total_tokens":2}}}`,
	}, "\n\n")
	info := &relaycommon.RelayInfo{
		ChannelMeta: &relaycommon.ChannelMeta{UpstreamModelName: "gpt-test"},
		IsStream:    true,
		RelayFormat: types.RelayFormatOpenAIResponses,
	}
	restore := openaichannel.BeginSmartResponsesSemanticWindow(ctx, info)
	defer restore()
	resp := &http.Response{
		StatusCode: http.StatusOK,
		Header:     http.Header{"Content-Type": []string{"text/event-stream"}},
		Body:       io.NopCloser(strings.NewReader(body)),
	}

	usage, apiErr := openaichannel.OaiResponsesStreamHandler(ctx, info, resp)
	require.Nil(t, apiErr)
	require.NotNil(t, usage)
	require.Equal(t, 2, usage.TotalTokens)
	require.NotNil(t, info.StreamStatus)
	require.True(t, info.StreamStatus.HasTerminal(), "response.done with an explicit completed status must be provider-complete")
	require.Nil(t, smartRouteStreamGuard(ctx, info))
}

func TestSmartResponsesDoneWithoutCompletedStatusDoesNotPassOuterGuard(t *testing.T) {
	gin.SetMode(gin.TestMode)
	for _, status := range []string{"incomplete", "failed", "cancelled", "canceled", "unknown"} {
		t.Run(status, func(t *testing.T) {
			recorder := httptest.NewRecorder()
			ctx, _ := gin.CreateTestContext(recorder)
			ctx.Request = httptest.NewRequest(http.MethodPost, "/v1/responses", nil)
			common.SetContextKey(ctx, constant.ContextKeyTokenRoutingPolicy, true)
			body := `data: {"type":"response.done","response":{"status":"` + status + `"}}`
			info := &relaycommon.RelayInfo{
				ChannelMeta: &relaycommon.ChannelMeta{UpstreamModelName: "gpt-test"},
				IsStream:    true,
				RelayFormat: types.RelayFormatOpenAIResponses,
			}
			restore := openaichannel.BeginSmartResponsesSemanticWindow(ctx, info)
			defer restore()
			resp := &http.Response{
				StatusCode: http.StatusOK,
				Header:     http.Header{"Content-Type": []string{"text/event-stream"}},
				Body:       io.NopCloser(strings.NewReader(body)),
			}

			_, handlerErr := openaichannel.OaiResponsesStreamHandler(ctx, info, resp)
			if handlerErr != nil {
				if info.StreamStatus != nil {
					assert.False(t, info.StreamStatus.HasTerminal())
				}
				return
			}
			require.NotNil(t, info.StreamStatus)
			assert.False(t, info.StreamStatus.HasTerminal())
			require.NotNil(t, smartRouteStreamGuard(ctx, info))
		})
	}
}

func TestEmitSmartStreamTerminationUsesClientProtocol(t *testing.T) {
	gin.SetMode(gin.TestMode)
	for _, test := range []struct {
		name   string
		format types.RelayFormat
		want   string
	}{
		{name: "openai chat", format: types.RelayFormatOpenAI, want: `"error"`},
		{name: "responses", format: types.RelayFormatOpenAIResponses, want: "event: error"},
		{name: "claude", format: types.RelayFormatClaude, want: "event: error"},
	} {
		t.Run(test.name, func(t *testing.T) {
			recorder := httptest.NewRecorder()
			ctx, _ := gin.CreateTestContext(recorder)
			ctx.Request = httptest.NewRequest("POST", "/v1/responses", nil)
			common.SetContextKey(ctx, constant.ContextKeyTokenRoutingPolicy, true)
			ctx.Set(common.RequestIdKey, "termination-test")
			info := &relaycommon.RelayInfo{IsStream: true, RelayFormat: test.format}
			err := types.NewOpenAIError(errors.New("upstream EOF"), types.ErrorCodeBadResponse, 502)

			EmitSmartStreamTermination(ctx, info, err)
			first := recorder.Body.String()
			EmitSmartStreamTermination(ctx, info, err)
			assert.Equal(t, first, recorder.Body.String(), "termination must be emitted once")
			assert.Contains(t, first, test.want)
			assert.Contains(t, first, "termination-test")
		})
	}
}
