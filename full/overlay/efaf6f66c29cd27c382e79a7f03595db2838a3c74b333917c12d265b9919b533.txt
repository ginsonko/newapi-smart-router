export type BootLifecycleController = {
  ready: () => void
  fail: (error?: unknown) => void
}

declare global {
  interface Window {
    __NEWAPI_EARLY_BOOT__?: {
      ready: () => void
      hasFailure?: () => boolean
    }
    __NEWAPI_BOOT__?: BootLifecycleController
  }
}

function getBootController(): BootLifecycleController | undefined {
  if (typeof window === 'undefined') return undefined
  return window.__NEWAPI_BOOT__
}

export function markApplicationReady(
  controller: BootLifecycleController | undefined = getBootController()
): void {
  if (controller) {
    controller.ready()
    return
  }

  if (typeof document !== 'undefined') {
    document.querySelector('#boot-shell')?.remove()
  }
}

export function reportApplicationBootFailure(
  error: unknown,
  controller: BootLifecycleController | undefined = getBootController()
): void {
  if (controller) {
    controller.fail(error)
    return
  }

  // eslint-disable-next-line no-console
  console.error('Application failed to start', error)
}
