var buildFS embed.FS

//go:embed web/default/dist/index.html

var classicBuildFS embed.FS

func main() {
	startTime := time.Now()

	err := InitResources()
	if err != nil {
		common.FatalLog("failed to initialize resources: " + err.Error())
		return
	}
	processRole, roleErr := smartRouterProcessRole()
	if roleErr != nil {
		common.FatalLog(roleErr.Error())
		return
	}
	workerOnly := processRole == "smart-router-worker"
	analyticsWorkerOnly := processRole == "route-analytics-worker"
	commissionWorkerOnly := processRole == "commission-worker"
	backgroundOnly := workerOnly || analyticsWorkerOnly

	common.SysLog("New API " + common.Version + " started")
	if os.Getenv("GIN_MODE") != "debug" {
		gin.SetMode(gin.ReleaseMode)
	}
	if common.DebugEnabled {
		common.SysLog("running in debug mode")
	}

	defer func() {
		err := model.CloseDB()
		if err != nil {
			common.FatalLog("failed to close database: " + err.Error())
		}
	}()
	if migrationOnlyRequested() {
		common.SysLog("database migration completed; MIGRATION_ONLY requested, exiting before background tasks and HTTP startup")
		return
	}
	// Downstream cost publishing is an explicitly configured asynchronous
	// worker. Missing/broken sink configuration never prevents the core API
	// from starting; events remain durable in the local outbox for pull/retry.
	var stopTenantCostOutbox func()
	var stopTenantCostProjectionRetry func()
	if !backgroundOnly && !commissionWorkerOnly {
		var outboxErr error
		stopTenantCostOutbox, outboxErr = service.StartTenantCostOutboxWorkerFromEnv()
		if outboxErr != nil {
			common.SysError("tenant cost outbox worker disabled: " + outboxErr.Error())
			stopTenantCostOutbox = nil
		}
		var projectionErr error
		stopTenantCostProjectionRetry, projectionErr = service.StartTenantCostProjectionRetryWorkerFromEnv()
		if projectionErr != nil {
			common.SysError("tenant cost projection retry worker disabled: " + projectionErr.Error())
			stopTenantCostProjectionRetry = nil
		}
	}
	if stopTenantCostOutbox != nil {
		defer stopTenantCostOutbox()
	}
	if stopTenantCostProjectionRetry != nil {
		defer stopTenantCostProjectionRetry()
	}
	if commissionWorkerOnly {
		worker, workerErr := service.StartCommissionSidecarWorker()
		if workerErr != nil {
			common.FatalLog("failed to initialize commission sidecar worker: " + workerErr.Error())
			return
		}
		waitForCommissionSidecarWorker(worker)
		return
	}
	var stopMediaCleanup func()
	var stopMediaMaterializer func()
	if !backgroundOnly {
		if mediaErr := mediaservice.InitializeStorage(); mediaErr == nil {
			stopMediaCleanup = mediaservice.StartCleanupLoop()
			stopMediaMaterializer = service.StartMediaMaterializer()
		} else if !errors.Is(mediaErr, mediaservice.ErrMediaDisabled) {
			common.SysError("media storage initialization failed; media APIs remain unavailable: " + mediaErr.Error())
		}
	}
	if stopMediaMaterializer != nil {
		defer stopMediaMaterializer()
	}
	if stopMediaCleanup != nil {
		defer stopMediaCleanup()
	}

	if common.RedisEnabled {
		// for compatibility with old versions
		common.MemoryCacheEnabled = true
	}
	if common.MemoryCacheEnabled {
		common.SysLog("memory cache enabled")
		common.SysLog(fmt.Sprintf("sync frequency: %d seconds", common.SyncFrequency))

		// Add panic recovery and retry for InitChannelCache
		func() {
			defer func() {
				if r := recover(); r != nil {
					common.SysLog(fmt.Sprintf("InitChannelCache panic: %v, retrying once", r))
					// Retry once
					_, _, fixErr := model.FixAbility()
					if fixErr != nil {
						common.FatalLog(fmt.Sprintf("InitChannelCache failed: %s", fixErr.Error()))
					}
				}
			}()
			model.InitChannelCache()
		}()

		go model.SyncChannelCache(common.SyncFrequency)
	}

	// 热更新配置
	go model.SyncOptions(common.SyncFrequency)

	// 周期性重载授权策略，保证多节点/多 master 部署下权限变更能传播到每个实例
	go authz.StartPolicySync(common.SyncFrequency)

	// 数据看板
	if !backgroundOnly {
		go model.UpdateQuotaData()
	}

	if !backgroundOnly && os.Getenv("CHANNEL_UPDATE_FREQUENCY") != "" {
		frequency, err := strconv.Atoi(os.Getenv("CHANNEL_UPDATE_FREQUENCY"))
		if err != nil {
			common.FatalLog("failed to parse CHANNEL_UPDATE_FREQUENCY: " + err.Error())
		}
		go controller.AutomaticallyUpdateChannels(frequency)
	}

	// Codex credential auto-refresh check every 10 minutes, refresh when expires within 1 day
	if !backgroundOnly {
		service.StartCodexCredentialAutoRefreshTask()

		// Subscription quota reset task (daily/weekly/monthly/custom)
		service.StartSubscriptionQuotaResetTask()
	}

	// Report this process as a system instance so the System Info page can show
	// all currently alive nodes in multi-instance deployments.
	service.StartSystemInstanceReporter()

	// Wire task polling adaptor factory (breaks service -> relay import cycle).
	// Must run before the system task runner starts: the async_task_poll handler
	// calls service.RunTaskPollingOnce, which needs this factory set.
	if !backgroundOnly {
		service.GetTaskAdaptorFunc = func(platform constant.TaskPlatform) service.TaskPollingAdaptor {
			if platform == constant.TaskPlatformOpenAIImage {
				return &service.ImageTaskAdaptor{}
			}
			a := relay.GetTaskAdaptor(platform)
			if a == nil {
				return nil
			}
			return a
		}

		// Register the periodic channel test, upstream model update, and async task
		// polling (Midjourney / Suno / video) jobs as scheduled system tasks
		// (DB-lease dedup across masters + run history), then start the runner that
		// schedules and executes them. Master-only execution and the UpdateTask
		// switch are enforced inside the runner and each handler's Enabled().
		controller.RegisterScheduledSystemTasks()
		service.StartSystemTaskRunner()
	}

	if !backgroundOnly && os.Getenv("BATCH_UPDATE_ENABLED") == "true" {
		common.BatchUpdateEnabled = true
		common.SysLog("batch update enabled with interval " + strconv.Itoa(common.BatchUpdateInterval) + "s")
		model.InitBatchUpdater()
	}

	// Start after BATCH_UPDATE_ENABLED is resolved so strict accounting cannot
	// observe the startup default during reconciliation.
	service.StartSmartRouteReconciliationTask()

	if !backgroundOnly && os.Getenv("ENABLE_PPROF") == "true" {
		gopool.Go(func() {
			log.Println(http.ListenAndServe("0.0.0.0:8005", nil))
		})
		go common.Monitor()
		common.SysLog("pprof enabled")
	}

	if !backgroundOnly {
		err = common.StartPyroScope()
		if err != nil {
			common.SysError(fmt.Sprintf("start pyroscope error : %v", err))
		}
	}

	var probeWorker *service.SmartRouteProbeWorker
	var routeAnalyticsWorker *service.RouteAnalyticsWorkerV1
	if workerOnly || analyticsWorkerOnly || processRole == "all" {
		engine, engineErr := service.GetDefaultSmartRouterEngine()
		if engineErr != nil {
			common.FatalLog("failed to initialize smart router worker: " + engineErr.Error())
			return
		}
		var workerErr error
		if !analyticsWorkerOnly {
			probeWorker, workerErr = service.NewSmartRouteProbeWorker(
				engine,
				controller.NewSmartRouteRelayProbeExecutor(),
			)
			if workerErr != nil {
				common.FatalLog("failed to initialize smart router probe worker: " + workerErr.Error())
				return
			}
			probeWorker.Start()
		}
		if service.RouteAnalyticsEnabledV1() {
			routeAnalyticsWorker, workerErr = service.NewRouteAnalyticsWorkerV1(engine)
			if workerErr != nil {
				common.SysError("route analytics worker remains disabled after initialization failure: " + workerErr.Error())
				routeAnalyticsWorker = nil
			} else {
				routeAnalyticsWorker.Start()
			}
		}
	}
	if backgroundOnly {
		waitForSmartRouterWorker(probeWorker, routeAnalyticsWorker)
		return
	}
	if probeWorker != nil {
		defer probeWorker.Close()
	}
	if routeAnalyticsWorker != nil {
		defer routeAnalyticsWorker.Close()
	}

	// Initialize HTTP server
	server := gin.New()
	server.Use(gin.CustomRecovery(func(c *gin.Context, err any) {
		common.SysLog(fmt.Sprintf("panic detected: %v", err))
		c.JSON(http.StatusInternalServerError, gin.H{
			"error": gin.H{
				"message": fmt.Sprintf("Panic detected, error: %v. Please submit a issue here: https://github.com/Calcium-Ion/new-api", err),
				"type":    "new_api_panic",
			},
		})
	}))
	// This will cause SSE not to work!!!
	//server.Use(gzip.Gzip(gzip.DefaultCompression))
	server.Use(middleware.RequestId())
	server.Use(middleware.Version())
	server.Use(middleware.I18n())
	middleware.SetUpLogger(server)
	// Initialize session store
	store := cookie.NewStore([]byte(common.SessionSecret))
	store.Options(sessions.Options{
		Path:     "/",
		MaxAge:   2592000, // 30 days
		HttpOnly: true,
		Secure:   common.SessionCookieSecure,
		SameSite: http.SameSiteStrictMode,
	})
	server.Use(sessions.Sessions("session", store))

	InjectUmamiAnalytics()
	InjectGoogleAnalytics()

	// 设置路由
	router.SetRouter(server, router.ThemeAssets{
		DefaultBuildFS:   buildFS,
		DefaultIndexPage: indexPage,
		ClassicBuildFS:   classicBuildFS,
		ClassicIndexPage: classicIndexPage,
	})
	var port = os.Getenv("PORT")
	if port == "" {
		port = strconv.Itoa(*common.Port)
	}

	srv := &http.Server{
		Addr:    ":" + port,
		Handler: server,
	}

	go func() {
		if err := srv.ListenAndServe(); err != nil && !errors.Is(err, http.ErrServerClosed) {
			common.FatalLog("failed to start HTTP server: " + err.Error())
		}
	}()

	time.Sleep(100 * time.Millisecond)

	common.LogStartupSuccess(startTime, port)

	quit := make(chan os.Signal, 1)
	signal.Notify(quit, syscall.SIGINT, syscall.SIGTERM)
	sig := <-quit
	common.SysLog(fmt.Sprintf("received signal: %v, shutting down...", sig))

	// SSE streams may run for minutes; give them time to finish before forced exit
	shutdownTimeout := time.Duration(common.GetEnvOrDefault("SHUTDOWN_TIMEOUT_SECONDS", 120)) * time.Second
	ctx, cancel := context.WithTimeout(context.Background(), shutdownTimeout)
	defer cancel()
	if err := srv.Shutdown(ctx); err != nil {
		common.SysError(fmt.Sprintf("server forced to shutdown: %v", err))
	}
	// 内存中的看板数据保存入库，避免重启丢失未落库数据 (issue #5679)
	if common.DataExportEnabled {
		model.SaveQuotaDataCache()
	}
	common.SysLog("server exited")
}

func smartRouterProcessRole() (string, error) {
	role := strings.ToLower(strings.TrimSpace(os.Getenv("PROCESS_ROLE")))
	if role == "" {
		return "web", nil
	}
	switch role {
	case "web", "smart-router-worker", "route-analytics-worker", "commission-worker", "all":
		return role, nil
	default:
		return "", fmt.Errorf("unsupported PROCESS_ROLE %q (expected web, smart-router-worker, route-analytics-worker, commission-worker, or all)", role)
	}
}

func waitForSmartRouterWorker(
	probeWorker *service.SmartRouteProbeWorker,
	routeAnalyticsWorker *service.RouteAnalyticsWorkerV1,
) {
	quit := make(chan os.Signal, 1)
	signal.Notify(quit, syscall.SIGINT, syscall.SIGTERM)
	defer signal.Stop(quit)
	sig := <-quit
	common.SysLog(fmt.Sprintf("received signal: %v, shutting down smart router worker...", sig))
	if routeAnalyticsWorker != nil {
		routeAnalyticsWorker.Close()
	}
	if probeWorker != nil {
		probeWorker.Close()
	}
	common.SysLog("smart router worker exited")
}

func waitForCommissionSidecarWorker(worker *service.CommissionSidecarWorker) {
	quit := make(chan os.Signal, 1)
	signal.Notify(quit, syscall.SIGINT, syscall.SIGTERM)
	defer signal.Stop(quit)
	sig := <-quit
	common.SysLog(fmt.Sprintf("received signal: %v, shutting down commission sidecar worker...", sig))
	worker.Close()
	common.SysLog("commission sidecar worker exited")
}

func InjectUmamiAnalytics() {
	analyticsInjectBuilder := &strings.Builder{}
	if os.Getenv("UMAMI_WEBSITE_ID") != "" {
		umamiSiteID := os.Getenv("UMAMI_WEBSITE_ID")
		umamiScriptURL := os.Getenv("UMAMI_SCRIPT_URL")
		if umamiScriptURL == "" {
			umamiScriptURL = "https://analytics.umami.is/script.js"
		}
		analyticsInjectBuilder.WriteString("<script defer src=\"")
		analyticsInjectBuilder.WriteString(umamiScriptURL)
		analyticsInjectBuilder.WriteString("\" data-website-id=\"")
		analyticsInjectBuilder.WriteString(umamiSiteID)
		analyticsInjectBuilder.WriteString("\"></script>")
	}
	analyticsInjectBuilder.WriteString("<!--Umami QuantumNous-->\n")
	analyticsInject := []byte(analyticsInjectBuilder.String())
	placeholder := []byte("<!--umami-->\n")
	indexPage = bytes.ReplaceAll(indexPage, placeholder, analyticsInject)
	classicIndexPage = bytes.ReplaceAll(classicIndexPage, placeholder, analyticsInject)
}

func InjectGoogleAnalytics() {
	analyticsInjectBuilder := &strings.Builder{}
	if os.Getenv("GOOGLE_ANALYTICS_ID") != "" {
		gaID := os.Getenv("GOOGLE_ANALYTICS_ID")
		// Google Analytics 4 (gtag.js)
		analyticsInjectBuilder.WriteString("<script async src=\"https://www.googletagmanager.com/gtag/js?id=")
		analyticsInjectBuilder.WriteString(gaID)
		analyticsInjectBuilder.WriteString("\"></script>")
		analyticsInjectBuilder.WriteString("<script>")
		analyticsInjectBuilder.WriteString("window.dataLayer = window.dataLayer || [];")
		analyticsInjectBuilder.WriteString("function gtag(){dataLayer.push(arguments);}")
		analyticsInjectBuilder.WriteString("gtag('js', new Date());")
		analyticsInjectBuilder.WriteString("gtag('config', '")
		analyticsInjectBuilder.WriteString(gaID)
		analyticsInjectBuilder.WriteString("');")
		analyticsInjectBuilder.WriteString("</script>")
	}
	analyticsInjectBuilder.WriteString("<!--Google Analytics QuantumNous-->\n")
	analyticsInject := []byte(analyticsInjectBuilder.String())
	placeholder := []byte("<!--Google Analytics-->\n")
	indexPage = bytes.ReplaceAll(indexPage, placeholder, analyticsInject)
	classicIndexPage = bytes.ReplaceAll(classicIndexPage, placeholder, analyticsInject)
}

func migrationOnlyRequested() bool {
	enabled, err := strconv.ParseBool(strings.TrimSpace(os.Getenv("MIGRATION_ONLY")))
	return err == nil && enabled
}

func InitResources() error {
	// Initialize resources here if needed
	// This is a placeholder function for future resource initialization
	err := godotenv.Load(".env")
	if err != nil {
		if common.DebugEnabled {
			common.SysLog("No .env file found, using default environment variables. If needed, please create a .env file and set the relevant variables.")
		}
	}

	// 加载环境变量
	common.InitEnv()

	logger.SetupLogger()

	// Initialize model settings
	ratio_setting.InitRatioSettings()

	service.InitHttpClient()

	service.InitTokenEncoders()

	// Database startup reconciles tenant bindings and invalidates token caches.
	// Establish the configured cache (including explicitly disabled Redis)
	// before either normal startup or migration-only performs reconciliation.
	if err = common.InitRedisClient(); err != nil {
		return err
	}

	// Initialize SQL Database
	err = model.InitDB()
	if err != nil {
		common.FatalLog("failed to initialize database: " + err.Error())
		return err
	}
	if err = model.EnsureRouteAnalyticsSchema(model.DB); err != nil {
		common.FatalLog("failed to initialize route analytics schema: " + err.Error())
		return err
	}
	if err = model.EnsureMediaStorageSchema(model.DB); err != nil {
		common.FatalLog("failed to initialize media storage schema: " + err.Error())
		return err
	}
	if err = authz.Init(model.DB); err != nil {
		common.FatalLog("failed to initialize authorization: " + err.Error())
		return err
	}

	model.CheckSetup()

	// Initialize options, should after model.InitDB()
	model.InitOptionMap()

	// 清理旧的磁盘缓存文件
	common.CleanupOldCacheFiles()

	// 初始化模型
	model.GetPricing()

	// Initialize SQL Database
	err = model.InitLogDB()
	if err != nil {
		return err
	}

	perfmetrics.Init()

	// 启动系统监控
	common.StartSystemMonitor()

	// Initialize i18n
	err = i18n.Init()
	if err != nil {
		common.SysError("failed to initialize i18n: " + err.Error())
		// Don't return error, i18n is not critical
	} else {
		common.SysLog("i18n initialized with languages: " + strings.Join(i18n.SupportedLanguages(), ", "))
	}
	// Register user language loader for lazy loading
	i18n.SetUserLangLoader(model.GetUserLanguage)

	// Load custom OAuth providers from database
	err = oauth.LoadCustomProviders()
	if err != nil {
		common.SysError("failed to load custom OAuth providers: " + err.Error())
		// Don't return error, custom OAuth is not critical
	}

	return nil
}
