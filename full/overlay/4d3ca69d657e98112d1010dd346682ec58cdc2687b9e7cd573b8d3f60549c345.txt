func GetAllLogs(c *gin.Context) {
	pageInfo, pageErr := logPageQuery(c)
	if pageErr != nil {
		common.ApiError(c, pageErr)
		return
	}
	logType, _ := strconv.Atoi(c.Query("type"))
	startTimestamp, _ := strconv.ParseInt(c.Query("start_timestamp"), 10, 64)
	endTimestamp, _ := strconv.ParseInt(c.Query("end_timestamp"), 10, 64)
	username := c.Query("username")
	tokenName := c.Query("token_name")
	modelName := c.Query("model_name")
	channel, _ := strconv.Atoi(c.Query("channel"))
	group := c.Query("group")
	requestId := c.Query("request_id")
	upstreamRequestId := c.Query("upstream_request_id")
	logs, total, err := model.GetAllLogs(c.Request.Context(), logType, startTimestamp, endTimestamp, modelName, username, tokenName, pageInfo.GetStartIdx(), pageInfo.GetPageSize(), channel, group, requestId, upstreamRequestId)
	if err != nil {
		logQueryError(c, err)
		return
	}
	pageInfo.SetTotal(int(min(total, model.LogSearchCountLimit)))
	pageInfo.SetItems(logs)
	common.ApiSuccess(c, struct {
		*common.PageInfo
		TotalCapped bool `json:"total_capped"`
		QueryLimit  int  `json:"query_limit"`
	}{pageInfo, total > model.LogSearchCountLimit, model.LogSearchCountLimit})
	return
}

// DeleteHistoryLogs is the legacy synchronous log cleanup endpoint (DELETE /api/log/).
// It deletes directly instead of going through the async system task. It is kept only
// for the classic frontend; the default frontend uses POST /api/system-task/log-cleanup.
// TODO: remove this handler (and its route) once the classic frontend is removed.
func DeleteHistoryLogs(c *gin.Context) {
	targetTimestamp, _ := strconv.ParseInt(c.Query("target_timestamp"), 10, 64)
	if targetTimestamp == 0 {
		c.JSON(http.StatusOK, gin.H{
			"success": false,
			"message": "target timestamp is required",
		})
		return
	}
	count, err := model.DeleteOldLog(c.Request.Context(), targetTimestamp, 100)
	if err != nil {
		common.ApiError(c, err)
		return
	}
	c.JSON(http.StatusOK, gin.H{
		"success": true,
		"message": "",
		"data":    count,
	})
	return
}

func logPageQuery(c *gin.Context) (*common.PageInfo, error) {
	page := common.GetPageQuery(c)
	if raw := c.Query("p"); raw != "" {
		value, err := strconv.Atoi(raw)
		if err != nil || value < 1 {
			return nil, model.ErrLogPagination
		}
		page.Page = value
	}
	for _, key := range []string{"page_size", "ps", "size"} {
		if raw := c.Query(key); raw != "" {
			value, err := strconv.Atoi(raw)
			if err != nil || value < 1 || value > 100 {
				return nil, model.ErrLogPagination
			}
			break
		}
	}
	if page.PageSize < 1 || page.PageSize > 100 || page.Page > (model.LogSearchCountLimit-1)/page.PageSize+1 {
		return nil, model.ErrLogPagination
	}
	return page, nil
}

func logQueryError(c *gin.Context, err error) {
	if errors.Is(err, model.ErrLogPagination) {
		common.ApiError(c, err)
		return
	}
	message := "查询日志失败，请稍后重试"
	if errors.Is(err, context.DeadlineExceeded) {
		message = "日志查询超时，请缩小时间或筛选范围后重试"
	} else if errors.Is(err, context.Canceled) {
		message = "日志查询已取消"
	}
	logger.LogWarn(c.Request.Context(), fmt.Sprintf("log query failed: %v", err))
	common.ApiError(c, errors.New(common.MessageWithRequestId(message, c.GetString(common.RequestIdKey))))
}
