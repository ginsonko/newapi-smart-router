import { z } from 'zod'

import { STUDIO_BATCH_MAX_COUNT } from './lib'

export const studioSchema = z
  .object({
    mode: z.enum([
      'images',
      'image-edits',
      'videos',
      'speech',
      'transcriptions',
      'translations',
    ]),
    group: z.string().min(1, 'Select a group'),
    model: z.string().min(1, 'Select a model'),
    prompt: z.string(),
    size: z.string(),
    aspectRatio: z.string(),
    resolution: z.string(),
    duration: z.number().int().min(1).max(3600),
    count: z.number().int().min(1).max(STUDIO_BATCH_MAX_COUNT),
    quality: z.string(),
    voice: z.string(),
    responseFormat: z.string(),
  })
  .superRefine((value, context) => {
    if (
      ['images', 'image-edits', 'videos', 'speech'].includes(value.mode) &&
      value.prompt.trim().length === 0
    ) {
      context.addIssue({
        code: 'custom',
        path: ['prompt'],
        message: 'Prompt is required',
      })
    }
    if (value.mode === 'videos' && value.aspectRatio.trim().length === 0) {
      context.addIssue({
        code: 'custom',
        path: ['aspectRatio'],
        message: 'Aspect ratio is required',
      })
    }
  })

export type StudioSchemaValue = z.infer<typeof studioSchema>
