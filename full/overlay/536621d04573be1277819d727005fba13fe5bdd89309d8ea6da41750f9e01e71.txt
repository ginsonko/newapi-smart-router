type mediaCancellationTransport func(*http.Request) (*http.Response, error)

func (f mediaCancellationTransport) RoundTrip(r *http.Request) (*http.Response, error) { return f(r) }

type mediaCancellationAdaptor struct {
	setup func()
}

func (a *mediaCancellationAdaptor) GetRequestURL(*relaycommon.RelayInfo) (string, error) {
	return "https://media-cancellation.invalid/generate", nil
}
func (a *mediaCancellationAdaptor) BuildRequestURL(info *relaycommon.RelayInfo) (string, error) {
	return a.GetRequestURL(info)
}
func (a *mediaCancellationAdaptor) SetupRequestHeader(*gin.Context, *http.Header, *relaycommon.RelayInfo) error {
	if a.setup != nil {
		a.setup()
	}
	return nil
}
func (a *mediaCancellationAdaptor) BuildRequestHeader(*gin.Context, *http.Request, *relaycommon.RelayInfo) error {
	if a.setup != nil {
		a.setup()
	}
	return nil
}

// Separate embeddings avoid ambiguous methods shared by the two interfaces.
type mediaCancellationAPIAdaptor struct {
	Adaptor
	delegate *mediaCancellationAdaptor
}

func (a *mediaCancellationAPIAdaptor) GetRequestURL(info *relaycommon.RelayInfo) (string, error) {
	return a.delegate.GetRequestURL(info)
}
func (a *mediaCancellationAPIAdaptor) SetupRequestHeader(c *gin.Context, h *http.Header, info *relaycommon.RelayInfo) error {
	return a.delegate.SetupRequestHeader(c, h, info)
}

type mediaCancellationTaskAdaptor struct {
	TaskAdaptor
	delegate *mediaCancellationAdaptor
}

func (a *mediaCancellationTaskAdaptor) BuildRequestURL(info *relaycommon.RelayInfo) (string, error) {
	return a.delegate.BuildRequestURL(info)
}
func (a *mediaCancellationTaskAdaptor) BuildRequestHeader(c *gin.Context, r *http.Request, info *relaycommon.RelayInfo) error {
	return a.delegate.BuildRequestHeader(c, r, info)
}

func TestMediaCancellationAtHTTPDispatch(t *testing.T) {
	gin.SetMode(gin.TestMode)
	if service.GetHttpClient() == nil {
		service.InitHttpClient()
	}
	client := service.GetHttpClient()
	previous := *client
	t.Cleanup(func() { *client = previous })
	for _, kind := range []string{"image_json", "image_form", "task_worker", "text"} {
		for _, phase := range []string{"before_headers", "during_headers", "during_transport", "active"} {
			t.Run(kind+"/"+phase, func(t *testing.T) {
				original, cancelOriginal := context.WithCancel(context.Background())
				cancelOriginal()
				require.ErrorIs(t, original.Err(), context.Canceled)
				// Like SubmitMediaAsyncTask, construct this request using the live
				// execution context, independently of the ended frontend request.
				execution, cancel := context.WithTimeout(context.Background(), time.Minute)
				defer cancel()
				c, _ := gin.CreateTestContext(httptest.NewRecorder())
				c.Request = httptest.NewRequest(http.MethodPost, "/v1/videos", nil).WithContext(execution)
				info := &relaycommon.RelayInfo{ChannelMeta: &relaycommon.ChannelMeta{}}
				switch kind {
				case "image_json":
					info.RelayMode = relayconstant.RelayModeImagesGenerations
				case "image_form":
					info.RelayMode = relayconstant.RelayModeImagesEdits
				case "task_worker":
					info.TaskRelayInfo = &relaycommon.TaskRelayInfo{PublicTaskID: "task-worker"}
				case "text":
					info.RelayMode = relayconstant.RelayModeChatCompletions
				}
				adaptor := &mediaCancellationAdaptor{}
				if phase == "before_headers" {
					cancel()
				}
				if phase == "during_headers" {
					adaptor.setup = cancel
				}
				calls := 0
				client.Timeout = 0
				client.Transport = mediaCancellationTransport(func(req *http.Request) (*http.Response, error) {
					calls++
					if kind == "text" {
						require.NoError(t, req.Context().Err(), "legacy text execution must stay unchanged")
						_, hasDeadline := req.Context().Deadline()
						require.False(t, hasDeadline)
					} else {
						require.True(t, execution == req.Context(), "media request must retain its execution context and deadline")
						if phase == "during_transport" {
							cancel()
						}
						if err := req.Context().Err(); err != nil {
							_ = req.Body.Close()
							return nil, err
						}
					}
					_ = req.Body.Close()
					return &http.Response{StatusCode: http.StatusOK, Header: http.Header{}, Body: io.NopCloser(strings.NewReader("{}")), Request: req}, nil
				})
				body := &trackingReadCloser{Reader: bytes.NewReader([]byte("payload"))}
				var response *http.Response
				var err error
				if kind == "task_worker" {
					response, err = DoTaskApiRequest(&mediaCancellationTaskAdaptor{delegate: adaptor}, c, info, body)
				} else if kind == "image_form" {
					response, err = DoFormRequest(&mediaCancellationAPIAdaptor{delegate: adaptor}, c, info, body)
				} else {
					response, err = DoApiRequest(&mediaCancellationAPIAdaptor{delegate: adaptor}, c, info, body)
				}
				if kind != "text" && phase != "active" {
					require.Error(t, err)
					require.True(t, errors.Is(err, context.Canceled), "cancellation cause must survive: %v", err)
					if phase == "during_transport" {
						require.Equal(t, 1, calls)
					} else {
						require.Zero(t, calls)
					}
				} else {
					require.NoError(t, err)
					require.Equal(t, 1, calls)
				}
				require.True(t, body.closed, "unsent bodies must be released")
				if response != nil {
					_ = response.Body.Close()
				}
			})
		}
	}
}
